bool CompoundCommand::execute() {
    bool success = true;
    size_t executedCommands = 0;

    for (size_t i = 0; i < m_commands.size(); ++i) {
        if (!m_commands[i]->execute()) {
            success = false;
            executedCommands = i;
            break;
        }
        executedCommands = i + 1;
    }

    if (!success) {
        for (size_t i = 0; i < executedCommands; ++i) {
            size_t index = executedCommands - i - 1;
            m_commands[index]->undo();
        }
    }

    return success;
}

bool CompoundCommand::undo() {
    bool success = true;

    for (const auto& m_command : std::ranges::reverse_view(m_commands))
        if (!m_command->undo()) {
            success = false;
            break;
        }

    return success;
}

bool CompoundCommand::redo() {
    bool success = true;

    for (const auto& command : m_commands)
        if (!command->redo()) {
            success = false;
            break;
        }

    return success;
}

AddControlPointCommand::AddControlPointCommand(std::shared_ptr<Curve> curve, const Point& point, int index)
    : CurveCommand(curve), m_point(point), m_index(index), m_addedIndex(-1) {
}

bool AddControlPointCommand::execute() {
    if (!m_curve || !m_curve->hasControlPoints())
        return false;

    if (m_index < 0 || m_index > static_cast<int>(m_curve->getControlPointCount())) {
        m_curve->addControlPoint(m_point);
        m_addedIndex = static_cast<int>(m_curve->getControlPointCount()) - 1;
    } else {
        m_curve->insertControlPoint(m_index, m_point);
        m_addedIndex = m_index;
    }

    return true;
}

bool AddControlPointCommand::undo() {
    if (!m_curve || m_addedIndex < 0 ||
        m_addedIndex >= static_cast<int>(m_curve->getControlPointCount())) {
        return false;
    }

    m_curve->removeControlPoint(m_addedIndex);
    return true;
}

        return false;

    if (m_pointIndex >= static_cast<int>(m_curve->getControlPointCount()))
        m_curve->addControlPoint(m_deletedPoint);

        m_curve->insertControlPoint(m_pointIndex, m_deletedPoint);

    return true;
}

bool CommandManager::executeCommand(std::unique_ptr<AbstractCommand> command) {
    if (!command)
        return false;

    if (const bool success = command->execute(); !success)
        return false;

    if (!m_redoStack.empty())
        m_redoStack.clear();

    m_undoStack.push_back(std::move(command));
    return true;
}

bool CommandManager::canUndo() const {
    return !m_undoStack.empty();
}

bool CommandManager::canRedo() const {
    return !m_redoStack.empty();
}

bool CommandManager::undo() {
    if (!canUndo())
        return false;

    auto command = std::move(m_undoStack.back());
    m_undoStack.pop_back();

    const bool success = command->undo();

    m_redoStack.push_back(std::move(command));
    return success;
}

bool CommandManager::redo() {
    if (!canRedo())
        return false;

    auto command = std::move(m_redoStack.back());
    m_redoStack.pop_back();

    const bool success = command->redo();

    m_undoStack.push_back(std::move(command));
    return success;
}

void CommandManager::clearHistory() {
    m_undoStack.clear();
    m_redoStack.clear();
}

int CommandManager::undoStackSize() const {
    return static_cast<int>(m_undoStack.size());
}

int CommandManager::redoStackSize() const {
    return static_cast<int>(m_redoStack.size());
}

BezierCurve::BezierCurve() : Curve(CurveType::Bezier) {
}

int BezierCurve::getDegree() const {
    return static_cast<int>(m_controlPoints.size()) - 1;
}

std::unique_ptr<Curve> BezierCurve::createClone() const {
    auto clone = std::make_unique<BezierCurve>();
    return clone;
}

void BezierCurve::copySpecificPropertiesTo(Curve* target) const {
    if (auto bezierTarget = dynamic_cast<BezierCurve*>(target)) {
        bezierTarget->m_weights = m_weights;
        bezierTarget->m_showWeights = m_showWeights;
    }
}

Point BezierCurve::getPointAt(double t) const {
    validateParameterRange(t);

    if (hasWeights() && !std::ranges::all_of(m_weights, [this](double w){ return std::abs(w - m_weights[0]) < 1e-9; }))
        return rationalDeCasteljau(t);
    return deCasteljau(t);
}

void BezierCurve::setWeights(const std::vector<double>& weights) {
    if (weights.size() != m_controlPoints.size())
        throw std::invalid_argument("Liczba wag musi być równa liczbie punktów kontrolnych");

    if (std::any_of(weights.begin(), weights.end(), [](double w) { return w <= 0.0; }))
        throw std::invalid_argument("Wszystkie wagi muszą być dodatnie");

    m_weights = weights;
    onControlPointsChanged();
}

const std::vector<double>& BezierCurve::getWeights() const {
    return m_weights;
}

void BezierCurve::setWeight(size_t index, double weight) {
    if (index >= m_controlPoints.size())
        throw std::out_of_range("Indeks poza zakresem");

    if (weight <= 0.0)
        throw std::invalid_argument("Waga musi być dodatnia");

    ensureWeightsSize();
    m_weights[index] = weight;
    onControlPointsChanged();
}

double BezierCurve::getWeight(size_t index) const {
    if (index >= m_controlPoints.size())
        throw std::out_of_range("Indeks poza zakresem");

    if (hasWeights() && index < m_weights.size())
        return m_weights[index];

    return 1.0;
}

bool BezierCurve::hasWeights() const {
    return !m_weights.empty() && m_weights.size() == m_controlPoints.size();
}

void BezierCurve::resetWeights() {
    if (m_weights.empty())
        ensureWeightsSize();

    std::ranges::fill(m_weights, 1.0);
    onControlPointsChanged();
}

void BezierCurve::normalizeWeights() {
    if (m_weights.empty()) return;

    if (const double sum = std::accumulate(m_weights.begin(), m_weights.end(), 0.0); sum > 0.0) {
        const double factor = static_cast<double>(m_weights.size()) / sum;
        for (double& weight : m_weights)
            weight *= factor;

        onControlPointsChanged();
    }
}

void BezierCurve::addControlPoint(const Point& point) {
    Curve::addControlPoint(point);

    if (!m_weights.empty()) m_weights.push_back(1.0);
}

void BezierCurve::insertControlPoint(const size_t index, const Point& point) {
    if (index <= m_controlPoints.size()) {
        Curve::insertControlPoint(index, point);

        if (!m_weights.empty() && index < m_weights.size())
            m_weights.insert(m_weights.begin() + index, 1.0);
        else if (!m_weights.empty())
            m_weights.push_back(1.0);
    }
}

void BezierCurve::removeControlPoint(size_t index) {
    if (index < m_controlPoints.size()) {
        Curve::removeControlPoint(index);

        if (!m_weights.empty() && index < m_weights.size())
            m_weights.erase(m_weights.begin() + index);
    }
}

void BezierCurve::setControlPoints(const std::vector<Point>& points) {
    Curve::setControlPoints(points);

    if (!m_weights.empty())
        m_weights.resize(points.size(), 1.0);
}

Point BezierCurve::deCasteljau(const double t) const {
    if (m_controlPoints.empty())
        return Point(0, 0, 0);

    if (m_controlPoints.size() == 1)
        return m_controlPoints[0];

    std::vector<Point> temp = m_controlPoints;

    for (size_t level = 1; level < m_controlPoints.size(); ++level)
        for (size_t i = 0; i < m_controlPoints.size() - level; ++i)
            temp[i] = interpolateLinear(temp[i], temp[i + 1], t);

    return temp[0];
}

Point BezierCurve::rationalDeCasteljau(double t) const {
    if (m_controlPoints.empty())
        return Point(0, 0, 0);

    if (m_controlPoints.size() == 1)
        return m_controlPoints[0];

    validateWeights();

    const size_t n = m_controlPoints.size();
    std::vector<Point> tempPoints = m_controlPoints;
    std::vector<double> tempWeights = m_weights;

    for (size_t level = 1; level < n; ++level) {
        for (size_t i = 0; i < n - level; ++i) {
            const double newWeight = (1.0 - t) * tempWeights[i] + t * tempWeights[i + 1];

            const double w1 = (1.0 - t) * tempWeights[i];
            const double w2 = t * tempWeights[i + 1];

            const Point& p1 = tempPoints[i];
            const Point& p2 = tempPoints[i + 1];

            const double invNewWeight = 1.0 / newWeight;
            tempPoints[i] = Point(
                (w1 * p1.x() + w2 * p2.x()) * invNewWeight,
                (w1 * p1.y() + w2 * p2.y()) * invNewWeight,
                (w1 * p1.z() + w2 * p2.z()) * invNewWeight
            );

            tempWeights[i] = newWeight;
        }
    }

    return tempPoints[0];
}

Vector BezierCurve::deCasteljauDerivative(double t) const {
    if (m_controlPoints.size() < 2)
        return Vector::Zero();

    std::vector<Point> derivativePoints;
    const int n = static_cast<int>(m_controlPoints.size()) - 1;

    for (size_t i = 0; i < m_controlPoints.size() - 1; ++i) {
        double dx = n * (m_controlPoints[i + 1].x() - m_controlPoints[i].x());
        double dy = n * (m_controlPoints[i + 1].y() - m_controlPoints[i].y());
        double dz = n * (m_controlPoints[i + 1].z() - m_controlPoints[i].z());
        derivativePoints.emplace_back(dx, dy, dz);
    }

    for (size_t level = 1; level < derivativePoints.size(); ++level)
        for (size_t i = 0; i < derivativePoints.size() - level; ++i) {
            derivativePoints[i] = interpolateLinear(derivativePoints[i], derivativePoints[i + 1], t);
        }

    if (derivativePoints.empty())
        return Vector::Zero();

    return Vector(derivativePoints[0].x(), derivativePoints[0].y(), derivativePoints[0].z());
}

Vector BezierCurve::rationalDeCasteljauDerivative(double t) const {
    if (m_controlPoints.size() < 2)
        return Vector::Zero();

    validateWeights();

    const double epsilon = 1e-6;
    const double t1 = std::max(0.0, t - epsilon);
    const double t2 = std::min(1.0, t + epsilon);

    const Point p1 = rationalDeCasteljau(t1);
    const Point p2 = rationalDeCasteljau(t2);
    const double dt = t2 - t1;

    return Vector(
        (p2.x() - p1.x()) / dt,
        (p2.y() - p1.y()) / dt,
        (p2.z() - p1.z()) / dt
    );
}

void BezierCurve::ensureWeightsSize() {
    if (m_weights.size() != m_controlPoints.size())
        m_weights.resize(m_controlPoints.size(), 1.0);
}

void BezierCurve::validateWeights() const {
    if (m_weights.size() != m_controlPoints.size())
        throw std::runtime_error("Rozmiar wektorów wag i punktów kontrolnych nie jest zgodny");

    if (std::ranges::any_of(m_weights, [](double w) { return w <= 0.0; }))
        throw std::runtime_error("Wszystkie wagi muszą być dodatnie");
}

void BezierCurve::setShowWeights(bool show) {
    m_showWeights = show;

    if (show && !hasWeights() && !m_controlPoints.empty()) {
        ensureWeightsSize();
        resetWeights();
    }
}

Vector Curve::getPosition() const {
    if (!hasControlPoints() || getControlPointCount() == 0)
        return Vector::Zero();

    const auto& points = getControlPoints();
    Vector sum = Vector::Zero();
    for (const auto& point : points)
        sum += Vector(point.x(), point.y(), point.z());

    return sum / static_cast<double>(points.size());
}

void Curve::setPosition(const Eigen::Vector3d& position) {
    if (!hasControlPoints() || getControlPointCount() == 0)
        return;

    const Vector currentCenter = getPosition();
    Vector delta = position - currentCenter;

    auto points = getControlPoints();
    for (auto& point : points) {
        point.setX(point.x() + delta.x());
        point.setY(point.y() + delta.y());
        point.setZ(point.z() + delta.z());
    }
    setControlPoints(points);
}

void Curve::transform(const TransformMatrix& matrix) {
    if (!hasControlPoints())
        return;

    auto points = getControlPoints();
    for (auto& point : points)
        point.transform(matrix);
    setControlPoints(points);
}

double Curve::calculateLengthByDiscretization(int numSamples) const {
    if (m_controlPoints.size() < 2)
        return 0.0;

    if (numSamples <= 0)
        numSamples = getDefaultLengthSamples();

    double length = 0.0;
    Point prevPoint = getPointAt(0.0);

    for (int i = 1; i <= numSamples; ++i) {
        double t = static_cast<double>(i) / numSamples;
        Point currentPoint = getPointAt(t);
        length += prevPoint.distanceTo(currentPoint);
        prevPoint = currentPoint;
    }

    return length;
}

bool Curve::validateParameterRange(double& t) const {
    t = std::max(0.0, std::min(1.0, t));
    return t >= 0.0 && t <= 1.0;
}

Point Curve::interpolateLinear(const Point& p1, const Point& p2, double t) const {
    const double oneMinusT = 1.0 - t;
    return Point(
        oneMinusT * p1.x() + t * p2.x(),
        oneMinusT * p1.y() + t * p2.y(),
        oneMinusT * p1.z() + t * p2.z()
    );
}

double Curve::calculateParameterForSegment(size_t segmentIndex, size_t totalSegments, double localT) const {
    double segmentLength = 1.0 / totalSegments;
    return segmentIndex * segmentLength + localT * segmentLength;
}

void InterpolatedCurve::rebuildInterpolation() const {
    if (!needsRebuild || m_controlPoints.size() < 2) {
        needsRebuild = false;
        return;
    }
    coefficients.clear();

    switch (type) {
        case InterpolationType::NIFS3:
            buildNIFS3();
        break;
        case InterpolationType::OIFS3:
            buildOIFS3();
        break;
        case InterpolationType::Lagrange:
            buildLagrange();
        break;
        case InterpolationType::Barycentric:
            buildBarycentric();
        break;
    }

    needsRebuild = false;
}

Point InterpolatedCurve::getPointAt(double t) const {
    if (m_controlPoints.empty()) return Point(0.0, 0.0, 0.0);
    if (m_controlPoints.size() == 1) return m_controlPoints[0];

    if (needsRebuild) {
        rebuildInterpolation();
    }

    validateParameterRange(t);

    size_t numSegments = (type == InterpolationType::OIFS3) ? m_controlPoints.size() : (m_controlPoints.size() - 1);
    double segmentLength = 1.0 / numSegments;
    size_t segmentIndex = std::min(static_cast<size_t>(t / segmentLength), numSegments - 1);
    double localT = (t - segmentIndex * segmentLength) / segmentLength;

    if (type == InterpolationType::NIFS3 || type == InterpolationType::OIFS3) {
        if (segmentIndex >= coefficients.size())
            return m_controlPoints[m_controlPoints.size() - 1];

        const auto& coef = coefficients[segmentIndex];
        Eigen::Vector4d tVec;
        tVec << 1.0, localT, localT*localT, localT*localT*localT;

        double x = tVec.dot(coef.col(0));
        double y = tVec.dot(coef.col(1));

        return Point(x, y, 0.0);
    }
    else if (type == InterpolationType::Lagrange) {
        double x = 0.0;
        double y = 0.0;

        for (size_t i = 0; i < m_controlPoints.size(); ++i) {
            double L = 1.0;
            for (size_t j = 0; j < m_controlPoints.size(); ++j) {
                if (i != j) {
                    double tj = static_cast<double>(j) / (m_controlPoints.size() - 1);
                    double ti = static_cast<double>(i) / (m_controlPoints.size() - 1);
                    L *= (t - tj) / (ti - tj);
                }
            }
            x += m_controlPoints[i].x() * L;
            y += m_controlPoints[i].y() * L;
        }

        return Point(x, y, 0.0);
    }
    else if (type == InterpolationType::Barycentric) {
        double x = 0.0;
        double y = 0.0;
        double sumW = 0.0;

        std::vector<double> weights(m_controlPoints.size());
        for (size_t i = 0; i < m_controlPoints.size(); ++i) {
            double ti = static_cast<double>(i) / (m_controlPoints.size() - 1);
            weights[i] = 1.0;
            for (size_t j = 0; j < m_controlPoints.size(); ++j) {
                if (i != j) {
                    double tj = static_cast<double>(j) / (m_controlPoints.size() - 1);
                    weights[i] *= (ti - tj);
                }
            }
            weights[i] = 1.0 / weights[i];
        }

        for (size_t i = 0; i < m_controlPoints.size(); ++i) {
            double ti = static_cast<double>(i) / (m_controlPoints.size() - 1);

            if (std::abs(t - ti) < 1e-10) {
                return m_controlPoints[i];
            }

            double w = weights[i] / (t - ti);
            x += m_controlPoints[i].x() * w;
            y += m_controlPoints[i].y() * w;
            sumW += w;
        }
        return {x / sumW, y / sumW, 0.0};
    }

    const size_t nextIndex = (segmentIndex + 1) % m_controlPoints.size();
    return interpolateLinear(m_controlPoints[segmentIndex], m_controlPoints[nextIndex], localT);
}

void InterpolatedCurve::buildNIFS3() const {
    const size_t n = m_controlPoints.size();
    if (n < 2) return;

    std::vector<double> xs(n), ys_x(n), ys_y(n);
    for (size_t i = 0; i < n; ++i) {
        xs[i] = static_cast<double>(i) / (n - 1);
        ys_x[i] = m_controlPoints[i].x();
        ys_y[i] = m_controlPoints[i].y();
    }

    std::vector<double> ds_x;
    std::vector<double> ds_y;
    for (size_t i = 1; i < n - 1; ++i) {
        const double term1_x = (ys_x[i + 1] - ys_x[i]) / (xs[i + 1] - xs[i]);
        const double term2_x = (ys_x[i] - ys_x[i - 1]) / (xs[i] - xs[i - 1]);
        const double term1_y = (ys_y[i + 1] - ys_y[i]) / (xs[i + 1] - xs[i]);
        const double term2_y = (ys_y[i] - ys_y[i - 1]) / (xs[i] - xs[i - 1]);
        ds_x.push_back(6.0 * (term1_x - term2_x) / (xs[i + 1] - xs[i - 1]));
        ds_y.push_back(6.0 * (term1_y - term2_y) / (xs[i + 1] - xs[i - 1]));
    }

    std::vector<double> ps;
    ps.push_back(0.0);
    for (size_t i = 1; i < n - 1; ++i) {
        double lambda = (xs[i] - xs[i - 1]) / (xs[i + 1] - xs[i - 1]);
        ps.push_back((lambda - 1.0) / (lambda * ps[i - 1] + 2.0));
    }

    std::vector<double> qs_x;
    std::vector<double> qs_y;
    qs_x.push_back(0.0);
    qs_y.push_back(0.0);
    for (size_t i = 1; i < n - 1; ++i) {
        const double lambda = (xs[i] - xs[i - 1]) / (xs[i + 1] - xs[i - 1]);
        qs_x.push_back((ds_x[i - 1] - lambda * qs_x[i - 1]) / (lambda * ps[i - 1] + 2.0));
        qs_y.push_back((ds_y[i - 1] - lambda * qs_y[i - 1]) / (lambda * ps[i - 1] + 2.0));
    }

    std::vector<double> M_x(n, 0.0), M_y(n, 0.0);

    M_x[n-1] = 0.0;
    M_y[n-1] = 0.0;
    M_x[n-2] = qs_x.back();
    M_y[n-2] = qs_y.back();

    for (int i = n - 3; i > 0; --i) {
        M_x[i] = ps[i] * M_x[i+1] + qs_x[i];
        M_y[i] = ps[i] * M_y[i+1] + qs_y[i];
    }

    coefficients.clear();
    for (size_t i = 0; i < n - 1; ++i) {
        double h = xs[i + 1] - xs[i];
        Eigen::Matrix<double, 4, 2> coef;

        coef(0, 0) = ys_x[i];
        coef(1, 0) = (ys_x[i + 1] - ys_x[i]) - (h * h / 6.0) * (M_x[i + 1] + 2.0 * M_x[i]);
        coef(2, 0) = (h * h / 2.0) * M_x[i];
        coef(3, 0) = (h * h / 6.0) * (M_x[i + 1] - M_x[i]);

        coef(0, 1) = ys_y[i];
        coef(1, 1) = (ys_y[i + 1] - ys_y[i]) - (h * h / 6.0) * (M_y[i + 1] + 2.0 * M_y[i]);
        coef(2, 1) = (h * h / 2.0) * M_y[i];
        coef(3, 1) = (h * h / 6.0) * (M_y[i + 1] - M_y[i]);

        coefficients.push_back(coef);
    }
}

void InterpolatedCurve::buildOIFS3() const {
    const size_t n = m_controlPoints.size();
    if (n < 3) return;

    Eigen::MatrixXd A = Eigen::MatrixXd::Zero(n, n);
    Eigen::VectorXd bx = Eigen::VectorXd::Zero(n);
    Eigen::VectorXd by = Eigen::VectorXd::Zero(n);

    for (size_t i = 0; i < n; ++i) {
        size_t prev = (i > 0) ? (i - 1) : (n - 1);
        size_t next = (i < n - 1) ? (i + 1) : 0;

        A(i, prev) = 1.0;
        A(i, i) = 4.0;
        A(i, next) = 1.0;

        bx(i) = 3.0 * (m_controlPoints[next].x() - m_controlPoints[prev].x());
        by(i) = 3.0 * (m_controlPoints[next].y() - m_controlPoints[prev].y());
    }

    Eigen::VectorXd cx = A.colPivHouseholderQr().solve(bx);
    Eigen::VectorXd cy = A.colPivHouseholderQr().solve(by);

    for (size_t i = 0; i < n; ++i) {
        size_t next = (i < n - 1) ? (i + 1) : 0;

        Eigen::Matrix<double, 4, 2> coef;

        coef(0, 0) = m_controlPoints[i].x();
        coef(1, 0) = cx(i);
        coef(2, 0) = 3.0 * (m_controlPoints[next].x() - m_controlPoints[i].x()) - 2.0 * cx(i) - cx(next);
        coef(3, 0) = 2.0 * (m_controlPoints[i].x() - m_controlPoints[next].x()) + cx(i) + cx(next);

        coef(0, 1) = m_controlPoints[i].y();
        coef(1, 1) = cy(i);
        coef(2, 1) = 3.0 * (m_controlPoints[next].y() - m_controlPoints[i].y()) - 2.0 * cy(i) - cy(next);
        coef(3, 1) = 2.0 * (m_controlPoints[i].y() - m_controlPoints[next].y()) + cy(i) + cy(next);

        coefficients.push_back(coef);
    }
}

void InterpolatedCurve::buildLagrange() const {
    const size_t n = m_controlPoints.size();
    if (n < 2) return;
}

void InterpolatedCurve::buildBarycentric() const {
    const size_t n = m_controlPoints.size();
    if (n < 2) return;
}

double Point::distanceTo(const Point& other) const {
    const double dx = other.x() - x();
    const double dy = other.y() - y();
    const double dz = other.z() - z();
    return std::sqrt(dx * dx + dy * dy + dz * dz);
}

Point Point::midPoint(const Point& other) const {
    return {
        (x() + other.x()) / 2.0,
        (y() + other.y()) / 2.0,
        (z() + other.z()) / 2.0
    };
}

void Point::transform(const TransformMatrix& matrix) {
    Eigen::Vector4d homogeneousPoint;
    homogeneousPoint << x(), y(), z(), 1.0;

    if (Eigen::Vector4d transformedPoint = matrix.getMatrix() * homogeneousPoint; std::abs(transformedPoint(3)) > 1e-10) {
        setX(transformedPoint(0) / transformedPoint(3));
        setY(transformedPoint(1) / transformedPoint(3));
        setZ(transformedPoint(2) / transformedPoint(3));
    }
}

void Canvas::removeObject(std::shared_ptr<GeometricObject> object)
{
    if (!object) return;

    if (m_selectedObject == object)
        clearSelection();

    auto it = std::find(m_objects.begin(), m_objects.end(), object);
    if (it != m_objects.end()) {
        m_objects.erase(it);
        emit objectRemoved(object);
        update();
    }

    clearSelection();
    update();

void Canvas::selectObject(std::shared_ptr<GeometricObject> object)
{
    if (m_selectedObject != object) {
        m_selectedObject = object;
        m_selectedControlPointIndex = -1;

        const auto curve = std::dynamic_pointer_cast<Curve>(object);

        if (bool hasControlPoints = curve && curve->hasControlPoints(); m_isEditingPoints && !hasControlPoints)
            finishEditing();

        emit objectSelected(object);
        update();
    }
}

void Canvas::clearSelection()
{
    if (m_selectedObject) {
        m_selectedObject = nullptr;
        m_selectedControlPointIndex = -1;

        if (m_isEditingPoints)

        emit selectionCleared();
        update();
    }
}

    update();

int Canvas::hitTestControlPoint(const QPointF& pos, double tolerance) const {
    if (!m_viewportManager) return -1;

    const auto selectedObj = selectedObject();
    if (!selectedObj) return -1;

    if (const auto curve = std::dynamic_pointer_cast<Curve>(selectedObj)) {
        if (!curve->hasControlPoints()) return -1;

        const auto& points = curve->getControlPoints();
        for (size_t i = 0; i < points.size(); ++i) {
            QPointF screenPos = m_viewportManager->worldToScreen(
                QPointF(points[i].x(), points[i].y()));
            if (QLineF(pos, screenPos).length() <= tolerance)
                return static_cast<int>(i);
        }
    }
    return -1;
}

bool Canvas::addControlPoint(const QPointF& screenPos) {
    if (!m_viewportManager) return false;

    auto selectedObj = selectedObject();
    if (!selectedObj) return false;

    if (auto curve = std::dynamic_pointer_cast<Curve>(selectedObj)) {
        if (!curve->hasControlPoints()) return false;

        const QPointF worldPos = m_viewportManager->screenToWorld(screenPos);
        const Point newPoint(worldPos.x(), worldPos.y());

        int insertIndex;

        if (m_selectedControlPointIndex >= 0 &&
            m_selectedControlPointIndex < static_cast<int>(curve->getControlPointCount())) {
            insertIndex = m_selectedControlPointIndex + 1;
            curve->insertControlPoint(insertIndex, newPoint);
            } else {
                curve->addControlPoint(newPoint);
                insertIndex = static_cast<int>(curve->getControlPointCount()) - 1;
            }

        selectControlPoint(insertIndex);
        emit controlPointAdded(insertIndex);
        update();
        return true;
    }
    return false;
}

bool Canvas::moveControlPoint(int pointIndex, const QPointF& screenPos) {
    if (!m_viewportManager || pointIndex < 0) return false;

    auto selectedObj = selectedObject();
    if (!selectedObj) return false;

    if (auto curve = std::dynamic_pointer_cast<Curve>(selectedObj)) {
        if (!curve->hasControlPoints()) return false;
        if (pointIndex >= static_cast<int>(curve->getControlPointCount())) return false;

        QPointF worldPos = m_viewportManager->screenToWorld(screenPos);
        const auto& points = curve->getControlPoints();
        auto newPoint = Point(worldPos.x(), worldPos.y(), points[pointIndex].z());
        curve->moveControlPoint(pointIndex, newPoint);

        emit controlPointMoved(pointIndex);
        update();
        return true;
    }
    return false;
}

void Canvas::selectControlPoint(const int pointIndex) {
    auto selectedObj = selectedObject();
    if (!selectedObj) return;

    if (auto curve = std::dynamic_pointer_cast<Curve>(selectedObj)) {
        if (!curve->hasControlPoints()) return;

        int maxPointCount = static_cast<int>(curve->getControlPointCount());

        if (pointIndex >= 0 && pointIndex < maxPointCount) {
            m_selectedControlPointIndex = pointIndex;
            emit controlPointSelected(pointIndex);
        } else {
            m_selectedControlPointIndex = -1;
        }
    } else {
        return;
    }

    update();
}

void Canvas::startAddingControlPoints() {
    setEditMode(EditMode::AddPoint);
}

void Canvas::startMovingControlPoints() {
    setEditMode(EditMode::MovePoint);
}

void Canvas::paintEvent(QPaintEvent* event)
{
    Q_UNUSED(event);

    if (!isVisible() || width() <= 0 || height() <= 0) return;

    QPainter painter(this);
    if (!painter.isActive()) {
        qWarning() << "Nie można zainicjować QPainter w Canvas::paintEvent";
        return;
    }

    painter.fillRect(rect(), Qt::white);

    if (!m_renderer) return;

    m_renderer->beginRender(&painter, m_viewportManager.get());

    m_renderer->renderGrid(1.0);
    m_renderer->renderAxes();

    for (const auto& object : m_objects)
        m_renderer->renderObject(*object);

    if (m_selectedObject) {
        const auto* geometricObj = dynamic_cast<const GeometricObject*>(m_selectedObject.get());
        if (geometricObj) {
            auto position = geometricObj->getPosition();
            QPointF selectionPos = m_viewportManager->worldToScreen(QPointF(position.x(), position.y()));
            m_renderer->renderSelectionMarker(selectionPos);
        }
    }

    if (m_isEditingPoints && m_selectedObject)
        drawControlPoints();

    m_renderer->endRender();
}

void Canvas::drawControlPoints() {
    if (!m_renderer) return;

    const auto selectedObj = selectedObject();
    if (!selectedObj) return;

    if (const auto curve = std::dynamic_pointer_cast<Curve>(selectedObj))
        if (curve->hasControlPoints()) {
            m_renderer->renderControlPoints(*curve, m_selectedControlPointIndex,
                                          m_isDraggingPoint ? m_selectedControlPointIndex : -1);
        }
}

void Canvas::resizeEvent(QResizeEvent* event)
{
    if (m_viewportManager) {
        m_viewportManager->setWindowSize(event->size().width(), event->size().height());
        emit viewportChanged();
    }
    QWidget::resizeEvent(event);
}

void Canvas::mousePressEvent(QMouseEvent* event)
{
    m_lastMousePos = event->pos();

    if (event->button() == Qt::LeftButton)
    {
        switch (m_editMode)
        {
            case EditMode::AddPoint:
                addControlPoint(event->pos());
                return;

            case EditMode::MovePoint:
                if (const int pointIndex = hitTestControlPoint(event->pos()); pointIndex >= 0) {
                    selectControlPoint(pointIndex);
                    m_isDraggingPoint = true;
                    setCursor(Qt::ClosedHandCursor);
                    return;
                }
                break;

            case EditMode::SelectPoint:
                if (const int pointIndex = hitTestControlPoint(event->pos()); pointIndex >= 0) {
                    selectControlPoint(pointIndex);
                    return;
                }
                break;

            case EditMode::None:
                    if (const auto hitObject = hitTestObject(event->pos())) {
                        selectObject(hitObject);
                        if (hasSelectedCurveWithControlPoints()) {
                            setEditMode(EditMode::SelectPoint);
                            emit editingStarted();
                        }
                        return;
                    }
                    clearSelection();
            break;

            default:
                qWarning() << "Nieobsługiwany tryb edycji:" << static_cast<int>(m_editMode);
            break;
        }
    }
    else if (event->button() == Qt::MiddleButton) {
        m_isPanning = true;
        setCursor(Qt::ClosedHandCursor);
    }

    QWidget::mousePressEvent(event);
}

void Canvas::mouseMoveEvent(QMouseEvent* event)
{
    const QPointF currentPos = event->pos();
    const QPointF delta = currentPos - m_lastMousePos;
    if (m_isDraggingPoint && m_selectedControlPointIndex >= 0) {
        moveControlPoint(m_selectedControlPointIndex, currentPos);
    }
    else if (m_isPanning) {
        if (m_viewportManager) {
            m_viewportManager->pan(delta);
            emit viewportChanged();
            update();
        }
    }
    else if (m_editMode == EditMode::MovePoint) {
        if (const int pointIndex = hitTestControlPoint(currentPos); pointIndex >= 0) {
            setCursor(Qt::OpenHandCursor);
        } else {
            setCursor(Qt::ArrowCursor);
        }
    }

    m_lastMousePos = currentPos;
    QWidget::mouseMoveEvent(event);
}

void Canvas::mouseReleaseEvent(QMouseEvent* event)
{
    if (event->button() == Qt::LeftButton) {
        if (m_isDraggingPoint) {
            m_isDraggingPoint = false;

            if (m_editMode == EditMode::MovePoint) {
                setCursor(Qt::OpenHandCursor);
            } else {
                setCursor(Qt::ArrowCursor);
            }
        }
    }
    else if (event->button() == Qt::MiddleButton) {
        m_isPanning = false;
        setCursor(Qt::ArrowCursor);
    }

    QWidget::mouseReleaseEvent(event);
}

void Canvas::wheelEvent(QWheelEvent* event)
{
    if (m_viewportManager) {
        QPointF center = event->position();

        double factor = event->angleDelta().y() > 0 ? 1.1 : 0.9;

        m_viewportManager->zoom(factor, center);
        emit viewportChanged();
        update();
    }

    event->accept();
}

void Canvas::keyPressEvent(QKeyEvent* event)
{
    if (event->key() == Qt::Key_Escape) {
        finishEditing();
    }
    else if (event->key() == Qt::Key_Delete && m_selectedControlPointIndex >= 0) {
        auto selectedObj = selectedObject();
        if (!selectedObj) return;

        if (auto curve = std::dynamic_pointer_cast<Curve>(selectedObj)) {
            if (curve->hasControlPoints() &&
                m_selectedControlPointIndex < static_cast<int>(curve->getControlPointCount())) {

                if (curve->getControlPointCount() > 2) {
                    curve->removeControlPoint(m_selectedControlPointIndex);
                    m_selectedControlPointIndex = -1;
                    update();
                } else {
                    QMessageBox::warning(this, "Usuwanie punktu",
                        "Krzywa musi mieć co najmniej 2 punkty kontrolne.");
                }
                }
        }
    }

    QWidget::keyPressEvent(event);
}

bool Canvas::hasSelectedCurveWithControlPoints() const {
    const auto selectedObj = selectedObject();
    if (!selectedObj) return false;

    if (const auto curve = std::dynamic_pointer_cast<Curve>(selectedObj))
        return curve->hasControlPoints();

    return false;
}

std::shared_ptr<GeometricObject> Canvas::hitTestObject(const QPointF& pos, double tolerance) const {
    if (!m_viewportManager) return nullptr;

    for (auto it = m_objects.rbegin(); it != m_objects.rend(); ++it) {
        const auto& object = *it;
        if (auto curve = std::dynamic_pointer_cast<Curve>(object)) {
            constexpr int testPoints = 100;
            for (int i = 0; i <= testPoints; ++i) {
                double t = static_cast<double>(i) / testPoints;
                Point curvePoint = curve->getPointAt(t);
                QPointF screenCurvePos = m_viewportManager->worldToScreen(
                    QPointF(curvePoint.x(), curvePoint.y()));

                if (QLineF(pos, screenCurvePos).length() <= tolerance)
                    return object;
            }
        }
    }

    return nullptr;
}

void QtRenderer::beginRender(QPainter* painter, ViewportManager* viewportManager)
{
    if (!painter || !viewportManager) {
        qDebug() << "Nieprawidłowy painter lub viewportManager";
        return;
    }
    if (!painter->isActive()) {
        qWarning() << "QPainter nie jest aktywny w beginRender()";
        return;
    }
    m_painter = painter;
    m_viewportManager = viewportManager;

    m_painter->save();
    m_painter->setTransform(viewportManager->viewTransform());
    m_painter->setRenderHint(QPainter::Antialiasing, true);
}

void QtRenderer::endRender()
{
    if (m_painter) {
        m_painter->restore();

        m_painter = nullptr;
        m_viewportManager = nullptr;
    }
}

void QtRenderer::renderPoint(const Point& point)
{
    if (!m_painter || !m_viewportManager || !m_painter->isActive())
        return;

    QPen pen(Qt::NoPen);
    QBrush brush(m_pointColor);
    m_painter->setPen(pen);
    m_painter->setBrush(brush);

    const double x = point.x(), y = point.y();

    m_painter->drawEllipse(QPointF(x, y), m_pointSize, m_pointSize);
}

void QtRenderer::renderCurve(const Curve& curve) {
    if (!m_painter || !m_viewportManager) return;

    renderMainCurve(curve);
    renderCurveExtras(curve);

    if (const auto* bezierCurve = dynamic_cast<const BezierCurve*>(&curve))
        if (bezierCurve->getShowWeights() && bezierCurve->hasWeights())
            renderWeights(*bezierCurve);
}

void QtRenderer::renderMainCurve(const Curve& curve) {
    if (!curve.getDrawMainCurve() || !curve.hasControlPoints() || curve.getControlPointCount() < 2)
        return;

    QColor color = curve.getColor();
    double width = curve.getLineWidth();
    Qt::PenStyle style = curve.getLineStyle();

    const QPen pen(color, width, style);
    m_painter->setPen(pen);
    m_painter->setBrush(Qt::NoBrush);

    constexpr int numSegments = 1000;
    QPolygonF points;
    for (int i = 0; i <= numSegments; ++i) {
        double t = static_cast<double>(i) / numSegments;
        Point p = curve.getPointAt(t);
        points.append(QPointF(p.x(), p.y()));
    }
    m_painter->drawPolyline(points);
}

void QtRenderer::renderCurveExtras(const Curve& curve) {
    if (!curve.hasControlPoints() || curve.getControlPointCount() < 2) return;

    renderCurvePolyline(curve);
    renderCurveConvexHull(curve);
}

void QtRenderer::renderCurvePolyline(const Curve& curve) {
    if (!curve.getDrawPolyline()) return;

    const auto& controlPoints = curve.getControlPoints();
    const QPen pen(curve.getPolylineColor(), curve.getPolylineWidth(), curve.getPolylineStyle());
    m_painter->setPen(pen);
    m_painter->setBrush(Qt::NoBrush);

    QPolygonF polylinePoints = pointsToPolygon(controlPoints);
    m_painter->drawPolyline(polylinePoints);
}

void QtRenderer::renderCurveConvexHull(const Curve& curve) {
    if (!curve.getDrawConvexHull() || curve.getControlPointCount() < 3) return;

    const auto& controlPoints = curve.getControlPoints();
    const std::vector<Point> hull = computeConvexHull(controlPoints);

    if (hull.size() < 3) return;

    const QPen pen(curve.getConvexHullColor(), curve.getConvexHullWidth(), curve.getConvexHullStyle());
    m_painter->setPen(pen);
    m_painter->setBrush(Qt::NoBrush);

    QPolygonF hullPolygon = pointsToPolygon(hull);
    if (!hull.empty()) {
        hullPolygon << QPointF(hull[0].x(), hull[0].y());
    }
    m_painter->drawPolyline(hullPolygon);
}

void QtRenderer::renderWeights(const BezierCurve& curve) {
    if (!curve.hasControlPoints() || !curve.hasWeights()) return;

    const auto& points = curve.getControlPoints();
    const auto& weights = curve.getWeights();

    m_painter->save();
    m_painter->resetTransform();

    QPen textPen(Qt::darkBlue);
    m_painter->setPen(textPen);

    QFont font = m_painter->font();
    font.setPointSize(9);
    font.setBold(true);
    m_painter->setFont(font);

    for (size_t i = 0; i < points.size() && i < weights.size(); ++i) {
        QPointF screenPos = m_viewportManager->worldToScreen(QPointF(points[i].x(), points[i].y()));

        QString weightText = QString("w%1: %2").arg(std::to_string(i)).arg(std::to_string(weights[i]));
        QFontMetrics fm(font);
        QRect textRect = fm.boundingRect(weightText);

        QPointF textPos = screenPos + QPointF(-textRect.width() - 5, -curve.getControlPointSize() - 5);

        QRectF bgRect(textPos.x() - 2, textPos.y() - textRect.height(),
                      textRect.width() + 4, textRect.height() + 2);
        m_painter->fillRect(bgRect, QColor(255, 255, 255, 200));

        m_painter->setPen(QPen(Qt::darkBlue, 1));
        m_painter->drawRect(bgRect);

        m_painter->setPen(textPen);
        m_painter->drawText(textPos, weightText);
    }

    m_painter->restore();
}

QPolygonF QtRenderer::pointsToPolygon(const std::vector<Point>& points) const {
    QPolygonF polygon;
    polygon.reserve(points.size());
    for (const auto& point : points) {
        polygon << QPointF(point.x(), point.y());
    }
    return polygon;
}

void QtRenderer::renderObject(const GeometricObject& object)

    if (const auto* point = dynamic_cast<const Point*>(&object)) {
        renderPoint(*point);
        return;
    }

    if (const auto* curve = dynamic_cast<const Curve*>(&object))
        renderCurve(*curve);
}

void QtRenderer::renderGrid(double gridSize) {
    if (!m_drawGrid || !m_painter) return;

    const QRectF visibleRect = m_viewportManager->visibleWorldRect();

    const QPen smallPen(QColor(230, 230, 230), 0.5, Qt::DotLine);
    m_painter->setPen(smallPen);

    for (double x = std::floor(visibleRect.left()); x <= visibleRect.right(); x += 1.0)
        m_painter->drawLine(QPointF(x, visibleRect.top()), QPointF(x, visibleRect.bottom()));

    for (double y = std::floor(visibleRect.top()); y <= visibleRect.bottom(); y += 1.0)
        m_painter->drawLine(QPointF(visibleRect.left(), y), QPointF(visibleRect.right(), y));

    const QPen mainPen(QColor(200, 200, 200), 1.0, Qt::SolidLine);
    m_painter->setPen(mainPen);

    for (double x = std::floor(visibleRect.left() / 5) * 5; x <= visibleRect.right(); x += 5.0)
        m_painter->drawLine(QPointF(x, visibleRect.top()), QPointF(x, visibleRect.bottom()));

    for (double y = std::floor(visibleRect.top() / 5) * 5; y <= visibleRect.bottom(); y += 5.0)
        m_painter->drawLine(QPointF(visibleRect.left(), y), QPointF(visibleRect.right(), y));
}

void QtRenderer::renderText(const QPointF& position, const QString& text) const {
    if (!m_painter) return;
    m_painter->save();
    const QTransform current = m_painter->transform();
    m_painter->resetTransform();
    const QPointF screenPos = current.map(position);
    m_painter->drawText(screenPos, text);
    m_painter->restore();
}

void QtRenderer::renderAxes()
{
    if (!m_drawAxes || !m_painter || !m_viewportManager) return;

    const QPen pen(m_axesColor, m_axesWidth, Qt::SolidLine);
    m_painter->setPen(pen);

    const QRectF visibleRect = m_viewportManager->visibleWorldRect();

    m_painter->drawLine(QPointF(visibleRect.left(), 0), QPointF(visibleRect.right(), 0));
    m_painter->drawLine(QPointF(0, visibleRect.top()), QPointF(0, visibleRect.bottom()));

    const double arrowSize = 7.0 / m_viewportManager->zoomFactor();

    QPolygonF arrowX;
    arrowX << QPointF(visibleRect.right(), 0)
          << QPointF(visibleRect.right() - arrowSize, arrowSize / 2)
          << QPointF(visibleRect.right() - arrowSize, -arrowSize / 2);
    m_painter->setBrush(m_axesColor);
    m_painter->drawPolygon(arrowX);

    QPolygonF arrowY;
    arrowY << QPointF(0, visibleRect.bottom())
          << QPointF(arrowSize / 2, visibleRect.bottom() - arrowSize)
          << QPointF(-arrowSize / 2, visibleRect.bottom() - arrowSize);
    m_painter->drawPolygon(arrowY);

    renderText(QPointF(visibleRect.right() - 2 * arrowSize, -arrowSize), "x");
    renderText(QPointF(arrowSize, visibleRect.bottom() - 2 * arrowSize), "y");

    renderAxisTicks();
}

void QtRenderer::renderAxisTicks() {
    if (!m_painter || !m_viewportManager || !m_painter->isActive()) return;

    const QRectF visibleRect = m_viewportManager->visibleWorldRect();
    const double zoomFactor = m_viewportManager->zoomFactor();

    const double viewportWidth = visibleRect.width(), viewportHeight = visibleRect.height();

    constexpr double desiredTickCount = 16.0;
    const double optimalSpacingX = viewportWidth / desiredTickCount, optimalSpacingY = viewportHeight / desiredTickCount;

    auto findNiceSpacing = [](double optimalSpacing) -> double {
        const double magnitude = pow(10, floor(log10(optimalSpacing)));
        const double normalized = optimalSpacing / magnitude;

        if (normalized < 1.2) return 1.0 * magnitude;
        if (normalized < 2.5) return 2.0 * magnitude;
        if (normalized < 7.0) return 5.0 * magnitude;
        return 10.0 * magnitude;
    };

    const double tickSpacingX = findNiceSpacing(optimalSpacingX), tickSpacingY = findNiceSpacing(optimalSpacingY);

    const double xMin = floor(visibleRect.left() / tickSpacingX) * tickSpacingX;
    const double xMax = ceil(visibleRect.right() / tickSpacingX) * tickSpacingX;
    const double yMin = floor(visibleRect.top() / tickSpacingY) * tickSpacingY;
    const double yMax = ceil(visibleRect.bottom() / tickSpacingY) * tickSpacingY;

    const double tickLength = 5.0 / zoomFactor;

    auto determinePrecision = [](const double spacing) -> int {
        if (spacing >= 1.0) return 0;
        return std::min(6, static_cast<int>(std::ceil(std::abs(std::log10(spacing)))));
    };

    const int xPrecision = determinePrecision(tickSpacingX), yPrecision = determinePrecision(tickSpacingY);

    for (double x = xMin; x <= xMax; x += tickSpacingX) {
        if (std::abs(x) < 1e-6) continue;

        m_painter->drawLine(QPointF(x, -tickLength/2), QPointF(x, tickLength/2));
        QString label = QString::number(x, 'f', xPrecision);
        while (label.contains('.') && label.endsWith('0'))
            label.chop(1);

        if (label.endsWith('.'))
            label.chop(1);

        renderText(QPointF(x, -tickLength - 5.0/zoomFactor), label);
    }

    for (double y = yMin; y <= yMax; y += tickSpacingY) {
        if (std::abs(y) < 1e-6) continue;

        m_painter->drawLine(QPointF(-tickLength/2, y), QPointF(tickLength/2, y));
        QString label = QString::number(y, 'f', yPrecision);
        while (label.contains('.') && label.endsWith('0'))
            label.chop(1);

        if (label.endsWith('.'))
            label.chop(1);

        renderText(QPointF(-tickLength - 15.0/zoomFactor, y), label);
    }
}

void QtRenderer::renderControlPoint(const QPointF& position, const Curve& curve, bool isSelected, bool isActive)
{
    if (!m_painter || !m_painter->isActive())
        return;

    m_painter->save();
    m_painter->resetTransform();

    QColor fillColor;
    if (isActive) {
        fillColor = curve.getControlPointActiveColor();
    } else if (isSelected) {
        fillColor = curve.getControlPointSelectedColor();
    } else {
        fillColor = curve.getControlPointNormalColor();
    }

    double pointSize = curve.getControlPointSize();

    QPen shadowPen(Qt::black, 1.0);
    QBrush shadowBrush(QColor(0, 0, 0, 50));
    m_painter->setPen(shadowPen);
    m_painter->setBrush(shadowBrush);
    m_painter->drawEllipse(position + QPointF(1, 1), pointSize + 1, pointSize + 1);

    QPen pen(Qt::black, 1.0);
    QBrush brush(fillColor);
    m_painter->setPen(pen);
    m_painter->setBrush(brush);
    m_painter->drawEllipse(position, pointSize, pointSize);

    if (isSelected) {
        m_painter->setPen(Qt::white);
        QFont font = m_painter->font();
        font.setPointSize(9);
        m_painter->setFont(font);

        QFontMetrics fm(font);
        QString text = QString::number(isActive ? 1 : 0);
        QRect textRect = fm.boundingRect(text);
        QPointF textPos = position - QPointF(textRect.width() / 2.0, -textRect.height() / 4.0);

        m_painter->drawText(textPos, text);
    }

    m_painter->restore();
}

void QtRenderer::renderControlPoints(const Curve& curve, int selectedIndex, int draggingIndex)
{
    if (!m_painter || !m_viewportManager || !curve.hasControlPoints())
        return;

    const auto& points = curve.getControlPoints();
    if (points.empty()) return;

    if (!curve.getShowControlPoints()) return;

    std::vector<QPointF> screenPositions;
    screenPositions.reserve(points.size());

    for (const auto& point : points)
        screenPositions.push_back(m_viewportManager->worldToScreen(QPointF(point.x(), point.y())));

    for (size_t i = 0; i < points.size(); ++i)
        if (static_cast<int>(i) != selectedIndex && static_cast<int>(i) != draggingIndex)
            renderControlPoint(screenPositions[i], curve, false, false);

    if (selectedIndex >= 0 && selectedIndex < static_cast<int>(points.size()))
        renderControlPoint(screenPositions[selectedIndex], curve, true, false);

    if (draggingIndex >= 0 && draggingIndex < static_cast<int>(points.size()) && draggingIndex != selectedIndex)
        renderControlPoint(screenPositions[draggingIndex], curve, false, true);

    m_painter->save();
    m_painter->resetTransform();
    QPen textPen(Qt::black);
    m_painter->setPen(textPen);

    QFont font = m_painter->font();
    font.setPointSize(8);
    m_painter->setFont(font);

    for (size_t i = 0; i < points.size(); ++i) {
        QString label = QString::number(i);
        QFontMetrics fm(font);
        QRect textRect = fm.boundingRect(label);
        QPointF textPos = screenPositions[i] + QPointF(curve.getControlPointSize() + 2,
                                                  curve.getControlPointSize() / 2.0);

        auto bgRect = QRectF(textPos.x() - 1, textPos.y() - textRect.height(),
                              textRect.width() + 2, textRect.height() + 2);
        m_painter->fillRect(bgRect, QColor(255, 255, 255, 180));

        m_painter->drawText(textPos, label);
    }

    m_painter->restore();
}

std::vector<Point> QtRenderer::computeConvexHull(const std::vector<Point>& points) const
{
    if (points.size() < 3) return points;

    std::vector<Point> pts = points;

    const auto pivotIt = std::ranges::min_element(pts,
                                                  [](const Point& a, const Point& b) {
                                                      if (std::abs(a.y() - b.y()) < 1e-9)
                                                          return a.x() < b.x();
                                                      return a.y() < b.y();
                                                  });
    Point pivot = *pivotIt;
    std::swap(*pivotIt, pts[0]);

    std::sort(pts.begin() + 1, pts.end(),
        [this, &pivot](const Point& a, const Point& b) {
            const double angleA = polarAngle(pivot, a), angleB = polarAngle(pivot, b);

            if (std::abs(angleA - angleB) < 1e-9)
                return distanceSquared(pivot, a) < distanceSquared(pivot, b);

            return angleA < angleB;
        });

    std::vector<Point> filtered;
    filtered.push_back(pts[0]);

    for (size_t i = 1; i < pts.size(); ++i) {
        for (;i < pts.size() - 1 &&
            std::abs(polarAngle(pivot, pts[i]) - polarAngle(pivot, pts[i + 1])) < 1e-9;
            i++) {}
        filtered.push_back(pts[i]);
    }

    if (filtered.size() < 3) return filtered;

    std::vector<Point> hull;
    hull.push_back(filtered[0]);
    hull.push_back(filtered[1]);

    for (size_t i = 2; i < filtered.size(); ++i) {
        while (hull.size() > 1 && crossProduct(hull[hull.size()-2], hull[hull.size()-1], filtered[i]) <= 0)
            hull.pop_back();
        hull.push_back(filtered[i]);
    }

    return hull;
}

double QtRenderer::crossProduct(const Point& O, const Point& A, const Point& B)
{
    return (A.x() - O.x()) * (B.y() - O.y()) - (A.y() - O.y()) * (B.x() - O.x());
}

double QtRenderer::polarAngle(const Point& pivot, const Point& point)
{
    return std::atan2(point.y() - pivot.y(), point.x() - pivot.x());
}

double QtRenderer::distanceSquared(const Point& a, const Point& b)
{
    const double dx = a.x() - b.x(), dy = a.y() - b.y();
    return dx * dx + dy * dy;
}

// Stałe domyślne
constexpr double DEFAULT_MIN_ZOOM = 0.1;
constexpr double DEFAULT_MAX_ZOOM = 100.0;
constexpr double DEFAULT_ZOOM_FACTOR = 1.0;
constexpr QPointF DEFAULT_PAN_OFFSET(0.0, 0.0);
constexpr QRectF DEFAULT_VIEWPORT(-10.0, -10.0, 20.0, 20.0);

ViewportManager::ViewportManager()
    : m_viewport(DEFAULT_VIEWPORT)
    , m_windowWidth(800)
    , m_windowHeight(600)
    , m_zoomFactor(DEFAULT_ZOOM_FACTOR)
    , m_panOffset(DEFAULT_PAN_OFFSET)
    , m_minZoom(DEFAULT_MIN_ZOOM)
    , m_maxZoom(DEFAULT_MAX_ZOOM)
{
    updateTransformations();
}

ViewportManager::~ViewportManager() = default;

void ViewportManager::pan(const QPointF& delta)
{
    const QPointF worldDelta = screenToWorld(delta) - screenToWorld(QPointF(0, 0));
    m_panOffset -= worldDelta;

    updateTransformations();
}

void ViewportManager::zoom(double factor, const QPointF& center)
{
    double newZoom = m_zoomFactor * factor;
    if (newZoom < m_minZoom) {
        factor = m_minZoom / m_zoomFactor;
        newZoom = m_minZoom;
    }
    else if (newZoom > m_maxZoom) {
        factor = m_maxZoom / m_zoomFactor;
        newZoom = m_maxZoom;
    }

    if (std::abs(factor - 1.0) < 1e-6)
        return;

    QPointF worldCenter = screenToWorld(center);
    m_zoomFactor = newZoom;

    QPointF newWorldCenter = screenToWorld(center);
    m_panOffset += worldCenter - newWorldCenter;

    updateTransformations();
}

void ViewportManager::resetView()
{
    m_zoomFactor = DEFAULT_ZOOM_FACTOR;
    m_panOffset = DEFAULT_PAN_OFFSET;
    m_viewport = DEFAULT_VIEWPORT;

    updateTransformations();
}

void ViewportManager::fitView(const QRectF& rect)
{
    if (rect.isEmpty())
        return;

    double scaleX = m_windowWidth / rect.width();
    double scaleY = m_windowHeight / rect.height();

    double newZoom = std::min(scaleX, scaleY) * 0.85;

    newZoom = std::max(m_minZoom, std::min(m_maxZoom, newZoom));

    m_zoomFactor = newZoom;
    m_panOffset = QPointF(rect.center().x(), rect.center().y());

    updateTransformations();
}

void ViewportManager::zoomIn(const QPointF& center) { zoom(1.2, center); }

void ViewportManager::zoomOut(const QPointF& center) { zoom(0.8, center); }

void ViewportManager::panBy(double dx, double dy)
{
    pan(QPointF(dx, dy));
}

    return m_inverseViewTransform.map(screenPoint);

void ViewportManager::setMinZoom(const double minZoom)
{
    m_minZoom = minZoom > 0 ? minZoom : DEFAULT_MIN_ZOOM;
    if (m_zoomFactor < m_minZoom) {
        m_zoomFactor = m_minZoom;
        updateTransformations();
    }
}

void ViewportManager::setMaxZoom(const double maxZoom)
{
    m_maxZoom = maxZoom > m_minZoom ? maxZoom : DEFAULT_MAX_ZOOM;
    if (m_zoomFactor > m_maxZoom) {
        m_zoomFactor = m_maxZoom;
        updateTransformations();
    }
}

void ViewportManager::setZoomConstraints(const double minZoom, const double maxZoom)
{
    m_minZoom = minZoom > 0 ? minZoom : DEFAULT_MIN_ZOOM;
    m_maxZoom = maxZoom > m_minZoom ? maxZoom : DEFAULT_MAX_ZOOM;

    if (m_zoomFactor < m_minZoom) m_zoomFactor = m_minZoom;
    else if (m_zoomFactor > m_maxZoom) m_zoomFactor = m_maxZoom;
    updateTransformations();
}

void ViewportManager::updateTransformations()
{
    m_viewTransform.reset();
    m_viewTransform.translate(m_windowWidth / 2.0, m_windowHeight / 2.0);

    m_viewTransform.scale(m_zoomFactor, -m_zoomFactor);

    m_viewTransform.translate(-m_panOffset.x(), -m_panOffset.y());

    m_inverseViewTransform = m_viewTransform.inverted();
}

std::shared_ptr<TransformMatrix> EigenMathAdapter::createRotationMatrix(double angle, const std::array<double, 3>& axis) {
    Eigen::Vector3d eigenAxis(axis[0], axis[1], axis[2]);
    Eigen::AngleAxisd rotation(angle, eigenAxis.normalized());
    Eigen::Matrix4d matrix = Eigen::Matrix4d::Identity();
    matrix.block<3, 3>(0, 0) = rotation.toRotationMatrix();
    return std::make_shared<StandardTransformMatrix>(matrix);
}

std::shared_ptr<TransformMatrix> EigenMathAdapter::createTranslationMatrix(const std::array<double, 3>& translation) {
    Eigen::Matrix4d matrix = Eigen::Matrix4d::Identity();
    matrix(0, 3) = translation[0];
    matrix(1, 3) = translation[1];
    matrix(2, 3) = translation[2];
    return std::make_shared<StandardTransformMatrix>(matrix);
}

std::shared_ptr<TransformMatrix> EigenMathAdapter::createScalingMatrix(const std::array<double, 3>& factors) {
    Eigen::Matrix4d matrix = Eigen::Matrix4d::Identity();
    matrix(0, 0) = factors[0];
    matrix(1, 1) = factors[1];
    matrix(2, 2) = factors[2];
    return std::make_shared<StandardTransformMatrix>(matrix);
}

std::vector<double> EigenMathAdapter::solveLinearSystem(
    const std::vector<std::vector<double>>& coefficients,
    const std::vector<double>& constants) {

    Eigen::MatrixXd A = toEigenMatrix(coefficients);
    Eigen::VectorXd b = Eigen::Map<const Eigen::VectorXd>(constants.data(), constants.size());

    Eigen::VectorXd x = A.colPivHouseholderQr().solve(b);

    return std::vector<double>(x.data(), x.data() + x.size());
}

Eigen::MatrixXd EigenMathAdapter::toEigenMatrix(const std::vector<std::vector<double>>& matrix) const {
    if (matrix.empty()) return Eigen::MatrixXd(0, 0);

    size_t rows = matrix.size();
    size_t cols = matrix[0].size();

    Eigen::MatrixXd result(rows, cols);
    for (size_t i = 0; i < rows; ++i) {
        for (size_t j = 0; j < cols && j < matrix[i].size(); ++j) {
            result(i, j) = matrix[i][j];
        }
    }
    return result;
}

void MainWindow::connectSignals() {
    connect(m_canvas, &Canvas::objectAdded, this, &MainWindow::onObjectAdded);
    connect(m_canvas, &Canvas::objectRemoved, this, &MainWindow::onObjectRemoved);
    connect(m_canvas, &Canvas::objectSelected, this, &MainWindow::onObjectSelected);
    connect(m_canvas, &Canvas::selectionCleared, this, &MainWindow::onSelectionCleared);
    connect(m_canvas, &Canvas::controlPointAdded, this, &MainWindow::onControlPointAdded);
    connect(m_canvas, &Canvas::controlPointMoved, this, &MainWindow::onControlPointMoved);
    connect(m_canvas, &Canvas::controlPointSelected, this, &MainWindow::onControlPointSelected);
    connect(m_canvas, &Canvas::editingStarted, this, &MainWindow::onEditingStarted);
    connect(m_canvas, &Canvas::editingFinished, this, &MainWindow::onEditingFinished);

    connect(m_objectsTree, &QTreeWidget::itemClicked, this, [this](QTreeWidgetItem* item, int column) {
        if (item->parent() == nullptr) {
            int index = item->data(0, Qt::UserRole).toInt();
            if (index >= 0 && index < static_cast<int>(m_canvas->objects().size())) {
                m_canvas->selectObject(m_canvas->objects()[index]);
                m_canvas->finishEditing();
                m_selectModeAction->setChecked(true);
                updateModeLabel();
            }
        }
        else {
            int pointIndex = item->data(0, Qt::UserRole).toInt();
            int objectIndex = item->data(0, Qt::UserRole + 1).toInt();
            if (objectIndex >= 0 && objectIndex < static_cast<int>(m_canvas->objects().size())) {
                m_canvas->selectObject(m_canvas->objects()[objectIndex]);

                auto curve = std::dynamic_pointer_cast<Curve>(m_canvas->selectedObject());
                if (curve && pointIndex >= 0 && pointIndex < static_cast<int>(curve->getControlPointCount())) {
                    if (m_canvas->getEditMode() == Canvas::EditMode::None) {
                        m_canvas->setEditMode(Canvas::EditMode::SelectPoint);
                        m_selectModeAction->setChecked(true);
                    }
                    m_canvas->selectControlPoint(pointIndex);
                    updateModeLabel();
                }
            }
        }
    });
    connect(m_showWeightsPanelAction, &QAction::toggled, m_weightsDock, &QDockWidget::setVisible);
    connect(m_weightsDock, &QDockWidget::visibilityChanged, m_showWeightsPanelAction, &QAction::setChecked);

    connect(m_weightsPanel, &WeightsPanel::weightChanged, this, &MainWindow::onWeightChanged);
    connect(m_weightsPanel, &WeightsPanel::weightsReset, this, &MainWindow::onWeightsReset);
    connect(m_weightsPanel, &WeightsPanel::weightsNormalized, this, &MainWindow::onWeightsNormalized);
    connect(m_canvas, &Canvas::objectSelected, this, &MainWindow::updateWeightsPanel);
    connect(m_canvas, &Canvas::selectionCleared, this, &MainWindow::updateWeightsPanel);
    connect(m_weightsPanel, &WeightsPanel::showWeightsToggled, this, &MainWindow::onShowWeightsToggled);
}

void MainWindow::updateObjectsTree() {
    m_objectsTree->clear();

    int index = 0;
    for (const auto& obj : m_canvas->objects()) {
        auto* item = new QTreeWidgetItem(m_objectsTree);
        item->setText(0, QString("Obiekt %1").arg(std::to_string(index + 1)));

        if (const auto curve = std::dynamic_pointer_cast<InterpolatedCurve>(obj)) {
            item->setText(1, "Krzywa");

            const auto& points = curve->getControlPoints();
            for (size_t i = 0; i < points.size(); ++i) {
                auto* pointItem = new QTreeWidgetItem(item);
                const Point& p = points[i];
                pointItem->setText(0, QString("Punkt %1").arg(std::to_string(i)));
                pointItem->setText(1, QString("(%1, %2)")
                    .arg(QString::number(p.x(), 'f', 2), QString::number(p.y(), 'f', 2)));

                pointItem->setData(0, Qt::UserRole, QVariant::fromValue(static_cast<int>(i)));
                pointItem->setData(0, Qt::UserRole + 1, QVariant::fromValue(index));
            }
        }
        else if (const auto bezierCurve = std::dynamic_pointer_cast<BezierCurve>(obj)) {
            item->setText(1, "Krzywa Béziera");

            const auto& points = bezierCurve->getControlPoints();
            for (size_t i = 0; i < points.size(); ++i) {
                auto* pointItem = new QTreeWidgetItem(item);
                const Point& p = points[i];
                pointItem->setText(0, QString("Punkt %1").arg(std::to_string(i)));
                pointItem->setText(1, QString("(%1, %2)")
                    .arg(QString::number(p.x(), 'f', 2), QString::number(p.y(), 'f', 2)));

                pointItem->setData(0, Qt::UserRole, QVariant::fromValue(static_cast<int>(i)));
                pointItem->setData(0, Qt::UserRole + 1, QVariant::fromValue(index));
            }
        }
        else {
            item->setText(1, "Inny");
        }

        item->setData(0, Qt::UserRole, QVariant::fromValue(index));
        index++;
    }

    m_objectsTree->expandAll();
}

    switch (m_canvas->getEditMode()) {
        case Canvas::EditMode::None:
            m_modeLabel->setText("Tryb: Normalny");
            break;
        case Canvas::EditMode::SelectPoint:
            m_modeLabel->setText("Tryb: Zaznaczanie obiektu");
            break;
        case Canvas::EditMode::AddPoint:
            m_modeLabel->setText("Tryb: Dodawanie punktu");
            break;
        case Canvas::EditMode::MovePoint:
            m_modeLabel->setText("Tryb: Przesuwanie punktu");
            break;
    }
}

void MainWindow::onObjectSelected(std::shared_ptr<GeometricObject> obj) {
    updateObjectsTree();
    updatePropertiesPanel();
    auto objects = m_canvas->objects();
    auto it = std::find(objects.begin(), objects.end(), obj);
    if (it != objects.end()) {
        int index = static_cast<int>(std::distance(objects.begin(), it));

        for (int i = 0; i < m_objectsTree->topLevelItemCount(); ++i) {
            QTreeWidgetItem* item = m_objectsTree->topLevelItem(i);
            if (item->data(0, Qt::UserRole).toInt() == index) {
                m_objectsTree->setCurrentItem(item);
                break;
            }
        }
    }

    statusBar()->showMessage("Wybrano obiekt");
}

void MainWindow::onControlPointSelected(int pointIndex) {
    updateObjectsTree();

    auto selectedObj = m_canvas->selectedObject();
    auto objects = m_canvas->objects();
    auto it = std::find(objects.begin(), objects.end(), selectedObj);
    if (it != objects.end()) {
        int objectIndex = static_cast<int>(std::distance(objects.begin(), it));

        for (int i = 0; i < m_objectsTree->topLevelItemCount(); ++i) {
            const QTreeWidgetItem* objectItem = m_objectsTree->topLevelItem(i);
            if (objectItem->data(0, Qt::UserRole).toInt() == objectIndex) {
                for (int j = 0; j < objectItem->childCount(); ++j) {
                    QTreeWidgetItem* pointItem = objectItem->child(j);
                    if (pointItem->data(0, Qt::UserRole).toInt() == pointIndex) {
                        m_objectsTree->setCurrentItem(pointItem);
                        break;
                    }
                }
                break;
            }
        }
    }

    statusBar()->showMessage(QString("Wybrano punkt kontrolny %1").arg(std::to_string(pointIndex)));
}

void MainWindow::updatePropertiesPanel() {
    QLayoutItem* child;
    while ((child = m_propertiesLayout->takeAt(0)) != nullptr) {
        delete child->widget();
        delete child;
    }

    auto selectedObject = m_canvas->selectedObject();
    if (!selectedObject) {
        m_propertiesLayout->addWidget(new QLabel("Wybierz obiekt, aby edytować jego właściwości"));
        return;
    }

    if (auto curve = std::dynamic_pointer_cast<InterpolatedCurve>(selectedObject)) {
        m_propertiesLayout->addWidget(new QLabel("<b>Właściwości krzywej interpolowanej</b>"));

        m_propertiesLayout->addWidget(new QLabel("Typ interpolacji:"));
        QComboBox* typeCombo = new QComboBox();
        typeCombo->addItem("NIFS3 (Natural Cubic Spline)", QVariant::fromValue(static_cast<int>(InterpolatedCurve::InterpolationType::NIFS3)));
        typeCombo->addItem("OIFS3 (Closed Cubic Spline)", QVariant::fromValue(static_cast<int>(InterpolatedCurve::InterpolationType::OIFS3)));
        typeCombo->addItem("Lagrange", QVariant::fromValue(static_cast<int>(InterpolatedCurve::InterpolationType::Lagrange)));
        typeCombo->addItem("Barycentric", QVariant::fromValue(static_cast<int>(InterpolatedCurve::InterpolationType::Barycentric)));

        typeCombo->setCurrentIndex(static_cast<int>(curve->getInterpolationType()));
        m_propertiesLayout->addWidget(typeCombo);

        connect(typeCombo, QOverload<int>::of(&QComboBox::currentIndexChanged),
            [this, curve](int index) {
                const auto type = static_cast<InterpolatedCurve::InterpolationType>(index);
                curve->setInterpolationType(type);
                m_canvas->refresh();
                QTimer::singleShot(0, [this]() { updatePropertiesPanel(); });

                QString typeName;
                switch (type) {
                    case InterpolatedCurve::InterpolationType::NIFS3: typeName = "NIFS3"; break;
                    case InterpolatedCurve::InterpolationType::OIFS3: typeName = "OIFS3"; break;
                    case InterpolatedCurve::InterpolationType::Lagrange: typeName = "Lagrange"; break;
                    case InterpolatedCurve::InterpolationType::Barycentric: typeName = "Barycentric"; break;
                }
                statusBar()->showMessage(QString("Zmieniono typ interpolacji na %1").arg(typeName));
            });

        addUnifiedCurveVisibilityControls(std::static_pointer_cast<Curve>(curve));
        addUnifiedCurveStyleControls(std::static_pointer_cast<Curve>(curve));
    }
    else if (auto bezierCurve = std::dynamic_pointer_cast<BezierCurve>(selectedObject)) {
        m_propertiesLayout->addWidget(new QLabel("<b>Właściwości krzywej Béziera</b>"));
        m_propertiesLayout->addWidget(new QLabel(QString("Liczba punktów kontrolnych: %1")
                                                .arg(QString::number(bezierCurve->getControlPointCount()))));
        m_propertiesLayout->addWidget(new QLabel(QString("Stopień krzywej: %1")
                                                .arg(QString::number(bezierCurve->getDegree()))));

        addUnifiedCurveVisibilityControls(std::static_pointer_cast<Curve>(bezierCurve));
        addUnifiedCurveStyleControls(std::static_pointer_cast<Curve>(bezierCurve));
    }
    else {
        m_propertiesLayout->addWidget(new QLabel("Wybierz obiekt"));
    }

    m_propertiesLayout->addStretch();
}

void MainWindow::onWeightChanged(int pointIndex, double weight) {
    auto selectedObj = m_canvas->selectedObject();
    if (auto bezierCurve = std::dynamic_pointer_cast<BezierCurve>(selectedObj)) {
        bezierCurve->setWeight(pointIndex, weight);
        m_canvas->refresh();
        statusBar()->showMessage(QString("Zmieniono wagę punktu %1 na %2").arg(std::to_string(pointIndex)).arg(std::to_string(weight)));
    }
    updatePropertiesPanel();
}

void MainWindow::onWeightsReset() {
    auto selectedObj = m_canvas->selectedObject();
    if (auto bezierCurve = std::dynamic_pointer_cast<BezierCurve>(selectedObj)) {
        bezierCurve->resetWeights();
        m_weightsPanel->updateWeights();
        m_canvas->refresh();
        statusBar()->showMessage("Zresetowano wagi do 1.0");
    }
}

void MainWindow::onWeightsNormalized() {
    auto selectedObj = m_canvas->selectedObject();
    if (auto bezierCurve = std::dynamic_pointer_cast<BezierCurve>(selectedObj)) {
        bezierCurve->normalizeWeights();
        m_weightsPanel->updateWeights();
        m_canvas->refresh();
        statusBar()->showMessage("Znormalizowano wagi");
    }
}

void MainWindow::updateWeightsPanel() {
    if (const auto bezierCurve = std::dynamic_pointer_cast<BezierCurve>(m_canvas->selectedObject())) {
        m_weightsPanel->setShowWeights(bezierCurve->getShowWeights());
        m_weightsPanel->setBezierCurve(bezierCurve);
    }
}

void MainWindow::onShowWeightsToggled(bool show) {
    auto selectedObj = m_canvas->selectedObject();
    if (auto bezierCurve = std::dynamic_pointer_cast<BezierCurve>(selectedObj)) {
        bezierCurve->setShowWeights(show);
        m_canvas->refresh();
        statusBar()->showMessage(show ? "Włączono wyświetlanie wag" : "Wyłączono wyświetlanie wag");
    }
}

void WeightsPanel::createLayout() {
    m_mainLayout = new QVBoxLayout(this);

    m_showWeightsCheckBox->setChecked(false);

    titleLabel->setStyleSheet("font-weight: bold; font-size: 12px;");

    m_scrollArea = new QScrollArea;
    m_scrollArea->setWidgetResizable(true);
    m_scrollArea->setHorizontalScrollBarPolicy(Qt::ScrollBarAlwaysOff);

    m_weightsWidget = new QWidget;
    m_weightsLayout = new QVBoxLayout(m_weightsWidget);
    m_scrollArea->setWidget(m_weightsWidget);
    m_mainLayout->addWidget(m_scrollArea);

    auto* controlsGroup = new QGroupBox("Operacje na wagach");
    auto* controlsLayout = new QVBoxLayout(controlsGroup);

    m_resetButton = new QPushButton("Resetuj (1.0)");
    m_normalizeButton = new QPushButton("Normalizuj");

    controlsLayout->addWidget(m_resetButton);
    controlsLayout->addWidget(m_normalizeButton);

    m_mainLayout->addStretch();

void WeightsPanel::connectSignals() {
    connect(m_resetButton, &QPushButton::clicked, this, &WeightsPanel::onResetWeights);
    connect(m_normalizeButton, &QPushButton::clicked, this, &WeightsPanel::onNormalizeWeights);
    connect(m_showWeightsCheckBox, &QCheckBox::toggled, this, &WeightsPanel::showWeightsToggled);
}

void WeightsPanel::setBezierCurve(std::shared_ptr<BezierCurve> curve) {
    m_curve = curve;
    updateWeights();
    setEnabled(curve != nullptr);
}

void WeightsPanel::updateWeights() {
    if (!m_curve) {
        clear();
        return;
    }

    updateWeightControls();
}

void WeightsPanel::updateWeightControls() {
    for (auto* spinBox : m_weightSpinBoxes)
        spinBox->deleteLater();

    for (auto* label : m_pointLabels)
        label->deleteLater();


    m_weightSpinBoxes.clear();
    m_pointLabels.clear();

    if (!m_curve || !m_curve->hasControlPoints()) return;

    size_t pointCount = m_curve->getControlPointCount();

    for (size_t i = 0; i < pointCount; ++i) {
        auto* layout = new QHBoxLayout;

        auto* label = new QLabel(QString("P%1:").arg(std::to_string(i)));
        label->setFixedWidth(30);
        m_pointLabels.push_back(label);
        layout->addWidget(label);

        auto* spinBox = new QDoubleSpinBox;
        spinBox->setRange(0.01, 100.0);
        spinBox->setValue(m_curve->getWeight(i));
        spinBox->setDecimals(3);
        spinBox->setSingleStep(0.1);
        spinBox->setProperty("pointIndex", static_cast<int>(i));

        connect(spinBox, QOverload<double>::of(&QDoubleSpinBox::valueChanged),
                this, &WeightsPanel::onWeightChanged);

        m_weightSpinBoxes.push_back(spinBox);
        layout->addWidget(spinBox);

        m_weightsLayout->addLayout(layout);
    }
}

void WeightsPanel::clear() {
    for (auto* spinBox : m_weightSpinBoxes)
        spinBox->deleteLater();

    for (auto* label : m_pointLabels)
        label->deleteLater();

    m_weightSpinBoxes.clear();
    m_pointLabels.clear();
}

void WeightsPanel::onWeightChanged() {
    auto* spinBox = qobject_cast<QDoubleSpinBox*>(sender());
    if (!spinBox || !m_curve) return;

    int pointIndex = spinBox->property("pointIndex").toInt();
    double weight = spinBox->value();

    emit weightChanged(pointIndex, weight);
}

void WeightsPanel::onResetWeights() {
    if (m_curve) emit weightsReset();
}

void WeightsPanel::onNormalizeWeights() {
    if (m_curve) emit weightsNormalized();
}

void WeightsPanel::setShowWeights(const bool show) const {
    m_showWeightsCheckBox->setChecked(show);
}
