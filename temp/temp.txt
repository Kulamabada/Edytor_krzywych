bool CompoundCommand::execute() {
    bool success = true;
    size_t executedCommands = 0;

    for (size_t i = 0; i < m_commands.size(); ++i) {
        if (!m_commands[i]->execute()) {
            success = false;
            executedCommands = i;
            break;
        }
        executedCommands = i + 1;
    }

    if (!success) {
        for (size_t i = 0; i < executedCommands; ++i) {
            size_t index = executedCommands - i - 1;
            m_commands[index]->undo();
        }
    }

    return success;
}

bool CompoundCommand::undo() {
    bool success = true;

    for (const auto& m_command : std::ranges::reverse_view(m_commands))
        if (!m_command->undo()) {
            success = false;
            break;
        }

    return success;
}

bool CompoundCommand::redo() {
    bool success = true;

    for (const auto& command : m_commands)
        if (!command->redo()) {
            success = false;
            break;
        }

    return success;
}


AddControlPointCommand::AddControlPointCommand(std::shared_ptr<InterpolatedCurve> curve, const Point& point, int index)
    : CurveCommand(curve), m_point(point), m_index(index), m_addedIndex(-1) {
}

bool AddControlPointCommand::execute() {
    if (!m_curve)
        return false;

    if (m_index < 0 || m_index > static_cast<int>(m_curve->getPointCount())) {
        m_curve->addPoint(m_point);
        m_addedIndex = static_cast<int>(m_curve->getPointCount()) - 1;
    } else {
        m_curve->insertPoint(m_index, m_point);
        m_addedIndex = m_index;
    }

    return true;
}

bool AddControlPointCommand::undo() {
    if (!m_curve || m_addedIndex < 0 ||
        m_addedIndex >= static_cast<int>(m_curve->getPointCount())) {
        return false;
    }

    m_curve->removePoint(m_addedIndex);
    return true;
}

                                               int pointIndex,

DeleteControlPointCommand::DeleteControlPointCommand(std::shared_ptr<InterpolatedCurve> curve,
                                                  int pointIndex)
    : CurveCommand(curve), m_pointIndex(pointIndex) {

    if (m_curve && m_pointIndex >= 0 && m_pointIndex < static_cast<int>(m_curve->getPointCount())) {
        const auto& points = m_curve->getPoints();
        m_deletedPoint = points[m_pointIndex];
    }
}

bool DeleteControlPointCommand::execute() {
    if (!m_curve || m_pointIndex < 0 ||
        m_pointIndex >= static_cast<int>(m_curve->getPointCount())) {
        return false;
    }

    if (m_curve->getPointCount() < 3)
        return false;

    m_curve->removePoint(m_pointIndex);
    return true;
}

bool DeleteControlPointCommand::undo() {
    if (!m_curve)
        return false;

    if (m_pointIndex >= static_cast<int>(m_curve->getPointCount()))
        m_curve->addPoint(m_deletedPoint);
    else
        m_curve->insertPoint(m_pointIndex, m_deletedPoint);

    return true;
}

    return true;

    return true;
}

#include "CommandManager.h"
#include "AbstractCommand.h"
#include <vector>

bool CommandManager::executeCommand(std::unique_ptr<AbstractCommand> command) {
    if (!command)
        return false;

    if (const bool success = command->execute(); !success)
        return false;

    if (!m_redoStack.empty())
        m_redoStack.clear();

    m_undoStack.push_back(std::move(command));
    return true;
}

bool CommandManager::canUndo() const {
    return !m_undoStack.empty();
}

bool CommandManager::canRedo() const {
    return !m_redoStack.empty();
}

bool CommandManager::undo() {
    if (!canUndo())
        return false;

    auto command = std::move(m_undoStack.back());
    m_undoStack.pop_back();

    const bool success = command->undo();

    m_redoStack.push_back(std::move(command));
    return success;
}

bool CommandManager::redo() {
    if (!canRedo())
        return false;

    auto command = std::move(m_redoStack.back());
    m_redoStack.pop_back();

    const bool success = command->redo();

    m_undoStack.push_back(std::move(command));
    return success;
}

void CommandManager::clearHistory() {
    m_undoStack.clear();
    m_redoStack.clear();
}

int CommandManager::undoStackSize() const {
    return static_cast<int>(m_undoStack.size());
}

int CommandManager::redoStackSize() const {
    return static_cast<int>(m_redoStack.size());
}

ImageObject::ImageObject(const QString& filePath) : m_image(QImage(filePath)) {
    const double width = m_image.width() / 20.0;  // Skalowanie do współrzędnych świata
    const double height = m_image.height() / 20.0;

    // Domyślnie tworzymy prostokąt o środku w (0,0)
    m_cornerPoints = {
        Point(-width/2, -height/2, 0),  // Lewy górny
        Point(width/2, -height/2, 0),   // Prawy górny
        Point(width/2, height/2, 0),    // Prawy dolny
        Point(-width/2, height/2, 0)    // Lewy dolny
    };
}

void ImageObject::setCornerPoints(const std::array<Point, 4>& points) {
    if (m_locked) return;
    m_cornerPoints = points;
}

void ImageObject::setOpacity(double opacity) {
    m_opacity = std::max(0.0, std::min(1.0, opacity));
}

void ImageObject::setLocked(bool locked) {
    m_locked = locked;
}

Vector ImageObject::getPosition() const {
    // Zwracamy środek prostokąta
    Vector sum = Vector::Zero();
    for (const auto& p : m_cornerPoints) {
        sum += Vector(p.x(), p.y(), p.z());
    }
    return sum / 4.0;
}

void ImageObject::setPosition(const Eigen::Vector3d& position) {
    if (m_locked) return;

    Vector currentPos = getPosition();
    Vector delta = position - currentPos;

    for (auto& p : m_cornerPoints) {
        p.setX(p.x() + delta.x());
        p.setY(p.y() + delta.y());
        p.setZ(p.z() + delta.z());
    }
}

void ImageObject::transform(const TransformMatrix& matrix) {
    if (m_locked) return;

    for (auto& p : m_cornerPoints) {
        p.transform(matrix);
    }
}

void InterpolatedCurve::addPoint(const Point& point) {
    points.push_back(point);
    needsRebuild = true;
}

void InterpolatedCurve::insertPoint(size_t index, const Point& point) {
    if (index <= points.size()) {
        points.insert(points.begin() + index, point);
        needsRebuild = true;
    }
}

void InterpolatedCurve::removePoint(size_t index) {
    if (index < points.size()) {
        points.erase(points.begin() + index);
        needsRebuild = true;
    }
}

void InterpolatedCurve::setPoints(const std::vector<Point>& newPoints) {
    points = newPoints;
    needsRebuild = true;
}

void InterpolatedCurve::movePoint(size_t index, const Point& newPosition) {
    if (index < points.size()) {
        points[index] = newPosition;
        needsRebuild = true;
    }
}

const std::vector<Point>& InterpolatedCurve::getPoints() const {
    return points;
}

size_t InterpolatedCurve::getPointCount() const {
    return points.size();
}

void InterpolatedCurve::setInterpolationType(InterpolationType newType) {
    if (type != newType) {
        type = newType;
        needsRebuild = true;
    }
}

InterpolatedCurve::InterpolationType InterpolatedCurve::getInterpolationType() const { return type; }

void InterpolatedCurve::rebuildInterpolation() const {
    if (!needsRebuild || points.size() < 2) {
        needsRebuild = false;
        return;
    }
    coefficients.clear();

    switch (type) {
        case InterpolationType::NIFS3:
            buildNIFS3();
        break;
        case InterpolationType::OIFS3:
            buildOIFS3();
        break;
        case InterpolationType::Lagrange:
            buildLagrange();
        break;
        case InterpolationType::Barycentric:
            buildBarycentric();
        break;
    }

    needsRebuild = false;
}

void InterpolatedCurve::buildNIFS3() const {
    const size_t n = points.size();
    if (n < 2) return;

    // Parametryzacja od 0 do 1
    std::vector<double> xs(n), ys_x(n), ys_y(n);
    for (size_t i = 0; i < n; ++i) {
        xs[i] = static_cast<double>(i) / (n - 1);
        ys_x[i] = points[i].x();
        ys_y[i] = points[i].y();
    }

    // Obliczanie ilorazów różnicowych
    std::vector<double> ds_x;
    std::vector<double> ds_y;
    for (size_t i = 1; i < n - 1; ++i) {
        const double term1_x = (ys_x[i + 1] - ys_x[i]) / (xs[i + 1] - xs[i]);
        const double term2_x = (ys_x[i] - ys_x[i - 1]) / (xs[i] - xs[i - 1]);
        const double term1_y = (ys_y[i + 1] - ys_y[i]) / (xs[i + 1] - xs[i]);
        const double term2_y = (ys_y[i] - ys_y[i - 1]) / (xs[i] - xs[i - 1]);
        ds_x.push_back(6.0 * (term1_x - term2_x) / (xs[i + 1] - xs[i - 1]));
        ds_y.push_back(6.0 * (term1_y - term2_y) / (xs[i + 1] - xs[i - 1]));
    }

    // Współczynniki p
    std::vector<double> ps;
    ps.push_back(0.0);
    for (size_t i = 1; i < n - 1; ++i) {
        double lambda = (xs[i] - xs[i - 1]) / (xs[i + 1] - xs[i - 1]);
        ps.push_back((lambda - 1.0) / (lambda * ps[i - 1] + 2.0));
    }

    // Współczynniki q dla x i y
    std::vector<double> qs_x;
    std::vector<double> qs_y;
    qs_x.push_back(0.0);
    qs_y.push_back(0.0);
    for (size_t i = 1; i < n - 1; ++i) {
        const double lambda = (xs[i] - xs[i - 1]) / (xs[i + 1] - xs[i - 1]);
        qs_x.push_back((ds_x[i - 1] - lambda * qs_x[i - 1]) / (lambda * ps[i - 1] + 2.0));
        qs_y.push_back((ds_y[i - 1] - lambda * qs_y[i - 1]) / (lambda * ps[i - 1] + 2.0));
    }

    // Momenty M - dokładnie jak w Pythonie
    std::vector<double> M_x(n, 0.0), M_y(n, 0.0);

    // Uzupełniamy wartości od końca, zgodnie z algorytmem Pythona
    M_x[n-1] = 0.0;  // Naturalne warunki brzegowe
    M_y[n-1] = 0.0;
    M_x[n-2] = qs_x.back();
    M_y[n-2] = qs_y.back();

    for (int i = n - 3; i > 0; --i) {
        M_x[i] = ps[i] * M_x[i+1] + qs_x[i];
        M_y[i] = ps[i] * M_y[i+1] + qs_y[i];
    }
    // M_x[0] i M_y[0] pozostają równe 0 (naturalne warunki brzegowe)

    // Obliczanie współczynników wielomianów
    coefficients.clear();
    for (size_t i = 0; i < n - 1; ++i) {
        double h = xs[i + 1] - xs[i];
        Eigen::Matrix<double, 4, 2> coef;

        // Obliczamy współczynniki dla wielomianu w formie standardowej
        // a + b*t + c*t^2 + d*t^3
        coef(0, 0) = ys_x[i];
        coef(1, 0) = (ys_x[i + 1] - ys_x[i]) - (h * h / 6.0) * (M_x[i + 1] + 2.0 * M_x[i]);
        coef(2, 0) = (h * h / 2.0) * M_x[i];
        coef(3, 0) = (h * h / 6.0) * (M_x[i + 1] - M_x[i]);

        coef(0, 1) = ys_y[i];
        coef(1, 1) = (ys_y[i + 1] - ys_y[i]) - (h * h / 6.0) * (M_y[i + 1] + 2.0 * M_y[i]);
        coef(2, 1) = (h * h / 2.0) * M_y[i];
        coef(3, 1) = (h * h / 6.0) * (M_y[i + 1] - M_y[i]);

        coefficients.push_back(coef);
    }
}

void InterpolatedCurve::buildOIFS3() const {
    const size_t n = points.size();
    if (n < 3) return;

    Eigen::MatrixXd A = Eigen::MatrixXd::Zero(n, n);
    Eigen::VectorXd bx = Eigen::VectorXd::Zero(n);
    Eigen::VectorXd by = Eigen::VectorXd::Zero(n);

    for (size_t i = 0; i < n; ++i) {
        size_t prev = (i > 0) ? (i - 1) : (n - 1);
        size_t next = (i < n - 1) ? (i + 1) : 0;

        A(i, prev) = 1.0;
        A(i, i) = 4.0;
        A(i, next) = 1.0;

        bx(i) = 3.0 * (points[next].x() - points[prev].x());
        by(i) = 3.0 * (points[next].y() - points[prev].y());
    }

    Eigen::VectorXd cx = A.colPivHouseholderQr().solve(bx);
    Eigen::VectorXd cy = A.colPivHouseholderQr().solve(by);

    for (size_t i = 0; i < n; ++i) {
        size_t next = (i < n - 1) ? (i + 1) : 0;

        Eigen::Matrix<double, 4, 2> coef;

        coef(0, 0) = points[i].x();
        coef(1, 0) = cx(i);
        coef(2, 0) = 3.0 * (points[next].x() - points[i].x()) - 2.0 * cx(i) - cx(next);
        coef(3, 0) = 2.0 * (points[i].x() - points[next].x()) + cx(i) + cx(next);

        coef(0, 1) = points[i].y();
        coef(1, 1) = cy(i);
        coef(2, 1) = 3.0 * (points[next].y() - points[i].y()) - 2.0 * cy(i) - cy(next);
        coef(3, 1) = 2.0 * (points[i].y() - points[next].y()) + cy(i) + cy(next);

        coefficients.push_back(coef);
    }
}

void InterpolatedCurve::buildLagrange() const {
    const size_t n = points.size();
    if (n < 2) return;
}

void InterpolatedCurve::buildBarycentric() const {
    const size_t n = points.size();
    if (n < 2) return;
}

Vector InterpolatedCurve::getTangentAt(double t) const {
    const double epsilon = 1e-6;
    Point p1 = getPointAt(t);
    Point p2 = getPointAt(t + epsilon);

    return Vector(
        (p2.x() - p1.x()) / epsilon,
        (p2.y() - p1.y()) / epsilon,
        (p2.z() - p1.z()) / epsilon
    );
}

Point InterpolatedCurve::getPointAt(double t) const {
    if (points.empty()) return Point(0.0, 0.0, 0.0);
    if (points.size() == 1) return points[0];

    if (needsRebuild) {
        rebuildInterpolation();
    }

    t = std::max(0.0, std::min(1.0, t));

    size_t numSegments = (type == InterpolationType::OIFS3) ? points.size() : (points.size() - 1);
    double segmentLength = 1.0 / numSegments;
    size_t segmentIndex = std::min(static_cast<size_t>(t / segmentLength), numSegments - 1);
    double localT = (t - segmentIndex * segmentLength) / segmentLength;

    if (type == InterpolationType::NIFS3 || type == InterpolationType::OIFS3) {
        if (segmentIndex >= coefficients.size())
            return points[points.size() - 1];

        const auto& coef = coefficients[segmentIndex];
        Eigen::Vector4d tVec;
        tVec << 1.0, localT, localT*localT, localT*localT*localT;

        double x = tVec.dot(coef.col(0));
        double y = tVec.dot(coef.col(1));

        return Point(x, y, 0.0);
    }
    else if (type == InterpolationType::Lagrange) {
        double x = 0.0;
        double y = 0.0;

        for (size_t i = 0; i < points.size(); ++i) {
            double L = 1.0;
            for (size_t j = 0; j < points.size(); ++j) {
                if (i != j) {
                    double tj = static_cast<double>(j) / (points.size() - 1);
                    double ti = static_cast<double>(i) / (points.size() - 1);
                    L *= (t - tj) / (ti - tj);
                }
            }
            x += points[i].x() * L;
            y += points[i].y() * L;
        }

        return Point(x, y, 0.0);
    }
    else if (type == InterpolationType::Barycentric) {
        double x = 0.0;
        double y = 0.0;
        double sumW = 0.0;

        std::vector<double> weights(points.size());
        for (size_t i = 0; i < points.size(); ++i) {
            double ti = static_cast<double>(i) / (points.size() - 1);
            weights[i] = 1.0;
            for (size_t j = 0; j < points.size(); ++j) {
                if (i != j) {
                    double tj = static_cast<double>(j) / (points.size() - 1);
                    weights[i] *= (ti - tj);
                }
            }
            weights[i] = 1.0 / weights[i];
        }

        for (size_t i = 0; i < points.size(); ++i) {
            double ti = static_cast<double>(i) / (points.size() - 1);

            if (std::abs(t - ti) < 1e-10) {
                return points[i];
            }

            double w = weights[i] / (t - ti);
            x += points[i].x() * w;
            y += points[i].y() * w;
            sumW += w;
        }
        return Point(x / sumW, y / sumW, 0.0);
    }

    size_t nextIndex = (segmentIndex + 1) % points.size();
    double oneMinusT = 1.0 - localT;
    double x = oneMinusT * points[segmentIndex].x() + localT * points[nextIndex].x();
    double y = oneMinusT * points[segmentIndex].y() + localT * points[nextIndex].y();
    double z = oneMinusT * points[segmentIndex].z() + localT * points[nextIndex].z();

    return Point(x, y, z);
}



double InterpolatedCurve::getLength() const {
    constexpr int steps = 200;
    double length = 0.0;
    Point prev = getPointAt(0.0);

    for (int i = 1; i <= steps; ++i) {
        const double t = static_cast<double>(i) / steps;
        Point curr = getPointAt(t);
        length += prev.distanceTo(curr);
        prev = curr;
    }

    return length;
}

void InterpolatedCurve::transform(const TransformMatrix& matrix) {
    for (auto& point : points) {
        point.transform(matrix);
    }
    needsRebuild = true;
}

std::unique_ptr<Curve> InterpolatedCurve::clone() const {
    auto copy = std::make_unique<InterpolatedCurve>(mathProvider, type);
    copy->setPoints(points);
    return copy;
}

Vector InterpolatedCurve::getPosition() const {
    if (points.empty()) return Vector::Zero();

    Vector sum = Vector::Zero();
    for (const auto& p : points) {
        sum += Vector(p.x(), p.y(), p.z());
    }
    return sum / static_cast<double>(points.size());
}

void InterpolatedCurve::setPosition(const Eigen::Vector3d& position) {
    if (points.empty()) return;

    Vector delta = position - getPosition();
    for (auto& p : points) {
        p.setX(p.x() + delta.x());
        p.setY(p.y() + delta.y());
        p.setZ(p.z() + delta.z());
    }
    needsRebuild = true;
}

double Point::distanceTo(const Point& other) const {
    const double dx = other.x() - x();
    const double dy = other.y() - y();
    const double dz = other.z() - z();
    return std::sqrt(dx * dx + dy * dy + dz * dz);
}

Point Point::midPoint(const Point& other) const {
    return Point(
        (x() + other.x()) / 2.0,
        (y() + other.y()) / 2.0,
        (z() + other.z()) / 2.0
    );
}

void Point::transform(const TransformMatrix& matrix) {
    Eigen::Vector4d homogeneousPoint;
    homogeneousPoint << x(), y(), z(), 1.0;

    Eigen::Vector4d transformedPoint = matrix.getMatrix() * homogeneousPoint;

    if (std::abs(transformedPoint(3)) > 1e-10) {
        setX(transformedPoint(0) / transformedPoint(3));
        setY(transformedPoint(1) / transformedPoint(3));
        setZ(transformedPoint(2) / transformedPoint(3));
    }
}

        m_selectedControlPointIndex = -1;

        emit objectSelected(object);
        update();
    }

int Canvas::hitTestControlPoint(const QPointF& pos, double tolerance) {
    auto curve = getSelectedCurve();
    if (!curve || !m_viewportManager) return -1;

    const auto& points = curve->getPoints();
    for (size_t i = 0; i < points.size(); ++i) {
        QPointF screenPos = m_viewportManager->worldToScreen(
            QPointF(points[i].x(), points[i].y()));

        if (QLineF(pos, screenPos).length() <= tolerance) {
            return static_cast<int>(i);
        }
    }

    return -1;
}

bool Canvas::addControlPoint(const QPointF& screenPos) {
    const auto curve = getSelectedCurve();
    if (!curve || !m_viewportManager) return false;

    const QPointF worldPos = m_viewportManager->screenToWorld(screenPos);

    curve->addPoint(Point(worldPos.x(), worldPos.y()));

    const int newPointIndex = static_cast<int>(curve->getPointCount()) - 1;

    selectControlPoint(newPointIndex);

    emit controlPointAdded(newPointIndex);

    update();
    return true;
}

bool Canvas::moveControlPoint(int pointIndex, const QPointF& screenPos) {
    auto curve = getSelectedCurve();
    if (!curve || !m_viewportManager ||
        pointIndex < 0 || pointIndex >= static_cast<int>(curve->getPointCount())) {
        return false;
    }

    QPointF worldPos = m_viewportManager->screenToWorld(screenPos);

    const auto& points = curve->getPoints();
    auto newPoint = Point(worldPos.x(), worldPos.y(), points[pointIndex].z());
    curve->movePoint(pointIndex, newPoint);

    emit controlPointMoved(pointIndex);

    update();
    return true;
}

void Canvas::selectControlPoint(const int pointIndex) {
    const auto curve = getSelectedCurve();
    if (!curve) return;

    if (pointIndex >= 0 && pointIndex < static_cast<int>(curve->getPointCount())) {
        m_selectedControlPointIndex = pointIndex;
        emit controlPointSelected(pointIndex);
    } else {
        m_selectedControlPointIndex = -1;
    }

    update();
}

void Canvas::startAddingControlPoints() {
    setEditMode(EditMode::AddPoint);
}

void Canvas::startMovingControlPoints() {
    setEditMode(EditMode::MovePoint);
}

void Canvas::paintEvent(QPaintEvent* event)
{
    Q_UNUSED(event);

    if (!isVisible() || width() <= 0 || height() <= 0) return;

    QPainter painter(this);
    if (!painter.isActive()) {
        qWarning() << "Nie można zainicjować QPainter w Canvas::paintEvent";
        return;
    }

    painter.fillRect(rect(), Qt::white);

    if (!m_renderer) return;

    m_renderer->beginRender(&painter, m_viewportManager.get());

    m_renderer->renderGrid(1.0);
    m_renderer->renderAxes();

    for (const auto& object : m_objects)
        m_renderer->renderObject(*object);

    if (m_selectedObject) {
        const auto* geometricObj = dynamic_cast<const GeometricObject*>(m_selectedObject.get());
        if (geometricObj) {
            auto position = geometricObj->getPosition();
            QPointF selectionPos = m_viewportManager->worldToScreen(QPointF(position.x(), position.y()));
            m_renderer->renderSelectionMarker(selectionPos);
        }
    }

    if (m_isEditingPoints && m_selectedObject)
        drawControlPoints();

    m_renderer->endRender();
}

void Canvas::drawControlPoints() {
    const auto curve = getSelectedCurve();
    if (!curve || !m_renderer) return;

    const auto& points = curve->getPoints();

    m_renderer->renderControlPoints(points, m_selectedControlPointIndex,m_isDraggingPoint ? m_selectedControlPointIndex : -1);
}

void Canvas::resizeEvent(QResizeEvent* event)
{
    if (m_viewportManager) {
        m_viewportManager->setWindowSize(event->size().width(), event->size().height());
        emit viewportChanged();
    }
    QWidget::resizeEvent(event);
}

void Canvas::mousePressEvent(QMouseEvent* event)
{
    m_lastMousePos = event->pos();

    if (event->button() == Qt::LeftButton)
    {
        switch (m_editMode)
        {
            case EditMode::AddPoint:
                addControlPoint(event->pos());
                return;

            case EditMode::MovePoint:
                if (const int pointIndex = hitTestControlPoint(event->pos()); pointIndex >= 0) {
                    selectControlPoint(pointIndex);
                    m_isDraggingPoint = true;
                    setCursor(Qt::ClosedHandCursor);
                    return;
                }
                break;

            case EditMode::SelectPoint:
                if (const int pointIndex = hitTestControlPoint(event->pos()); pointIndex >= 0) {
                    selectControlPoint(pointIndex);
                    return;
                }
                break;

            case EditMode::None:
                    break;

            default:
                qWarning() << "Nieobsługiwany tryb edycji:" << static_cast<int>(m_editMode);
            break;
        }
    }
    else if (event->button() == Qt::MiddleButton) {
        m_isPanning = true;
        setCursor(Qt::ClosedHandCursor);
    }

    QWidget::mousePressEvent(event);
}

void Canvas::mouseMoveEvent(QMouseEvent* event)
{
    const QPointF currentPos = event->pos();
    const QPointF delta = currentPos - m_lastMousePos;
    if (m_isDraggingPoint && m_selectedControlPointIndex >= 0) {
        moveControlPoint(m_selectedControlPointIndex, currentPos);
    }
    else if (m_isPanning) {
        if (m_viewportManager) {
            m_viewportManager->pan(delta);
            emit viewportChanged();
            update();
        }
    }
    else if (m_editMode == EditMode::MovePoint) {
        if (const int pointIndex = hitTestControlPoint(currentPos); pointIndex >= 0) {
            setCursor(Qt::OpenHandCursor);
        } else {
            setCursor(Qt::ArrowCursor);
        }
    }

    m_lastMousePos = currentPos;
    QWidget::mouseMoveEvent(event);
}

void Canvas::mouseReleaseEvent(QMouseEvent* event)
{
    if (event->button() == Qt::LeftButton) {
        if (m_isDraggingPoint) {
            m_isDraggingPoint = false;

            if (m_editMode == EditMode::MovePoint) {
                setCursor(Qt::OpenHandCursor);
            } else {
                setCursor(Qt::ArrowCursor);
            }
        }
    }
    else if (event->button() == Qt::MiddleButton) {
        m_isPanning = false;
        setCursor(Qt::ArrowCursor);
    }

    QWidget::mouseReleaseEvent(event);
}

void Canvas::wheelEvent(QWheelEvent* event)
{
    if (m_viewportManager) {
        QPointF center = event->position();

        double factor = event->angleDelta().y() > 0 ? 1.1 : 0.9;

        m_viewportManager->zoom(factor, center);
        emit viewportChanged();
        update();
    }

    event->accept();
}

void Canvas::keyPressEvent(QKeyEvent* event)
{
    if (event->key() == Qt::Key_Escape) {
        finishEditing();
    }
    else if (event->key() == Qt::Key_Delete && m_selectedControlPointIndex >= 0) {
        auto curve = getSelectedCurve();
        if (curve && m_selectedControlPointIndex >= 0 &&
            m_selectedControlPointIndex < static_cast<int>(curve->getPointCount())) {

            if (curve->getPointCount() > 2) {
                curve->removePoint(m_selectedControlPointIndex);
                m_selectedControlPointIndex = -1;
                update();
            } else {
                QMessageBox::warning(this, "Usuwanie punktu",
                    "Krzywa musi mieć co najmniej 2 punkty.");
            }
        }
    }

    QWidget::keyPressEvent(event);
}

QtRenderer::QtRenderer() = default;

void QtRenderer::beginRender(QPainter* painter, ViewportManager* viewportManager)
{
    if (!painter || !viewportManager) {
        qDebug() << "Nieprawidłowy painter lub viewportManager";
        return;
    }
    if (!painter->isActive()) {
        qWarning() << "QPainter nie jest aktywny w beginRender()";
        return;
    }
    m_painter = painter;
    m_viewportManager = viewportManager;

    m_painter->save();
    m_painter->setTransform(viewportManager->viewTransform());
    m_painter->setRenderHint(QPainter::Antialiasing, true);
}

void QtRenderer::endRender()
{
    if (m_painter) {
        m_painter->restore();

        m_painter = nullptr;
        m_viewportManager = nullptr;
    }
}

void QtRenderer::renderPoint(const Point& point)
{
    if (!m_painter || !m_viewportManager || !m_painter->isActive())
        return;

    QPen pen(Qt::NoPen);
    QBrush brush(m_pointColor);
    m_painter->setPen(pen);
    m_painter->setBrush(brush);

    const double x = point.x(), y = point.y();

    m_painter->drawEllipse(QPointF(x, y), m_pointSize, m_pointSize);
}

void QtRenderer::renderCurve(const Curve& curve)
{
    if (!m_painter || !m_viewportManager)
        return;

    const auto interpolatedCurve = dynamic_cast<const InterpolatedCurve*>(&curve);
    if (!interpolatedCurve) return;

    if (interpolatedCurve->getDrawInterpolatedCurve()) {
        QColor color = interpolatedCurve->getColor();
        double width = interpolatedCurve->getLineWidth();
        Qt::PenStyle style = interpolatedCurve->getLineStyle();

        const QPen pen(color, width, style);
        m_painter->setPen(pen);
        m_painter->setBrush(Qt::NoBrush);

        constexpr int numSegments = 5000;
        QPolygonF points;

        for (int i = 0; i <= numSegments; ++i) {
            double t = static_cast<double>(i) / numSegments;
            Point p = curve.getPointAt(t);
            points.append(QPointF(p.x(), p.y()));
        }

        m_painter->drawPolyline(points);
    }

    if (interpolatedCurve->getDrawPolyline())
        renderPolyline(interpolatedCurve->getPoints());

    if (interpolatedCurve->getDrawConvexHull())
        renderConvexHull(interpolatedCurve->getPoints());
}

void QtRenderer::renderObject(const GeometricObject& object)
{
    if (const auto* point = dynamic_cast<const Point*>(&object)) {
        renderPoint(*point);
        return;
    }

    if (const auto* curve = dynamic_cast<const Curve*>(&object))
        renderCurve(*curve);
}

void QtRenderer::renderGrid(double gridSize) {
    if (!m_drawGrid || !m_painter) return;

    QRectF visibleRect = m_viewportManager->visibleWorldRect();

    QPen smallPen(QColor(230, 230, 230), 0.5, Qt::DotLine);
    m_painter->setPen(smallPen);

    for (double x = std::floor(visibleRect.left()); x <= visibleRect.right(); x += 1.0) {
        m_painter->drawLine(QPointF(x, visibleRect.top()), QPointF(x, visibleRect.bottom()));
    }
    for (double y = std::floor(visibleRect.top()); y <= visibleRect.bottom(); y += 1.0) {
        m_painter->drawLine(QPointF(visibleRect.left(), y), QPointF(visibleRect.right(), y));
    }

    QPen mainPen(QColor(200, 200, 200), 1.0, Qt::SolidLine);
    m_painter->setPen(mainPen);

    for (double x = std::floor(visibleRect.left() / 5) * 5; x <= visibleRect.right(); x += 5.0) {
        m_painter->drawLine(QPointF(x, visibleRect.top()), QPointF(x, visibleRect.bottom()));
    }
    for (double y = std::floor(visibleRect.top() / 5) * 5; y <= visibleRect.bottom(); y += 5.0) {
        m_painter->drawLine(QPointF(visibleRect.left(), y), QPointF(visibleRect.right(), y));
    }
}

void QtRenderer::renderText(const QPointF& position, const QString& text) {
    if (!m_painter) return;
    m_painter->save();
    QTransform current = m_painter->transform();
    m_painter->resetTransform();
    QPointF screenPos = current.map(position);
    m_painter->drawText(screenPos, text);
    m_painter->restore();
}

void QtRenderer::renderAxes()
{
    if (!m_drawAxes || !m_painter || !m_viewportManager) return;

    QPen pen(m_axesColor, m_axesWidth, Qt::SolidLine);
    m_painter->setPen(pen);

    QRectF visibleRect = m_viewportManager->visibleWorldRect();

    m_painter->drawLine(QPointF(visibleRect.left(), 0), QPointF(visibleRect.right(), 0));
    m_painter->drawLine(QPointF(0, visibleRect.top()), QPointF(0, visibleRect.bottom()));

    double arrowSize = 7.0 / m_viewportManager->zoomFactor();

    QPolygonF arrowX;
    arrowX << QPointF(visibleRect.right(), 0)
          << QPointF(visibleRect.right() - arrowSize, arrowSize / 2)
          << QPointF(visibleRect.right() - arrowSize, -arrowSize / 2);
    m_painter->setBrush(m_axesColor);
    m_painter->drawPolygon(arrowX);

    QPolygonF arrowY;
    arrowY << QPointF(0, visibleRect.bottom())
          << QPointF(arrowSize / 2, visibleRect.bottom() - arrowSize)
          << QPointF(-arrowSize / 2, visibleRect.bottom() - arrowSize);
    m_painter->drawPolygon(arrowY);

    renderText(QPointF(visibleRect.right() - 2 * arrowSize, -arrowSize), "x");
    renderText(QPointF(arrowSize, visibleRect.bottom() - 2 * arrowSize), "y");

    renderAxisTicks();
}

void QtRenderer::renderAxisTicks() {
    if (!m_painter || !m_viewportManager || !m_painter->isActive()) return;

    QRectF visibleRect = m_viewportManager->visibleWorldRect();
    double zoomFactor = m_viewportManager->zoomFactor();

    double viewportWidth = visibleRect.width();
    double viewportHeight = visibleRect.height();

    double desiredTickCount = 16.0;
    double optimalSpacingX = viewportWidth / desiredTickCount;
    double optimalSpacingY = viewportHeight / desiredTickCount;

    auto findNiceSpacing = [](double optimalSpacing) -> double {
        double magnitude = pow(10, floor(log10(optimalSpacing)));
        double normalized = optimalSpacing / magnitude;

        if (normalized < 1.2) return 1.0 * magnitude;
        if (normalized < 2.5) return 2.0 * magnitude;
        if (normalized < 7.0) return 5.0 * magnitude;
        return 10.0 * magnitude;
    };

    double tickSpacingX = findNiceSpacing(optimalSpacingX);
    double tickSpacingY = findNiceSpacing(optimalSpacingY);

    double xMin = floor(visibleRect.left() / tickSpacingX) * tickSpacingX;
    double xMax = ceil(visibleRect.right() / tickSpacingX) * tickSpacingX;
    double yMin = floor(visibleRect.top() / tickSpacingY) * tickSpacingY;
    double yMax = ceil(visibleRect.bottom() / tickSpacingY) * tickSpacingY;

    double tickLength = 5.0 / zoomFactor;

    auto determinePrecision = [](double spacing) -> int {
        if (spacing >= 1.0) return 0;
        return std::min(6, static_cast<int>(std::ceil(std::abs(std::log10(spacing)))));
    };

    int xPrecision = determinePrecision(tickSpacingX);
    int yPrecision = determinePrecision(tickSpacingY);

    for (double x = xMin; x <= xMax; x += tickSpacingX) {
        if (std::abs(x) < 1e-6) continue;

        m_painter->drawLine(QPointF(x, -tickLength/2), QPointF(x, tickLength/2));
        QString label = QString::number(x, 'f', xPrecision);
        while (label.contains('.') && label.endsWith('0'))
            label.chop(1);

        if (label.endsWith('.'))
            label.chop(1);

        renderText(QPointF(x, -tickLength - 5.0/zoomFactor), label);
    }

    for (double y = yMin; y <= yMax; y += tickSpacingY) {
        if (std::abs(y) < 1e-6) continue;

        m_painter->drawLine(QPointF(-tickLength/2, y), QPointF(tickLength/2, y));
        QString label = QString::number(y, 'f', yPrecision);
        while (label.contains('.') && label.endsWith('0'))
            label.chop(1);

        if (label.endsWith('.'))
            label.chop(1);

        renderText(QPointF(-tickLength - 15.0/zoomFactor, y), label);
    }
}

void QtRenderer::renderSelectionMarker(const QPointF& position)
{
    if (!m_painter || !m_painter->isActive())
        return;

    m_painter->save();
    m_painter->resetTransform();

    QPen pen(m_selectionColor, 1.0, Qt::SolidLine);
    m_painter->setPen(pen);
    m_painter->setBrush(Qt::NoBrush);

    double size = m_selectionSize;
    m_painter->drawEllipse(position, size, size);

    m_painter->drawLine(position - QPointF(size / 2, 0),
                       position + QPointF(size / 2, 0));
    m_painter->drawLine(position - QPointF(0, size / 2),
                       position + QPointF(0, size / 2));

    m_painter->restore();
}

void QtRenderer::renderControlPoint(const QPointF& position, bool isSelected, bool isActive)
{
    if (!m_painter || !m_painter->isActive()) {
        return;
    }

    m_painter->save();
    m_painter->resetTransform();

    QColor fillColor;
    if (isActive) {
        fillColor = m_controlPointActiveColor;
    } else if (isSelected) {
        fillColor = m_controlPointSelectedColor;
    } else {
        fillColor = m_controlPointNormalColor;
    }

    QPen shadowPen(Qt::black, 1.0);
    QBrush shadowBrush(QColor(0, 0, 0, 50));
    m_painter->setPen(shadowPen);
    m_painter->setBrush(shadowBrush);
    m_painter->drawEllipse(position + QPointF(1, 1), m_controlPointSize + 1, m_controlPointSize + 1);

    QPen pen(Qt::black, 1.0);
    QBrush brush(fillColor);
    m_painter->setPen(pen);
    m_painter->setBrush(brush);
    m_painter->drawEllipse(position, m_controlPointSize, m_controlPointSize);

    if (isSelected) {
        m_painter->setPen(Qt::white);
        QFont font = m_painter->font();
        font.setPointSize(9);
        m_painter->setFont(font);

        QFontMetrics fm(font);
        QString text = QString::number(isActive ? 1 : 0);
        QRect textRect = fm.boundingRect(text);
        QPointF textPos = position - QPointF(textRect.width() / 2.0, -textRect.height() / 4.0);

        m_painter->drawText(textPos, text);
    }

    m_painter->restore();
}

void QtRenderer::renderControlPoints(const std::vector<Point>& points, int selectedIndex, int draggingIndex)
{
    if (!m_painter || !m_viewportManager || points.empty())
        return;

    if (m_drawControlPolygon && points.size() >= 2)
        renderControlPolygon(points);

    std::vector<QPointF> screenPositions;
    screenPositions.reserve(points.size());

    for (const auto& point : points)
        screenPositions.push_back(m_viewportManager->worldToScreen(QPointF(point.x(), point.y())));

    for (size_t i = 0; i < points.size(); ++i)
        if (static_cast<int>(i) != selectedIndex && static_cast<int>(i) != draggingIndex)
            renderControlPoint(screenPositions[i], false, false);

    if (selectedIndex >= 0 && selectedIndex < static_cast<int>(points.size()))
        renderControlPoint(screenPositions[selectedIndex], true, false);

    if (draggingIndex >= 0 && draggingIndex < static_cast<int>(points.size()) && draggingIndex != selectedIndex)
        renderControlPoint(screenPositions[draggingIndex], false, true);


    m_painter->save();
    m_painter->resetTransform();
    QPen textPen(Qt::black);
    m_painter->setPen(textPen);

    QFont font = m_painter->font();
    font.setPointSize(8);
    m_painter->setFont(font);

    for (size_t i = 0; i < points.size(); ++i) {
        QString label = QString::number(i);
        QFontMetrics fm(font);
        QRect textRect = fm.boundingRect(label);
        QPointF textPos = screenPositions[i] + QPointF(m_controlPointSize + 2,
                                                  m_controlPointSize / 2.0);

        QRectF bgRect = QRectF(textPos.x() - 1, textPos.y() - textRect.height(),
                              textRect.width() + 2, textRect.height() + 2);
        m_painter->fillRect(bgRect, QColor(255, 255, 255, 180));

        m_painter->drawText(textPos, label);
    }

    m_painter->restore();
}

void QtRenderer::renderControlPolygon(const std::vector<Point>& points)
{
    if (!m_painter || !m_viewportManager || points.size() < 2)
        return;

    m_painter->save();
    m_painter->resetTransform();

    QPen pen(m_controlPolygonColor, m_controlPolygonWidth, m_controlPolygonStyle);
    m_painter->setPen(pen);
    m_painter->setBrush(Qt::NoBrush);

    QPolygonF polygon;
    for (const auto& point : points) {
        QPointF screenPos = m_viewportManager->worldToScreen(QPointF(point.x(), point.y()));
        polygon << screenPos;
    }

    m_painter->drawPolyline(polygon);

    m_painter->restore();
}

void QtRenderer::renderConvexHull(const std::vector<Point>& points)

    if (!m_painter || !m_viewportManager || points.size() < 3)
        return;

    // TODO: Implementacja algorytmu otoczki wypukłej (np. Graham Scan)
    // Na razie placeholder - rysuj prostokąt otaczający

    QPen pen(m_convexHullColor, m_convexHullWidth, m_convexHullStyle);
    m_painter->setPen(pen);
    m_painter->setBrush(Qt::NoBrush);

    // Znajdź bounding box jako placeholder
    double minX = points[0].x(), maxX = points[0].x();
    double minY = points[0].y(), maxY = points[0].y();

    for (const auto& point : points) {
        minX = std::min(minX, point.x());
        maxX = std::max(maxX, point.x());
        minY = std::min(minY, point.y());
        maxY = std::max(maxY, point.y());
    }

    QRectF boundingRect(minX, minY, maxX - minX, maxY - minY);
    m_painter->drawRect(boundingRect);

    // Dodaj tekst informujący że to placeholder
    m_painter->save();
    m_painter->resetTransform();
    QPointF screenPos = m_viewportManager->worldToScreen(QPointF(minX, maxY));
    m_painter->setPen(Qt::black);
    QFont font = m_painter->font();
    font.setPointSize(8);
    m_painter->setFont(font);
    m_painter->drawText(screenPos + QPointF(5, -5), "Convex Hull (placeholder)");
    m_painter->restore();
}

// Stałe domyślne
constexpr double DEFAULT_MIN_ZOOM = 0.1;
constexpr double DEFAULT_MAX_ZOOM = 100.0;
constexpr double DEFAULT_ZOOM_FACTOR = 1.0;
constexpr QPointF DEFAULT_PAN_OFFSET(0.0, 0.0);
constexpr QRectF DEFAULT_VIEWPORT(-10.0, -10.0, 20.0, 20.0);

ViewportManager::ViewportManager()
    : m_viewport(DEFAULT_VIEWPORT)
    , m_windowWidth(800)
    , m_windowHeight(600)
    , m_zoomFactor(DEFAULT_ZOOM_FACTOR)
    , m_panOffset(DEFAULT_PAN_OFFSET)
    , m_minZoom(DEFAULT_MIN_ZOOM)
    , m_maxZoom(DEFAULT_MAX_ZOOM)
{
    updateTransformations();
}

ViewportManager::~ViewportManager() = default;

void ViewportManager::pan(const QPointF& delta)
{
    const QPointF worldDelta = screenToWorld(delta) - screenToWorld(QPointF(0, 0));
    m_panOffset -= worldDelta;

    updateTransformations();
}

void ViewportManager::zoom(double factor, const QPointF& center)
{
    double newZoom = m_zoomFactor * factor;
    if (newZoom < m_minZoom) {
        factor = m_minZoom / m_zoomFactor;
        newZoom = m_minZoom;
    }
    else if (newZoom > m_maxZoom) {
        factor = m_maxZoom / m_zoomFactor;
        newZoom = m_maxZoom;
    }

    if (std::abs(factor - 1.0) < 1e-6)
        return;

    QPointF worldCenter = screenToWorld(center);
    m_zoomFactor = newZoom;

    QPointF newWorldCenter = screenToWorld(center);
    m_panOffset += worldCenter - newWorldCenter;

    updateTransformations();
}

void ViewportManager::resetView()
{
    m_zoomFactor = DEFAULT_ZOOM_FACTOR;
    m_panOffset = DEFAULT_PAN_OFFSET;
    m_viewport = DEFAULT_VIEWPORT;

    updateTransformations();
}

void ViewportManager::fitView(const QRectF& rect)
{
    if (rect.isEmpty())
        return;

    double scaleX = m_windowWidth / rect.width();
    double scaleY = m_windowHeight / rect.height();

    double newZoom = std::min(scaleX, scaleY) * 0.85;

    newZoom = std::max(m_minZoom, std::min(m_maxZoom, newZoom));

    m_zoomFactor = newZoom;
    m_panOffset = QPointF(rect.center().x(), rect.center().y());

    updateTransformations();
}

void ViewportManager::zoomIn(const QPointF& center) { zoom(1.2, center); }

void ViewportManager::zoomOut(const QPointF& center) { zoom(0.8, center); }

void ViewportManager::panBy(double dx, double dy)
{
    pan(QPointF(dx, dy));
}

QPointF ViewportManager::screenToWorld(const QPointF& screenPoint) const
{
    return m_inverseViewTransform.map(screenPoint);
}

{
    return m_viewTransform.map(worldPoint);
}

{
    return m_inverseViewTransform.mapRect(screenRect);
}

QRectF ViewportManager::worldToScreen(const QRectF& worldRect) const
{
    return m_viewTransform.mapRect(worldRect);
}

    return m_inverseViewTransform;

void ViewportManager::setMinZoom(const double minZoom)
{
    m_minZoom = minZoom > 0 ? minZoom : DEFAULT_MIN_ZOOM;
    if (m_zoomFactor < m_minZoom) {
        m_zoomFactor = m_minZoom;
        updateTransformations();
    }
}

void ViewportManager::setMaxZoom(const double maxZoom)
{
    m_maxZoom = maxZoom > m_minZoom ? maxZoom : DEFAULT_MAX_ZOOM;
    if (m_zoomFactor > m_maxZoom) {
        m_zoomFactor = m_maxZoom;
        updateTransformations();
    }
}

void ViewportManager::setZoomConstraints(const double minZoom, const double maxZoom)
{
    m_minZoom = minZoom > 0 ? minZoom : DEFAULT_MIN_ZOOM;
    m_maxZoom = maxZoom > m_minZoom ? maxZoom : DEFAULT_MAX_ZOOM;

    if (m_zoomFactor < m_minZoom) m_zoomFactor = m_minZoom;
    else if (m_zoomFactor > m_maxZoom) m_zoomFactor = m_maxZoom;
    updateTransformations();
}

void ViewportManager::updateTransformations()
{
    m_viewTransform.reset();
    m_viewTransform.translate(m_windowWidth / 2.0, m_windowHeight / 2.0);

    m_viewTransform.scale(m_zoomFactor, -m_zoomFactor);

    m_viewTransform.translate(-m_panOffset.x(), -m_panOffset.y());

    m_inverseViewTransform = m_viewTransform.inverted();
}

EigenMathAdapter::EigenMathAdapter() {
}

std::shared_ptr<TransformMatrix> EigenMathAdapter::createRotationMatrix(double angle, const std::array<double, 3>& axis) {
    Eigen::Vector3d eigenAxis(axis[0], axis[1], axis[2]);
    Eigen::AngleAxisd rotation(angle, eigenAxis.normalized());
    Eigen::Matrix4d matrix = Eigen::Matrix4d::Identity();
    matrix.block<3, 3>(0, 0) = rotation.toRotationMatrix();
    return std::make_shared<StandardTransformMatrix>(matrix);
}

std::shared_ptr<TransformMatrix> EigenMathAdapter::createTranslationMatrix(const std::array<double, 3>& translation) {
    Eigen::Matrix4d matrix = Eigen::Matrix4d::Identity();
    matrix(0, 3) = translation[0];
    matrix(1, 3) = translation[1];
    matrix(2, 3) = translation[2];
    return std::make_shared<StandardTransformMatrix>(matrix);
}

std::shared_ptr<TransformMatrix> EigenMathAdapter::createScalingMatrix(const std::array<double, 3>& factors) {
    Eigen::Matrix4d matrix = Eigen::Matrix4d::Identity();
    matrix(0, 0) = factors[0];
    matrix(1, 1) = factors[1];
    matrix(2, 2) = factors[2];
    return std::make_shared<StandardTransformMatrix>(matrix);
}

std::vector<double> EigenMathAdapter::solveLinearSystem(
    const std::vector<std::vector<double>>& coefficients,
    const std::vector<double>& constants) {

    Eigen::MatrixXd A = toEigenMatrix(coefficients);
    Eigen::VectorXd b = Eigen::Map<const Eigen::VectorXd>(constants.data(), constants.size());

    Eigen::VectorXd x = A.colPivHouseholderQr().solve(b);

    return std::vector<double>(x.data(), x.data() + x.size());
}

Eigen::MatrixXd EigenMathAdapter::toEigenMatrix(const std::vector<std::vector<double>>& matrix) const {
    if (matrix.empty()) return Eigen::MatrixXd(0, 0);

    size_t rows = matrix.size();
    size_t cols = matrix[0].size();

    Eigen::MatrixXd result(rows, cols);
    for (size_t i = 0; i < rows; ++i) {
        for (size_t j = 0; j < cols && j < matrix[i].size(); ++j) {
            result(i, j) = matrix[i][j];
        }
    }
    return result;
}

    if (m_mainWindow.getCanvas()->getSelectedCurve()) {
        m_mainWindow.getCanvas()->startMovingControlPoints();
        m_mainWindow.updateModeLabel();
        m_mainWindow.getStatusBar()->showMessage("Tryb przesuwania punktów kontrolnych. Kliknij i przeciągnij punkt kontrolny.");
    } else {
        QMessageBox::warning(&m_mainWindow, "Edycja punktów kontrolnych",
                            "Najpierw zaznacz krzywą, aby przesuwać jej punkty kontrolne.");
        m_mainWindow.m_selectModeAction->setChecked(true);
    }
}

void ActionHandler::onDeletePointAction() {
    if (auto curve = m_mainWindow.getCanvas()->getSelectedCurve()) {
        QKeyEvent keyEvent(QEvent::KeyPress, Qt::Key_Delete, Qt::NoModifier);
        QApplication::sendEvent(m_mainWindow.getCanvas(), &keyEvent);

        m_mainWindow.updateObjectsTree();
        m_mainWindow.updatePropertiesPanel();
        m_mainWindow.getStatusBar()->showMessage("Usunięto punkt kontrolny");
    } else {
        m_mainWindow.getStatusBar()->showMessage("Nie wybrano krzywej lub punktu kontrolnego");
    }
}

    curve->setColor(Qt::blue);
    curve->setLineWidth(1.0);
    curve->setLineStyle(Qt::SolidLine);

    curve->addPoint(Point(0, 0));
    curve->addPoint(Point(50, 100));
    curve->addPoint(Point(100, 0));
    m_mainWindow.getCanvas()->addObject(curve);
    m_mainWindow.getCanvas()->selectObject(curve);
    m_mainWindow.getCanvas()->refresh();
    m_mainWindow.updateObjectsTree();
    m_mainWindow.getStatusBar()->showMessage(QString("Utworzono krzywą interpolowaną typu %1").arg(typeName));
}

void MainWindow::connectSignals() {
    connect(m_canvas, &Canvas::objectAdded, this, &MainWindow::onObjectAdded);
    connect(m_canvas, &Canvas::objectRemoved, this, &MainWindow::onObjectRemoved);
    connect(m_canvas, &Canvas::objectSelected, this, &MainWindow::onObjectSelected);
    connect(m_canvas, &Canvas::selectionCleared, this, &MainWindow::onSelectionCleared);
    connect(m_canvas, &Canvas::controlPointAdded, this, &MainWindow::onControlPointAdded);
    connect(m_canvas, &Canvas::controlPointMoved, this, &MainWindow::onControlPointMoved);
    connect(m_canvas, &Canvas::controlPointSelected, this, &MainWindow::onControlPointSelected);
    connect(m_canvas, &Canvas::editingStarted, this, &MainWindow::onEditingStarted);
    connect(m_canvas, &Canvas::editingFinished, this, &MainWindow::onEditingFinished);

    connect(m_objectsTree, &QTreeWidget::itemClicked, this, [this](QTreeWidgetItem* item, int column) {
        if (item->parent() == nullptr) {
            int index = item->data(0, Qt::UserRole).toInt();
            if (index >= 0 && index < static_cast<int>(m_canvas->objects().size())) {
                m_canvas->selectObject(m_canvas->objects()[index]);
                m_canvas->finishEditing();
                m_selectModeAction->setChecked(true);
                updateModeLabel();
            }
        }
        else {
            int pointIndex = item->data(0, Qt::UserRole).toInt();
            int objectIndex = item->data(0, Qt::UserRole + 1).toInt();
            if (objectIndex >= 0 && objectIndex < static_cast<int>(m_canvas->objects().size())) {
                m_canvas->selectObject(m_canvas->objects()[objectIndex]);

                auto curve = m_canvas->getSelectedCurve();
                if (curve && pointIndex >= 0 && pointIndex < static_cast<int>(curve->getPointCount())) {
                    if (m_canvas->getEditMode() == Canvas::EditMode::None) {
                        m_canvas->setEditMode(Canvas::EditMode::SelectPoint);
                        m_selectModeAction->setChecked(true);
                    }
                    m_canvas->selectControlPoint(pointIndex);
                    updateModeLabel();
                }
            }
        }
    });
}

void MainWindow::updateObjectsTree() {
    m_objectsTree->clear();

    int index = 0;
    for (const auto& obj : m_canvas->objects()) {
        auto* item = new QTreeWidgetItem(m_objectsTree);
        item->setText(0, QString("Obiekt %1").arg(std::to_string(index + 1)));

        if (const auto curve = std::dynamic_pointer_cast<InterpolatedCurve>(obj)) {
            item->setText(1, "Krzywa");

            const auto& points = curve->getPoints();
            for (size_t i = 0; i < points.size(); ++i) {
                auto* pointItem = new QTreeWidgetItem(item);
                const Point& p = points[i];
                pointItem->setText(0, QString("Punkt %1").arg(std::to_string(i)));
                pointItem->setText(1, QString("(%1, %2)")
                    .arg(QString::number(p.x(), 'f', 2), QString::number(p.y(), 'f', 2)));

                pointItem->setData(0, Qt::UserRole, QVariant::fromValue(static_cast<int>(i)));
                pointItem->setData(0, Qt::UserRole + 1, QVariant::fromValue(index));
            }
        }
        else if (const auto image = std::dynamic_pointer_cast<ImageObject>(obj)) {
            item->setText(1, "Obraz");

            const auto& corners = image->getCornerPoints();
            for (size_t i = 0; i < corners.size(); ++i) {
                auto* cornerItem = new QTreeWidgetItem(item);
                const Point& p = corners[i];
                cornerItem->setText(0, QString("Narożnik %1").arg(std::to_string(i)));
                cornerItem->setText(1, QString("(%1, %2)")
                    .arg(QString::number(p.x(), 'f', 2),
                         QString::number(p.y(), 'f', 2)));
            }
        } else {
            item->setText(1, "Inny");
        }

        item->setData(0, Qt::UserRole, QVariant::fromValue(index));
        index++;
    }

    m_objectsTree->expandAll();
}

void MainWindow::updatePropertiesPanel() {
    QLayoutItem* child;
    while ((child = m_propertiesLayout->takeAt(0)) != nullptr) {
        delete child->widget();
        delete child;
    }

    auto selectedObject = m_canvas->selectedObject();
    if (!selectedObject) {
        m_propertiesLayout->addWidget(new QLabel("Wybierz obiekt, aby edytować jego właściwości"));
        return;
    }

    if (auto curve = std::dynamic_pointer_cast<InterpolatedCurve>(selectedObject)) {
        m_propertiesLayout->addWidget(new QLabel("<b>Właściwości krzywej interpolowanej</b>"));

        m_propertiesLayout->addWidget(new QLabel("Typ interpolacji:"));

        QComboBox* typeCombo = new QComboBox();
        typeCombo->addItem("NIFS3 (Natural Cubic Spline)", QVariant::fromValue(static_cast<int>(InterpolatedCurve::InterpolationType::NIFS3)));
        typeCombo->addItem("OIFS3 (Closed Cubic Spline)", QVariant::fromValue(static_cast<int>(InterpolatedCurve::InterpolationType::OIFS3)));
        typeCombo->addItem("Lagrange", QVariant::fromValue(static_cast<int>(InterpolatedCurve::InterpolationType::Lagrange)));
        typeCombo->addItem("Barycentric", QVariant::fromValue(static_cast<int>(InterpolatedCurve::InterpolationType::Barycentric)));

        typeCombo->setCurrentIndex(static_cast<int>(curve->getInterpolationType()));
        m_propertiesLayout->addWidget(typeCombo);

        connect(typeCombo, QOverload<int>::of(&QComboBox::currentIndexChanged),
            [this, curve](int index) {
                const auto type = static_cast<InterpolatedCurve::InterpolationType>(index);
                curve->setInterpolationType(type);
                m_canvas->refresh();
                QTimer::singleShot(0, [this]() {
                        updatePropertiesPanel();
                    });

                QString typeName;
                switch (type) {
                    case InterpolatedCurve::InterpolationType::NIFS3: typeName = "NIFS3"; break;
                    case InterpolatedCurve::InterpolationType::OIFS3: typeName = "OIFS3"; break;
                    case InterpolatedCurve::InterpolationType::Lagrange: typeName = "Lagrange"; break;
                    case InterpolatedCurve::InterpolationType::Barycentric: typeName = "Barycentric"; break;
                }

                statusBar()->showMessage(QString("Zmieniono typ interpolacji na %1").arg(typeName));
            });

        // === SEKCJA KONTROLI RENDEROWANIA ===
        m_propertiesLayout->addWidget(new QLabel("<b>Kontrola wyświetlania:</b>"));

        // Checkbox dla krzywej interpolowanej
        auto* drawCurveCheckBox = new QCheckBox("Rysuj krzywą interpolowaną");
        drawCurveCheckBox->setChecked(curve->getDrawInterpolatedCurve());
        m_propertiesLayout->addWidget(drawCurveCheckBox);

        connect(drawCurveCheckBox, &QCheckBox::toggled, [this, curve](bool checked) {
            curve->setDrawInterpolatedCurve(checked);
            m_canvas->refresh();
            statusBar()->showMessage(checked ? "Włączono rysowanie krzywej" : "Wyłączono rysowanie krzywej");
        });

        // Checkbox dla łamanej
        auto* drawPolylineCheckBox = new QCheckBox("Rysuj łamaną łączącą punkty");
        drawPolylineCheckBox->setChecked(curve->getDrawPolyline());
        m_propertiesLayout->addWidget(drawPolylineCheckBox);

        connect(drawPolylineCheckBox, &QCheckBox::toggled, [this, curve](bool checked) {
            curve->setDrawPolyline(checked);
            m_canvas->refresh();
            statusBar()->showMessage(checked ? "Włączono rysowanie łamanej" : "Wyłączono rysowanie łamanej");
        });

        // Checkbox dla otoczki wypukłej
        auto* drawConvexHullCheckBox = new QCheckBox("Rysuj otoczkę wypukłą punktów");
        drawConvexHullCheckBox->setChecked(curve->getDrawConvexHull());
        m_propertiesLayout->addWidget(drawConvexHullCheckBox);

        connect(drawConvexHullCheckBox, &QCheckBox::toggled, [this, curve](bool checked) {
            curve->setDrawConvexHull(checked);
            m_canvas->refresh();
            statusBar()->showMessage(checked ? "Włączono rysowanie otoczki wypukłej" : "Wyłączono rysowanie otoczki wypukłej");
        });

        // === WŁAŚCIWOŚCI KRZYWEJ INTERPOLOWANEJ ===
        m_propertiesLayout->addWidget(new QLabel("<b>Styl krzywej interpolowanej:</b>"));

        m_propertiesLayout->addWidget(new QLabel("Kolor linii:"));

        QPushButton* colorButton = new QPushButton();
        colorButton->setFixedSize(80, 25);
        QColor currentColor = curve->getColor();
        QString styleSheet = QString("background-color: %1;").arg(currentColor.name());
        colorButton->setStyleSheet(styleSheet);

        m_propertiesLayout->addWidget(colorButton);

        connect(colorButton, &QPushButton::clicked, [this, curve, colorButton]() {
            QColor color = QColorDialog::getColor(curve->getColor(), nullptr, "Wybierz kolor linii");
            if (color.isValid()) {
                curve->setColor(color);
                QString styleSheet = QString("background-color: %1;").arg(color.name());
                colorButton->setStyleSheet(styleSheet);
                m_canvas->refresh();
                statusBar()->showMessage("Zmieniono kolor linii");
            }
        });

        m_propertiesLayout->addWidget(new QLabel("Grubość linii:"));

        auto* widthSpinBox = new QDoubleSpinBox();
        widthSpinBox->setRange(0.1, 10.0);
        widthSpinBox->setSingleStep(0.1);
        widthSpinBox->setValue(curve->getLineWidth());

        m_propertiesLayout->addWidget(widthSpinBox);

        connect(widthSpinBox, QOverload<double>::of(&QDoubleSpinBox::valueChanged), [this, curve](double value) {
            curve->setLineWidth(value);
            m_canvas->refresh();
            statusBar()->showMessage(QString("Zmieniono grubość linii na %1").arg(std::to_string(value)));
        });

        m_propertiesLayout->addWidget(new QLabel("Styl linii:"));

        auto* lineStyleCombo = new QComboBox();
        lineStyleCombo->addItem("Ciągła", static_cast<int>(Qt::SolidLine));
        lineStyleCombo->addItem("Przerywana", static_cast<int>(Qt::DashLine));
        lineStyleCombo->addItem("Kropkowana", static_cast<int>(Qt::DotLine));
        lineStyleCombo->addItem("Kreska-kropka", static_cast<int>(Qt::DashDotLine));
        lineStyleCombo->addItem("Kreska-kropka-kropka", static_cast<int>(Qt::DashDotDotLine));

        int styleIndex = 0;
        switch (curve->getLineStyle()) {
            case Qt::SolidLine: styleIndex = 0; break;
            case Qt::DashLine: styleIndex = 1; break;
            case Qt::DotLine: styleIndex = 2; break;
            case Qt::DashDotLine: styleIndex = 3; break;
            case Qt::DashDotDotLine: styleIndex = 4; break;
            default: styleIndex = 0; break;
        }

        lineStyleCombo->setCurrentIndex(styleIndex);

        m_propertiesLayout->addWidget(lineStyleCombo);

        connect(lineStyleCombo, QOverload<int>::of(&QComboBox::currentIndexChanged), [this, curve, lineStyleCombo](int index) {
            Qt::PenStyle style = static_cast<Qt::PenStyle>(lineStyleCombo->itemData(index).toInt());
            curve->setLineStyle(style);
            m_canvas->refresh();
            statusBar()->showMessage("Zmieniono styl linii");
        });

        // === WŁAŚCIWOŚCI ŁAMANEJ ===
        m_propertiesLayout->addWidget(new QLabel("<b>Styl łamanej:</b>"));

        // Kolor łamanej
        m_propertiesLayout->addWidget(new QLabel("Kolor łamanej:"));
        QPushButton* polylineColorButton = new QPushButton();
        polylineColorButton->setFixedSize(80, 25);
        QColor currentPolylineColor = curve->getPolylineColor();
        polylineColorButton->setStyleSheet(QString("background-color: %1;").arg(currentPolylineColor.name()));
        m_propertiesLayout->addWidget(polylineColorButton);

        connect(polylineColorButton, &QPushButton::clicked, [this, curve, polylineColorButton]() {
            QColor color = QColorDialog::getColor(curve->getPolylineColor(), nullptr, "Wybierz kolor łamanej");
            if (color.isValid()) {
                curve->setPolylineColor(color);
                polylineColorButton->setStyleSheet(QString("background-color: %1;").arg(color.name()));

                // Ustaw styl w rendererze
                if (auto qtRenderer = dynamic_cast<QtRenderer*>(m_renderer.get())) {
                    qtRenderer->setPolylineStyle(color, curve->getPolylineWidth(), curve->getPolylineStyle());
                }

                m_canvas->refresh();
                statusBar()->showMessage("Zmieniono kolor łamanej");
            }
        });

        // Grubość łamanej
        m_propertiesLayout->addWidget(new QLabel("Grubość łamanej:"));
        auto* polylineWidthSpinBox = new QDoubleSpinBox();
        polylineWidthSpinBox->setRange(0.1, 10.0);
        polylineWidthSpinBox->setSingleStep(0.1);
        polylineWidthSpinBox->setValue(curve->getPolylineWidth());
        m_propertiesLayout->addWidget(polylineWidthSpinBox);

        connect(polylineWidthSpinBox, QOverload<double>::of(&QDoubleSpinBox::valueChanged), [this, curve](double value) {
            curve->setPolylineWidth(value);

            // Ustaw styl w rendererze
            if (auto qtRenderer = dynamic_cast<QtRenderer*>(m_renderer.get())) {
                qtRenderer->setPolylineStyle(curve->getPolylineColor(), value, curve->getPolylineStyle());
            }

            m_canvas->refresh();
            statusBar()->showMessage(QString("Zmieniono grubość łamanej na %1").arg(std::to_string(value)));
        });

        // === WŁAŚCIWOŚCI OTOCZKI WYPUKŁEJ ===
        m_propertiesLayout->addWidget(new QLabel("<b>Styl otoczki wypukłej:</b>"));

        // Kolor otoczki wypukłej
        m_propertiesLayout->addWidget(new QLabel("Kolor otoczki wypukłej:"));
        QPushButton* convexHullColorButton = new QPushButton();
        convexHullColorButton->setFixedSize(80, 25);
        QColor currentConvexHullColor = curve->getConvexHullColor();
        convexHullColorButton->setStyleSheet(QString("background-color: %1;").arg(currentConvexHullColor.name()));
        m_propertiesLayout->addWidget(convexHullColorButton);

        connect(convexHullColorButton, &QPushButton::clicked, [this, curve, convexHullColorButton]() {
            QColor color = QColorDialog::getColor(curve->getConvexHullColor(), nullptr, "Wybierz kolor otoczki wypukłej");
            if (color.isValid()) {
                curve->setConvexHullColor(color);
                convexHullColorButton->setStyleSheet(QString("background-color: %1;").arg(color.name()));

                // Ustaw styl w rendererze
                if (auto qtRenderer = dynamic_cast<QtRenderer*>(m_renderer.get())) {
                    qtRenderer->setConvexHullStyle(color, curve->getConvexHullWidth(), curve->getConvexHullStyle());
                }

                m_canvas->refresh();
                statusBar()->showMessage("Zmieniono kolor otoczki wypukłej");
            }
        });

        // Informacja o implementacji
        m_propertiesLayout->addWidget(new QLabel("<i>Uwaga: Otoczka wypukła to obecnie placeholder</i>"));

        m_propertiesLayout->addWidget(new QLabel(QString("Długość krzywej: %1")
            .arg(QString::number(curve->getLength(), 'f', 2))));
    }
    else {
        m_propertiesLayout->addWidget(new QLabel("Wybierz obiekt"));
    }
    m_propertiesLayout->addStretch();
}

void MainWindow::updateModeLabel() {
    switch (m_canvas->getEditMode()) {
        case Canvas::EditMode::None:
            m_modeLabel->setText("Tryb: Normalny");
            break;
        case Canvas::EditMode::SelectPoint:
            m_modeLabel->setText("Tryb: Zaznaczanie obiektu");
            break;
        case Canvas::EditMode::AddPoint:
            m_modeLabel->setText("Tryb: Dodawanie punktu");
            break;
        case Canvas::EditMode::MovePoint:
            m_modeLabel->setText("Tryb: Przesuwanie punktu");
            break;
    }
}

void MainWindow::onObjectAdded(std::shared_ptr<GeometricObject> obj) {
    updateObjectsTree();
    statusBar()->showMessage("Dodano nowy obiekt");
}

void MainWindow::onObjectRemoved(std::shared_ptr<GeometricObject> obj) {
    updateObjectsTree();
    statusBar()->showMessage("Usunięto obiekt");
}

void MainWindow::onObjectSelected(std::shared_ptr<GeometricObject> obj) {
    updateObjectsTree();
    updatePropertiesPanel();
    auto objects = m_canvas->objects();
    auto it = std::find(objects.begin(), objects.end(), obj);
    if (it != objects.end()) {
        int index = static_cast<int>(std::distance(objects.begin(), it));

        for (int i = 0; i < m_objectsTree->topLevelItemCount(); ++i) {
            QTreeWidgetItem* item = m_objectsTree->topLevelItem(i);
            if (item->data(0, Qt::UserRole).toInt() == index) {
                m_objectsTree->setCurrentItem(item);
                break;
            }
        }
    }

    statusBar()->showMessage("Wybrano obiekt");
}

void MainWindow::onSelectionCleared() {
    m_objectsTree->clearSelection();
    updatePropertiesPanel();
    statusBar()->showMessage("Anulowano zaznaczenie");
}

void MainWindow::onControlPointAdded(int pointIndex) {
    updateObjectsTree();
    updatePropertiesPanel();
    statusBar()->showMessage(QString("Dodano punkt kontrolny %1").arg(std::to_string(pointIndex)));
}

void MainWindow::onControlPointMoved(int pointIndex) {
    updateObjectsTree();
    updatePropertiesPanel();
    statusBar()->showMessage(QString("Przesunięto punkt kontrolny %1").arg(std::to_string(pointIndex)));
}

void MainWindow::onControlPointSelected(int pointIndex) {
    updateObjectsTree();

    auto selectedObj = m_canvas->selectedObject();
    auto objects = m_canvas->objects();
    auto it = std::find(objects.begin(), objects.end(), selectedObj);
    if (it != objects.end()) {
        int objectIndex = static_cast<int>(std::distance(objects.begin(), it));

        for (int i = 0; i < m_objectsTree->topLevelItemCount(); ++i) {
            QTreeWidgetItem* objectItem = m_objectsTree->topLevelItem(i);
            if (objectItem->data(0, Qt::UserRole).toInt() == objectIndex) {
                for (int j = 0; j < objectItem->childCount(); ++j) {
                    QTreeWidgetItem* pointItem = objectItem->child(j);
                    if (pointItem->data(0, Qt::UserRole).toInt() == pointIndex) {
                        m_objectsTree->setCurrentItem(pointItem);
                        break;
                    }
                }
                break;
            }
        }
    }

    statusBar()->showMessage(QString("Wybrano punkt kontrolny %1").arg(std::to_string(pointIndex)));
}

void MainWindow::onEditingStarted() {
    updateModeLabel();
    statusBar()->showMessage("Rozpoczęto edycję punktów kontrolnych");
}

void MainWindow::onEditingFinished() {
    m_selectModeAction->setChecked(true);
    updateModeLabel();
    statusBar()->showMessage("Zakończono edycję punktów kontrolnych");
}
