    if (command) {
        m_commands.push_back(std::move(command));
    }

bool CompoundCommand::execute() {
    bool success = true;
    size_t executedCommands = 0;

    // Wykonaj wszystkie komendy po kolei
    for (size_t i = 0; i < m_commands.size(); ++i) {
        if (!m_commands[i]->execute()) {
            success = false;
            executedCommands = i;
            break;
        }
        executedCommands = i + 1;
    }

    // Jeśli którakolwiek komenda nie powiodła się, cofnij wszystkie wykonane w odwrotnej kolejności
    if (!success) {
        for (size_t i = 0; i < executedCommands; ++i) {
            size_t index = executedCommands - i - 1;
            m_commands[index]->undo();
        }
    }

    return success;
}

bool CompoundCommand::undo() {
    bool success = true;

    // Cofnij wszystkie komendy w odwrotnej kolejności
    for (auto it = m_commands.rbegin(); it != m_commands.rend(); ++it) {
        if (!(*it)->undo()) {
            success = false;
            break;
        }
    }

    return success;
}

bool CompoundCommand::redo() {
    bool success = true;

    // Ponownie wykonaj wszystkie komendy po kolei
    for (auto& command : m_commands) {
        if (!command->redo()) {
            success = false;
            break;
        }
    }

    return success;
}

// Implementacja PIMPL
class CommandManager::CommandManagerImpl {
public:
    CommandManagerImpl() = default;
    ~CommandManagerImpl() = default;

    bool executeCommand(std::unique_ptr<AbstractCommand> command) {
        if (!command) {
            return false;
        }

        // Wykonanie komendy
        bool success = command->execute();
        if (!success) {
            return false;
        }

        // Po wykonaniu nowej komendy, czyścimy stos redo
        if (!m_redoStack.empty()) {
            m_redoStack.clear();
        }

        // Dodanie komendy do stosu undo
        m_undoStack.push_back(std::move(command));
        return true;
    }

    bool canUndo() const {
        return !m_undoStack.empty();
    }

    bool canRedo() const {
        return !m_redoStack.empty();
    }

    bool undo() {
        if (!canUndo()) {
            return false;
        }

        // Pobranie ostatniej komendy ze stosu undo
        auto command = std::move(m_undoStack.back());
        m_undoStack.pop_back();

        // Cofnięcie komendy
        bool success = command->undo();

        // Przeniesienie komendy na stos redo
        m_redoStack.push_back(std::move(command));
        return success;
    }

    bool redo() {
        if (!canRedo()) {
            return false;
        }

        // Pobranie ostatniej komendy ze stosu redo
        auto command = std::move(m_redoStack.back());
        m_redoStack.pop_back();

        // Ponowne wykonanie komendy
        bool success = command->redo();

        // Przeniesienie komendy na stos undo
        m_undoStack.push_back(std::move(command));
        return success;
    }

    void clearHistory() {
        m_undoStack.clear();
        m_redoStack.clear();
    }

    int undoStackSize() const {
        return static_cast<int>(m_undoStack.size());
    }

    int redoStackSize() const {
        return static_cast<int>(m_redoStack.size());
    }

private:
    std::vector<std::unique_ptr<AbstractCommand>> m_undoStack;
    std::vector<std::unique_ptr<AbstractCommand>> m_redoStack;
};

void InterpolatedCurve::rebuildInterpolation() const {
    if (!needsRebuild || points.size() < 2) {
        needsRebuild = false;
        return;
    }

    coefficients.clear();

    switch (type) {
        case InterpolationType::CubicSpline:
            buildCubicSpline();
            break;
        case InterpolationType::CatmullRom:
            buildCatmullRom();
            break;
        case InterpolationType::Linear:
            // Dla interpolacji liniowej nie potrzebujemy współczynników
            break;
    }

    needsRebuild = false;
}

void InterpolatedCurve::buildCubicSpline() const {
    const size_t n = points.size();
    if (n < 2) return;

    // Utwórz trójdiagonalną macierz dla obliczania współczynników spline
    Eigen::MatrixXd A = Eigen::MatrixXd::Zero(n, n);
    Eigen::VectorXd bx = Eigen::VectorXd::Zero(n);
    Eigen::VectorXd by = Eigen::VectorXd::Zero(n);

    // Naturalne warunki brzegowe
    A(0, 0) = 1.0;
    A(n-1, n-1) = 1.0;

    // Wypełnianie macierzy dla punktów wewnętrznych
    for (size_t i = 1; i < n - 1; ++i) {
        A(i, i-1) = 1.0;
        A(i, i) = 4.0;
        A(i, i+1) = 1.0;

        bx(i) = 3.0 * (points[i+1].x() - points[i-1].x());
        by(i) = 3.0 * (points[i+1].y() - points[i-1].y());
    }

    // Rozwiązanie układu równań dla pochodnych
    Eigen::VectorXd cx = A.colPivHouseholderQr().solve(bx);
    Eigen::VectorXd cy = A.colPivHouseholderQr().solve(by);

    // Obliczenie współczynników dla każdego segmentu
    for (size_t i = 0; i < n - 1; ++i) {
        Eigen::Matrix<double, 4, 2> coef;

        // Współczynniki dla x
        coef(0, 0) = points[i].x();
        coef(1, 0) = cx(i);
        coef(2, 0) = 3.0 * (points[i+1].x() - points[i].x()) - 2.0 * cx(i) - cx(i+1);
        coef(3, 0) = 2.0 * (points[i].x() - points[i+1].x()) + cx(i) + cx(i+1);

        // Współczynniki dla y
        coef(0, 1) = points[i].y();
        coef(1, 1) = cy(i);
        coef(2, 1) = 3.0 * (points[i+1].y() - points[i].y()) - 2.0 * cy(i) - cy(i+1);
        coef(3, 1) = 2.0 * (points[i].y() - points[i+1].y()) + cy(i) + cy(i+1);

        coefficients.push_back(coef);
    }
}

void InterpolatedCurve::buildCatmullRom() const {
    if (points.size() < 2) return;

    // Macierz bazowa Catmull-Rom
    Eigen::Matrix4d basis;
    basis << 0, 1, 0, 0,
             -0.5, 0, 0.5, 0,
             1, -2.5, 2, -0.5,
             -0.5, 1.5, -1.5, 0.5;

    // Dla każdego segmentu
    for (size_t i = 0; i < points.size() - 1; ++i) {
        // Punkty kontrolne z ekstrapolacją na krańcach
        Point p0 = (i > 0) ? points[i-1] : points[i];
        Point p1 = points[i];
        Point p2 = points[i+1];
        Point p3 = (i+1 < points.size()-1) ? points[i+2] : points[i+1];

        // Macierz punktów kontrolnych
        Eigen::Matrix<double, 4, 2> controls;
        controls << p0.x(), p0.y(),
                   p1.x(), p1.y(),
                   p2.x(), p2.y(),
                   p3.x(), p3.y();

        // Obliczenie współczynników
        coefficients.push_back(basis * controls);
    }
}

Point InterpolatedCurve::getPointAt(double t) const {
    if (points.empty()) return Point(0.0, 0.0, 0.0);
    if (points.size() == 1) return points[0];

    // Aktualizacja danych interpolacji jeśli potrzebna
    if (needsRebuild) {
        rebuildInterpolation();
    }

    // Ograniczenie parametru t do [0, 1]
    t = std::max(0.0, std::min(1.0, t));

    // Obliczenie segmentu i lokalnego parametru
    size_t numSegments = points.size() - 1;
    double segmentLength = 1.0 / numSegments;
    size_t segmentIndex = std::min(static_cast<size_t>(t / segmentLength), numSegments - 1);
    double localT = (t - segmentIndex * segmentLength) / segmentLength;

    if (type == InterpolationType::Linear) {
        // Prosta interpolacja liniowa
        double oneMinusT = 1.0 - localT;
        double x = oneMinusT * points[segmentIndex].x() + localT * points[segmentIndex + 1].x();
        double y = oneMinusT * points[segmentIndex].y() + localT * points[segmentIndex + 1].y();
        double z = oneMinusT * points[segmentIndex].z() + localT * points[segmentIndex + 1].z();
        return Point(x, y, z);
    } else {
        // Interpolacja krzywoliniowa
        if (segmentIndex >= coefficients.size()) {
            return points[segmentIndex]; // Awaryjnie
        }

        const auto& coef = coefficients[segmentIndex];
        Eigen::Vector4d tVec;
        tVec << 1.0, localT, localT*localT, localT*localT*localT;

        double x = tVec.dot(coef.col(0));
        double y = tVec.dot(coef.col(1));

        return Point(x, y, 0.0);
    }
}

Vector InterpolatedCurve::getTangentAt(double t) const {
    // Numeryczne obliczenie pochodnej
    const double epsilon = 1e-6;
    Point p1 = getPointAt(t);
    Point p2 = getPointAt(t + epsilon);

    return Vector(
        (p2.x() - p1.x()) / epsilon,
        (p2.y() - p1.y()) / epsilon,
        (p2.z() - p1.z()) / epsilon
    );
}

double InterpolatedCurve::getCurvatureAt(double t) const {
    // Obliczenie krzywizny przez przybliżenie drugiej pochodnej
    const double h = 1e-6;
    Vector first = getTangentAt(t);
    Vector second = (getTangentAt(t + h) - first) / h;

    double numerator = std::abs(first.x() * second.y() - first.y() * second.x());
    double denominator = std::pow(first.x() * first.x() + first.y() * first.y(), 1.5);

    return (denominator < 1e-10) ? 0.0 : (numerator / denominator);
}

bool InterpolatedCurve::isPointOnCurve(const Point& point, double tolerance) const {
    // Sprawdzenie przez próbkowanie krzywej
    const int steps = 50;
    for (int i = 0; i <= steps; ++i) {
        double t = static_cast<double>(i) / steps;
        if (getPointAt(t).distanceTo(point) <= tolerance) {
            return true;
        }
    }
    return false;
}

double InterpolatedCurve::getLength() const {
    // Przybliżenie długości przez sumowanie odcinków
    const int steps = 20;
    double length = 0.0;
    Point prev = getPointAt(0.0);

    for (int i = 1; i <= steps; ++i) {
        double t = static_cast<double>(i) / steps;
        Point curr = getPointAt(t);
        length += prev.distanceTo(curr);
        prev = curr;
    }

    return length;
}

std::vector<Point> InterpolatedCurve::getIntersectionsWith(const Curve& other) const {
    // Uproszczona implementacja - brute force z optymalizacją
    std::vector<Point> result;
    const int steps = 25;
    const double tolerance = 1e-4;

    for (int i = 0; i <= steps; ++i) {
        double t1 = static_cast<double>(i) / steps;
        Point p1 = getPointAt(t1);

        for (int j = 0; j <= steps; ++j) {
            double t2 = static_cast<double>(j) / steps;
            Point p2 = other.getPointAt(t2);

            if (p1.distanceTo(p2) < tolerance) {
                // Sprawdź czy punkt już istnieje w wynikach
                bool exists = false;
                for (const auto& p : result) {
                    if (p.distanceTo(p1) < tolerance) {
                        exists = true;
                        break;
                    }
                }
                if (!exists) {
                    result.push_back(p1);
                }
                break;
            }
        }
    }

    return result;
}

void InterpolatedCurve::transform(const TransformMatrix& matrix) {
    for (auto& point : points) {
        point.transform(matrix);
    }
    needsRebuild = true;
}

std::unique_ptr<Curve> InterpolatedCurve::clone() const {
    auto copy = std::make_unique<InterpolatedCurve>(mathProvider, type);
    copy->setPoints(points);
    return copy;
}

Vector InterpolatedCurve::getPosition() const {
    if (points.empty()) return Vector::Zero();

    Vector sum = Vector::Zero();
    for (const auto& p : points) {
        sum += Vector(p.x(), p.y(), p.z());
    }
    return sum / static_cast<double>(points.size());
}

void InterpolatedCurve::setPosition(const Eigen::Vector3d& position) {
    if (points.empty()) return;

    Vector delta = position - getPosition();
    for (auto& p : points) {
        p.setX(p.x() + delta.x());
        p.setY(p.y() + delta.y());
        p.setZ(p.z() + delta.z());
    }
    needsRebuild = true;
}

Vector InterpolatedCurve::getScale() const {
    return Vector::Ones();
}

void InterpolatedCurve::setScale(const Vector& scale) {
    if (points.empty()) return;

    Vector center = getPosition();
    for (auto& p : points) {
        Vector relPos(p.x() - center.x(), p.y() - center.y(), p.z() - center.z());
        p.setX(center.x() + relPos.x() * scale.x());
        p.setY(center.y() + relPos.y() * scale.y());
        p.setZ(center.z() + relPos.z() * scale.z());
    }
    needsRebuild = true;
}

Eigen::Quaterniond InterpolatedCurve::getRotation() const {
    return Eigen::Quaterniond::Identity();
}

void InterpolatedCurve::setRotation(const Eigen::Quaterniond& rotation) {
    if (points.empty()) return;

    Vector center = getPosition();
    for (auto& p : points) {
        Vector relPos(p.x() - center.x(), p.y() - center.y(), p.z() - center.z());
        Vector rotated = rotation * relPos;
        p.setX(center.x() + rotated.x());
        p.setY(center.y() + rotated.y());
        p.setZ(center.z() + rotated.z());
    }
    needsRebuild = true;
}

double Point::distanceTo(const Point& other) const {
    const double dx = other.x() - x();
    const double dy = other.y() - y();
    const double dz = other.z() - z();
    return std::sqrt(dx * dx + dy * dy + dz * dz);
}

Point Point::midPoint(const Point& other) const {
    return Point(
        (x() + other.x()) / 2.0,
        (y() + other.y()) / 2.0,
        (z() + other.z()) / 2.0
    );
}

void Point::transform(const TransformMatrix& matrix) {
    // Konwertujemy punkt do formatu Eigen::Vector4d (reprezentacja homogeniczna)
    Eigen::Vector4d homogeneousPoint;
    homogeneousPoint << x(), y(), z(), 1.0;

    // Pobieramy macierz transformacji i aplikujemy transformację
    Eigen::Vector4d transformedPoint = matrix.getMatrix() * homogeneousPoint;

    // Konwertujemy z powrotem do współrzędnych kartezjańskich (z normalizacją)
    if (std::abs(transformedPoint(3)) > 1e-10) {  // Unikamy dzielenia przez zero
        setX(transformedPoint(0) / transformedPoint(3));
        setY(transformedPoint(1) / transformedPoint(3));
        setZ(transformedPoint(2) / transformedPoint(3));
    }
}

void Canvas::paintEvent(QPaintEvent* event)
{
    Q_UNUSED(event);

    QPainter painter(this);
    // Sprawdzamy, czy painter został prawidłowo zainicjowany
    if (!painter.isActive()) {
        qWarning() << "Nie można zainicjować QPainter w Canvas::paintEvent";
        return;
    }
    painter.fillRect(rect(), Qt::white);

    if (!m_dPtr->renderer) {
        return;
    }

    m_dPtr->renderer->beginRender(&painter, m_dPtr->viewportManager.get());

    // Rysowanie elementów pomocniczych
    m_dPtr->renderer->renderGrid(1.0);
    m_dPtr->renderer->renderAxes();

    // Rysowanie obiektów geometrycznych
    for (const auto& object : m_dPtr->objects) {
        m_dPtr->renderer->renderObject(*object);
    }

    // Rysowanie zaznaczenia
    if (m_dPtr->selectedObject) {
        const auto* geometricObj = dynamic_cast<const GeometricObject*>(m_dPtr->selectedObject.get());
        if (geometricObj) {
            auto position = geometricObj->getPosition();
            QPointF selectionPos = m_dPtr->viewportManager->worldToScreen(QPointF(position.x(), position.y()));
            m_dPtr->renderer->renderSelectionMarker(selectionPos);
        }
    }

    m_dPtr->renderer->endRender();
}

{
    if (m_dPtr->viewportManager) {
        m_dPtr->viewportManager->setWindowSize(event->size().width(), event->size().height());
        emit viewportChanged();
    }

    QWidget::resizeEvent(event);
}

void Canvas::mousePressEvent(QMouseEvent* event)
{
    m_dPtr->lastMousePos = event->pos();

    if (event->button() == Qt::LeftButton) {
        m_dPtr->isSelecting = true;
        m_dPtr->selectionPoint = event->pos();

        // Tutaj można dodać kod do zaznaczania obiektów pod kursorem
    }
    else if (event->button() == Qt::MiddleButton ||
            (event->button() == Qt::LeftButton && event->modifiers() & Qt::ControlModifier)) {
        m_dPtr->isPanning = true;
        setCursor(Qt::ClosedHandCursor);
    }

    QWidget::mousePressEvent(event);
}

void Canvas::mouseMoveEvent(QMouseEvent* event)
{
    QPointF currentPos = event->pos();
    QPointF delta = currentPos - m_dPtr->lastMousePos;

    if (m_dPtr->isPanning) {
        if (m_dPtr->viewportManager) {
            m_dPtr->viewportManager->pan(delta);
            emit viewportChanged();
            update();
        }
    }

    m_dPtr->lastMousePos = currentPos;
    QWidget::mouseMoveEvent(event);
}

void Canvas::mouseReleaseEvent(QMouseEvent* event)
{
    if (event->button() == Qt::LeftButton) {
        if (m_dPtr->isSelecting) {
            // Tutaj logika zaznaczania obiektu
            m_dPtr->isSelecting = false;
        }
    }
    else if (event->button() == Qt::MiddleButton ||
            (event->button() == Qt::LeftButton && event->modifiers() & Qt::ControlModifier)) {
        m_dPtr->isPanning = false;
        setCursor(Qt::ArrowCursor);
    }

    QWidget::mouseReleaseEvent(event);
}

void Canvas::wheelEvent(QWheelEvent* event)
{
    if (m_dPtr->viewportManager) {
        QPointF center = event->position();

        double factor = event->angleDelta().y() > 0 ? 1.1 : 0.9;

        m_dPtr->viewportManager->zoom(factor, center);
        emit viewportChanged();
        update();
    }

    event->accept();
}

void QtRenderer::beginRender(QPainter* painter, ViewportManager* viewportManager)
{
    if (!painter || !viewportManager) {
        qDebug() << "Nieprawidłowy painter lub viewportManager";
        return;
    }

    if (!painter->isActive()) {
        qWarning() << "QPainter nie jest aktywny w beginRender()";
        return;
    }

    m_dPtr->painter = painter;
    m_dPtr->viewportManager = viewportManager;

    m_dPtr->painter->save();
    m_dPtr->painter->setTransform(viewportManager->viewTransform());
    m_dPtr->painter->setRenderHint(QPainter::Antialiasing, true);

        m_dPtr->painter->restore();

        m_dPtr->painter = nullptr;
        m_dPtr->viewportManager = nullptr;
    }
}

void QtRenderer::renderPoint(const Point& point)
{
    if (!m_dPtr->painter || !m_dPtr->viewportManager || !m_dPtr->painter->isActive()) {
        return;
    }

    QPen pen(Qt::NoPen);
    QBrush brush(m_dPtr->pointColor);
    m_dPtr->painter->setPen(pen);
    m_dPtr->painter->setBrush(brush);

    double x = point.x();
    double y = point.y();

    m_dPtr->painter->drawEllipse(QPointF(x, y), m_dPtr->pointSize, m_dPtr->pointSize);
}

void QtRenderer::renderCurve(const Curve& curve)
{
    if (!m_dPtr->painter || !m_dPtr->viewportManager) {
        return;
    }

    QPen pen(m_dPtr->curveColor, m_dPtr->curveWidth, m_dPtr->curveStyle);
    m_dPtr->painter->setPen(pen);
    m_dPtr->painter->setBrush(Qt::NoBrush);

    const int numSegments = 100;
    QPolygonF points;

    for (int i = 0; i <= numSegments; ++i) {
        double t = static_cast<double>(i) / numSegments;
        Point p = curve.getPointAt(t);
        points.append(QPointF(p.x(), p.y()));
    }

    m_dPtr->painter->drawPolyline(points);
}

void QtRenderer::renderObject(const GeometricObject& object)
{

    if (!m_dPtr->painter || !m_dPtr->painter->isActive()) {
        return;
    }

    const Point* point = dynamic_cast<const Point*>(&object);
    if (point) {
        renderPoint(*point);
        return;
    }

    const Curve* curve = dynamic_cast<const Curve*>(&object);
    if (curve) {
        renderCurve(*curve);
        return;
    }
}

void QtRenderer::renderGrid(double gridSize)
{
    if (!m_dPtr->drawGrid || !m_dPtr->painter || !m_dPtr->viewportManager) {
        return;
    }

    QPen pen(m_dPtr->gridColor, m_dPtr->gridWidth, m_dPtr->gridStyle);
    m_dPtr->painter->setPen(pen);

    QRectF visibleRect = m_dPtr->viewportManager->visibleWorldRect();

    double xMin = std::floor(visibleRect.left() / gridSize) * gridSize;
    double xMax = std::ceil(visibleRect.right() / gridSize) * gridSize;
    double yMin = std::floor(visibleRect.top() / gridSize) * gridSize;
    double yMax = std::ceil(visibleRect.bottom() / gridSize) * gridSize;

    // Rysuj linie pionowe
    for (double x = xMin; x <= xMax; x += gridSize) {
        m_dPtr->painter->drawLine(QPointF(x, yMin), QPointF(x, yMax));
    }

    // Rysuj linie poziome
    for (double y = yMin; y <= yMax; y += gridSize) {
        m_dPtr->painter->drawLine(QPointF(xMin, y), QPointF(xMax, y));
    }
}

void QtRenderer::renderAxes()
{
    if (!m_dPtr->drawAxes || !m_dPtr->painter || !m_dPtr->viewportManager) {
        return;
    }

    QPen pen(m_dPtr->axesColor, m_dPtr->axesWidth, Qt::SolidLine);
    m_dPtr->painter->setPen(pen);

    QRectF visibleRect = m_dPtr->viewportManager->visibleWorldRect();

    // Rysuj oś X
    m_dPtr->painter->drawLine(QPointF(visibleRect.left(), 0),
                             QPointF(visibleRect.right(), 0));

    // Rysuj oś Y
    m_dPtr->painter->drawLine(QPointF(0, visibleRect.top()),
                             QPointF(0, visibleRect.bottom()));

    // Dodaj strzałki wskazujące kierunek osi
    double arrowSize = 10.0 / m_dPtr->viewportManager->zoomFactor();

    // Strzałka dla osi X
    QPolygonF arrowX;
    arrowX << QPointF(visibleRect.right(), 0)
          << QPointF(visibleRect.right() - arrowSize, arrowSize / 2)
          << QPointF(visibleRect.right() - arrowSize, -arrowSize / 2);
    m_dPtr->painter->setBrush(m_dPtr->axesColor);
    m_dPtr->painter->drawPolygon(arrowX);

    // Strzałka dla osi Y
    QPolygonF arrowY;
    arrowY << QPointF(0, visibleRect.top())
          << QPointF(arrowSize / 2, visibleRect.top() + arrowSize)
          << QPointF(-arrowSize / 2, visibleRect.top() + arrowSize);
    m_dPtr->painter->drawPolygon(arrowY);

    // Dodaj etykiety osi
    m_dPtr->painter->drawText(QPointF(visibleRect.right() - arrowSize, -arrowSize), "x");
    m_dPtr->painter->drawText(QPointF(arrowSize, visibleRect.top() + arrowSize), "y");
}

void QtRenderer::renderSelectionMarker(const QPointF& position)
{
    if (!m_dPtr->painter || !m_dPtr->painter->isActive()) {
        return;
    }

    m_dPtr->painter->save();
    m_dPtr->painter->resetTransform();

    QPen pen(m_dPtr->selectionColor, 1.0, Qt::SolidLine);
    m_dPtr->painter->setPen(pen);
    m_dPtr->painter->setBrush(Qt::NoBrush);

    double size = m_dPtr->selectionSize;
    m_dPtr->painter->drawEllipse(position, size, size);

    // Narysuj krzyżyk w środku
    m_dPtr->painter->drawLine(position - QPointF(size / 2, 0),
                             position + QPointF(size / 2, 0));
    m_dPtr->painter->drawLine(position - QPointF(0, size / 2),
                             position + QPointF(0, size / 2));

    m_dPtr->painter->restore();
}

#include "ViewportManager.h"

// Stałe domyślne
const double DEFAULT_MIN_ZOOM = 0.1;
const double DEFAULT_MAX_ZOOM = 100.0;
const double DEFAULT_ZOOM_FACTOR = 1.0;
const QPointF DEFAULT_PAN_OFFSET(0.0, 0.0);
const QRectF DEFAULT_VIEWPORT(-10.0, -10.0, 20.0, 20.0);

ViewportManager::ViewportManager()
    : m_viewport(DEFAULT_VIEWPORT)
    , m_windowWidth(800)
    , m_windowHeight(600)
    , m_zoomFactor(DEFAULT_ZOOM_FACTOR)
    , m_panOffset(DEFAULT_PAN_OFFSET)
    , m_minZoom(DEFAULT_MIN_ZOOM)
    , m_maxZoom(DEFAULT_MAX_ZOOM)
{
    updateTransformations();
}

ViewportManager::~ViewportManager() = default;

void ViewportManager::pan(const QPointF& delta)
{
    // Obliczenie przesunięcia w koordynatach świata
    QPointF worldDelta = screenToWorld(delta) - screenToWorld(QPointF(0, 0));

    // Aktualizacja przesunięcia
    m_panOffset -= worldDelta;

    updateTransformations();
}

void ViewportManager::zoom(double factor, const QPointF& center)
{
    // Sprawdzenie ograniczeń zoomowania
    double newZoom = m_zoomFactor * factor;
    if (newZoom < m_minZoom) {
        factor = m_minZoom / m_zoomFactor;
        newZoom = m_minZoom;
    }
    else if (newZoom > m_maxZoom) {
        factor = m_maxZoom / m_zoomFactor;
        newZoom = m_maxZoom;
    }

    // Jeśli nie ma zmiany, zakończ
    if (std::abs(factor - 1.0) < 1e-6) {
        return;
    }

    // Pobierz punkt środkowy w koordynatach świata
    QPointF worldCenter = screenToWorld(center);

    // Aktualizacja współczynnika zoomowania
    m_zoomFactor = newZoom;

    // Korekta przesunięcia, aby punkt centralny pozostał w miejscu
    QPointF newWorldCenter = screenToWorld(center);
    m_panOffset += worldCenter - newWorldCenter;

    updateTransformations();
}

void ViewportManager::resetView()
{
    // Przywrócenie domyślnych wartości
    m_zoomFactor = DEFAULT_ZOOM_FACTOR;
    m_panOffset = DEFAULT_PAN_OFFSET;
    m_viewport = DEFAULT_VIEWPORT;

    updateTransformations();
}

void ViewportManager::fitView(const QRectF& rect)
{
    if (rect.isEmpty()) {
        return;
    }

    // Obliczenie współczynnika skalowania
    double scaleX = m_windowWidth / rect.width();
    double scaleY = m_windowHeight / rect.height();

    // Użycie mniejszego współczynnika z marginesem 10%
    double newZoom = std::min(scaleX, scaleY) * 0.9;

    // Ograniczenie zoomu do dozwolonych wartości
    newZoom = std::max(m_minZoom, std::min(m_maxZoom, newZoom));

    m_zoomFactor = newZoom;
    m_panOffset = QPointF(rect.center().x(), rect.center().y());

    updateTransformations();
}

void ViewportManager::zoomIn(const QPointF& center)
{
    zoom(1.2, center); // Zbliżenie o 20%
}

void ViewportManager::zoomOut(const QPointF& center)
{
    zoom(0.8, center); // Oddalenie o 20%
}

void ViewportManager::panBy(double dx, double dy)
{
    pan(QPointF(dx, dy));
}

QPointF ViewportManager::screenToWorld(const QPointF& screenPoint) const
{
    return m_inverseViewTransform.map(screenPoint);
}

QPointF ViewportManager::worldToScreen(const QPointF& worldPoint) const
{
    return m_viewTransform.map(worldPoint);
}

QRectF ViewportManager::screenToWorld(const QRectF& screenRect) const
{
    return m_inverseViewTransform.mapRect(screenRect);
}

QRectF ViewportManager::worldToScreen(const QRectF& worldRect) const
{
    return m_viewTransform.mapRect(worldRect);
}

QTransform ViewportManager::viewTransform() const
{
    return m_viewTransform;
}

QTransform ViewportManager::inverseViewTransform() const
{
    return m_inverseViewTransform;
}

double ViewportManager::zoomFactor() const
{
    return m_zoomFactor;
}

QPointF ViewportManager::panOffset() const
{
    return m_panOffset;
}

QRectF ViewportManager::visibleWorldRect() const
{
    return screenToWorld(QRectF(0, 0, m_windowWidth, m_windowHeight));
}

void ViewportManager::setViewport(const QRectF& viewport)
{
    m_viewport = viewport;
    updateTransformations();
}

void ViewportManager::setWindowSize(int width, int height)
{
    m_windowWidth = width;
    m_windowHeight = height;
    updateTransformations();
}

void ViewportManager::setZoomFactor(double factor)
{
    m_zoomFactor = std::max(m_minZoom, std::min(m_maxZoom, factor));
    updateTransformations();
}

void ViewportManager::setPanOffset(const QPointF& offset)
{
    m_panOffset = offset;
    updateTransformations();
}

void ViewportManager::setMinZoom(double minZoom)
{
    m_minZoom = minZoom > 0 ? minZoom : DEFAULT_MIN_ZOOM;
    if (m_zoomFactor < m_minZoom) {
        m_zoomFactor = m_minZoom;
        updateTransformations();
    }
}

void ViewportManager::setMaxZoom(double maxZoom)
{
    m_maxZoom = maxZoom > m_minZoom ? maxZoom : DEFAULT_MAX_ZOOM;
    if (m_zoomFactor > m_maxZoom) {
        m_zoomFactor = m_maxZoom;
        updateTransformations();
    }
}

void ViewportManager::setZoomConstraints(double minZoom, double maxZoom)
{
    m_minZoom = minZoom > 0 ? minZoom : DEFAULT_MIN_ZOOM;
    m_maxZoom = maxZoom > m_minZoom ? maxZoom : DEFAULT_MAX_ZOOM;

    // Dostosowanie aktualnego zoomu do nowych ograniczeń
    if (m_zoomFactor < m_minZoom) {
        m_zoomFactor = m_minZoom;
    }
    else if (m_zoomFactor > m_maxZoom) {
        m_zoomFactor = m_maxZoom;
    }

    updateTransformations();
}

void ViewportManager::updateTransformations()
{
    // Zresetowanie transformacji
    m_viewTransform.reset();

    // Sekwencja transformacji:

    // 1. Przesunięcie początku układu współrzędnych do środka okna
    m_viewTransform.translate(m_windowWidth / 2.0, m_windowHeight / 2.0);

    // 2. Zastosowanie współczynnika zoomowania (odwrócenie osi Y)
    m_viewTransform.scale(m_zoomFactor, -m_zoomFactor);

    // 3. Zastosowanie przesunięcia widoku
    m_viewTransform.translate(-m_panOffset.x(), -m_panOffset.y());

    // Obliczenie transformacji odwrotnej
    m_inverseViewTransform = m_viewTransform.inverted();
}

// Implementacja metod operacji na macierzach
std::shared_ptr<TransformMatrix> EigenMathAdapter::createRotationMatrix(double angle, const std::array<double, 3>& axis) {
    Eigen::Vector3d eigenAxis(axis[0], axis[1], axis[2]);
    Eigen::AngleAxisd rotation(angle, eigenAxis.normalized());
    Eigen::Matrix4d matrix = Eigen::Matrix4d::Identity();
    matrix.block<3, 3>(0, 0) = rotation.toRotationMatrix();
    return std::make_shared<StandardTransformMatrix>(matrix);
}

std::shared_ptr<TransformMatrix> EigenMathAdapter::createTranslationMatrix(const std::array<double, 3>& translation) {
    Eigen::Matrix4d matrix = Eigen::Matrix4d::Identity();
    matrix(0, 3) = translation[0];
    matrix(1, 3) = translation[1];
    matrix(2, 3) = translation[2];
    return std::make_shared<StandardTransformMatrix>(matrix);
}

std::shared_ptr<TransformMatrix> EigenMathAdapter::createScalingMatrix(const std::array<double, 3>& factors) {
    Eigen::Matrix4d matrix = Eigen::Matrix4d::Identity();
    matrix(0, 0) = factors[0];
    matrix(1, 1) = factors[1];
    matrix(2, 2) = factors[2];
    return std::make_shared<StandardTransformMatrix>(matrix);
}

std::vector<double> EigenMathAdapter::solveLinearSystem(
    const std::vector<std::vector<double>>& coefficients,
    const std::vector<double>& constants) {

    // Konwersja danych wejściowych
    Eigen::MatrixXd A = toEigenMatrix(coefficients);
    Eigen::VectorXd b = Eigen::Map<const Eigen::VectorXd>(constants.data(), constants.size());

    // Rozwiązanie układu równań (Ax = b)
    Eigen::VectorXd x = A.colPivHouseholderQr().solve(b);

    // Konwersja wyniku
    return std::vector<double>(x.data(), x.data() + x.size());
}

Eigen::MatrixXd EigenMathAdapter::toEigenMatrix(const std::vector<std::vector<double>>& matrix) const {
    if (matrix.empty()) return Eigen::MatrixXd(0, 0);

    size_t rows = matrix.size();
    size_t cols = matrix[0].size();

    Eigen::MatrixXd result(rows, cols);
    for (size_t i = 0; i < rows; ++i) {
        for (size_t j = 0; j < cols && j < matrix[i].size(); ++j) {
            result(i, j) = matrix[i][j];
        }
    }
    return result;
}

// File: IMathProvider.h
#pragma once

#include <memory>
#include <vector>
#include <array>

class TransformMatrix;

class IMathProvider {
public:
    virtual ~IMathProvider() = default;

    // Podstawowe operacje arytmetyczne (inline)
    virtual double add(double a, double b) const { return a + b; }
    virtual double subtract(double a, double b) const { return a - b; }
    virtual double multiply(double a, double b) const { return a * b; }
    virtual double divide(double a, double b) const {
        if (b == 0) throw std::invalid_argument("Division by zero");
        return a / b;
    }

    // Operacje macierzowe
    virtual std::shared_ptr<TransformMatrix> createRotationMatrix(double angle, const std::array<double, 3>& axis) = 0;
    virtual std::shared_ptr<TransformMatrix> createTranslationMatrix(const std::array<double, 3>& translation) = 0;
    virtual std::shared_ptr<TransformMatrix> createScalingMatrix(const std::array<double, 3>& factors) = 0;

    // Operacje algebry liniowej
    virtual std::vector<double> solveLinearSystem(
        const std::vector<std::vector<double>>& coefficients,
        const std::vector<double>& constants) = 0;
};

#include <QApplication>

int main(int argc, char *argv[]) {

    QAction *openAction = fileMenu->addAction("Otwórz...");

    QAction *saveAction = fileMenu->addAction("Zapisz");
    saveAction->setShortcut(QKeySequence::Save);
    QAction *saveAsAction = fileMenu->addAction("Zapisz jako...");
    saveAsAction->setShortcut(QKeySequence::SaveAs);
    fileMenu->addSeparator();
    QAction *exitAction = fileMenu->addAction("Wyjście");
    exitAction->setShortcut(QKeySequence::Quit);

    // Menu Edycja
    QMenu *editMenu = menuBar->addMenu("Edycja");
    QAction *undoAction = editMenu->addAction("Cofnij");
    undoAction->setShortcut(QKeySequence::Undo);
    QAction *redoAction = editMenu->addAction("Ponów");
    redoAction->setShortcut(QKeySequence::Redo);
    editMenu->addSeparator();
    QAction *deleteAction = editMenu->addAction("Usuń");
    deleteAction->setShortcut(QKeySequence::Delete);

    // Menu Widok
    QMenu *viewMenu = menuBar->addMenu("Widok");
    QAction *zoomInAction = viewMenu->addAction("Przybliż");
    zoomInAction->setShortcut(QKeySequence::ZoomIn);
    QAction *zoomOutAction = viewMenu->addAction("Oddal");
    zoomOutAction->setShortcut(QKeySequence::ZoomOut);
    QAction *resetViewAction = viewMenu->addAction("Resetuj widok");
    resetViewAction->setShortcut(QKeySequence("Ctrl+0"));
    QAction *fitContentAction = viewMenu->addAction("Dopasuj do zawartości");

    // Menu Krzywe
    QMenu *curvesMenu = menuBar->addMenu("Krzywe");
    QAction *bezierAction = curvesMenu->addAction("Krzywa Beziera");
    QAction *interpolatedAction = curvesMenu->addAction("Krzywa interpolowana");
    QAction *parametricAction = curvesMenu->addAction("Krzywa parametryczna");

    // Menu Pomoc
    QMenu *helpMenu = menuBar->addMenu("Pomoc");
    QAction *aboutAction = helpMenu->addAction("O programie");

    // Pasek narzędzi
    QToolBar *toolBar = new QToolBar("Narzędzia główne", &mainWindow);
    mainWindow.addToolBar(toolBar);

    toolBar->addAction(newAction);
    toolBar->addAction(openAction);
    toolBar->addAction(saveAction);
    toolBar->addSeparator();
    toolBar->addAction(undoAction);
    toolBar->addAction(redoAction);
    toolBar->addSeparator();
    toolBar->addAction(bezierAction);
    toolBar->addAction(interpolatedAction);
    toolBar->addAction(parametricAction);
    toolBar->addSeparator();
    toolBar->addAction(zoomInAction);
    toolBar->addAction(zoomOutAction);
    toolBar->addAction(resetViewAction);

    // Centralny widget
    QWidget *centralWidget = new QWidget(&mainWindow);
    QVBoxLayout *layout = new QVBoxLayout(centralWidget);

    // Canvas do rysowania
    Canvas *canvas = new Canvas(centralWidget);
    layout->addWidget(canvas);

    auto renderer = std::make_shared<QtRenderer>();
    canvas->setRenderer(renderer);

    // Ustawienie centralnego widgetu
    mainWindow.setCentralWidget(centralWidget);

    // Doker z listą obiektów
    QDockWidget *objectsDock = new QDockWidget("Obiekty", &mainWindow);
    objectsDock->setAllowedAreas(Qt::LeftDockWidgetArea | Qt::RightDockWidgetArea);

    QListWidget *objectsList = new QListWidget(objectsDock);
    objectsDock->setWidget(objectsList);

    mainWindow.addDockWidget(Qt::RightDockWidgetArea, objectsDock);

    // Doker z właściwościami
    QDockWidget *propertiesDock = new QDockWidget("Właściwości", &mainWindow);
    propertiesDock->setAllowedAreas(Qt::LeftDockWidgetArea | Qt::RightDockWidgetArea);

    QWidget *propertiesWidget = new QWidget(propertiesDock);
    QVBoxLayout *propertiesLayout = new QVBoxLayout(propertiesWidget);
    propertiesLayout->addWidget(new QLabel("Wybierz obiekt, aby edytować jego właściwości"));
    propertiesDock->setWidget(propertiesWidget);

    mainWindow.addDockWidget(Qt::RightDockWidgetArea, propertiesDock);

    // Pasek statusu
    QStatusBar *statusBar = mainWindow.statusBar();
    statusBar->showMessage("Gotowy");

    // Inicjalizacja narzędzi matematycznych
    auto mathProvider = std::make_shared<EigenMathAdapter>();

    // Inicjalizacja managera poleceń
    CommandManager commandManager;

    // Połączenia akcji
    QObject::connect(exitAction, &QAction::triggered, &app, &QApplication::quit);

    QObject::connect(newAction, &QAction::triggered, [&]() {
        canvas->clearObjects();
        commandManager.clearHistory();
        canvas->resetView();
        objectsList->clear();
        statusBar->showMessage("Utworzono nowy projekt");
    });

    QObject::connect(openAction, &QAction::triggered, [&]() {
        QString filePath = QFileDialog::getOpenFileName(&mainWindow, "Otwórz projekt",
            QString(), "Pliki projektu (*.crv);;Wszystkie pliki (*.*)");
        if (!filePath.isEmpty()) {
            statusBar->showMessage("Otwieranie projektu nie jest jeszcze zaimplementowane");
        }
    });

    QObject::connect(saveAction, &QAction::triggered, [&]() {
        statusBar->showMessage("Zapisywanie projektu nie jest jeszcze zaimplementowane");
    });

    QObject::connect(saveAsAction, &QAction::triggered, [&]() {
        QString filePath = QFileDialog::getSaveFileName(&mainWindow, "Zapisz projekt jako",
            QString(), "Pliki projektu (*.crv);;Wszystkie pliki (*.*)");
        if (!filePath.isEmpty()) {
            statusBar->showMessage("Zapisywanie projektu nie jest jeszcze zaimplementowane");
        }
    });

    QObject::connect(undoAction, &QAction::triggered, [&]() {
        if (commandManager.canUndo()) {
            commandManager.undo();
            canvas->refresh();
            statusBar->showMessage("Cofnięto ostatnią operację");
        }
    });

    QObject::connect(redoAction, &QAction::triggered, [&]() {
        if (commandManager.canRedo()) {
            commandManager.redo();
            canvas->refresh();
            statusBar->showMessage("Ponowiono operację");
        }
    });

    QObject::connect(deleteAction, &QAction::triggered, [&]() {
        auto selectedObject = canvas->selectedObject();
        if (selectedObject) {
            canvas->removeObject(selectedObject);
            canvas->refresh();
            statusBar->showMessage("Usunięto wybrany obiekt");
        }
    });

    QObject::connect(zoomInAction, &QAction::triggered, [&]() {
        canvas->zoomIn();
        statusBar->showMessage("Przybliżono widok");
    });

    QObject::connect(zoomOutAction, &QAction::triggered, [&]() {
        canvas->zoomOut();
        statusBar->showMessage("Oddalono widok");
    });

    QObject::connect(resetViewAction, &QAction::triggered, [&]() {
        canvas->resetView();
        statusBar->showMessage("Zresetowano widok");
    });

    QObject::connect(fitContentAction, &QAction::triggered, [&]() {
        canvas->fitToContents();
        statusBar->showMessage("Dopasowano widok do zawartości");
    });

    QObject::connect(interpolatedAction, &QAction::triggered, [&]() {
        auto curve = std::make_shared<InterpolatedCurve>(mathProvider);
        // Dodanie przykładowych punktów
        curve->addPoint(Point(0, 0));
        curve->addPoint(Point(50, 100));
        curve->addPoint(Point(100, 0));
        canvas->addObject(curve);
        canvas->refresh();
        statusBar->showMessage("Utworzono krzywą interpolowaną");
    });

    QObject::connect(aboutAction, &QAction::triggered, [&]() {
        QMessageBox::about(&mainWindow, "O programie",
            "<h3>Edytor Krzywych</h3>"
            "<p>Wersja 1.0</p>"
            "<p>Program do tworzenia i edycji krzywych matematycznych.</p>"
            "<p>Używa biblioteki Eigen do obliczeń matematycznych.</p>");
    });

    // Aktualizacja listy obiektów
    QObject::connect(canvas, &Canvas::objectAdded, [objectsList, &statusBar](std::shared_ptr<GeometricObject> obj) {
        const QString str = "Obiekt " + QString::number(objectsList->count() + 1);
        objectsList->addItem(str);
        statusBar->showMessage("Dodano nowy obiekt");
    });

    QObject::connect(canvas, &Canvas::objectRemoved, [objectsList, canvas, &statusBar](std::shared_ptr<GeometricObject> obj) {
        objectsList->clear();
        for (size_t i = 0; i < canvas->objects().size(); ++i) {
            const QString str = "Obiekt " + QString::number(i + 1);
            objectsList->addItem(QString(str));
        }
        statusBar->showMessage("Usunięto obiekt");
    });

    QObject::connect(objectsList, &QListWidget::currentRowChanged, [canvas, objectsList, &statusBar](int row) {
        if (row >= 0 && row < static_cast<int>(canvas->objects().size())) {
            canvas->selectObject(canvas->objects()[row]);
            canvas->refresh();
            const QString message = "Wybrano obiekt " + QString::number(row + 1);
            statusBar->showMessage(message);
        }
    });

    QObject::connect(canvas, &Canvas::objectSelected, [canvas, objectsList](std::shared_ptr<GeometricObject> obj) {
        auto objects = canvas->objects();
        auto it = std::find(objects.begin(), objects.end(), obj);
        if (it != objects.end()) {
            int row = static_cast<int>(std::distance(objects.begin(), it));
            objectsList->setCurrentRow(row);
        }
    });

    // Wyświetlenie okna
    mainWindow.show();
    statusBar->showMessage("Aplikacja gotowa do użycia");

    // Uruchomienie pętli zdarzeń aplikacji
    return app.exec();
}
