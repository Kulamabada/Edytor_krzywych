void CompoundCommand::addCommand(std::unique_ptr<AbstractCommand> command) {
    if (command) {
        m_commands.push_back(std::move(command));
    }
}

bool CompoundCommand::execute() {
    bool success = true;
    size_t executedCommands = 0;

    // Wykonaj wszystkie komendy po kolei
    for (size_t i = 0; i < m_commands.size(); ++i) {
        if (!m_commands[i]->execute()) {
            success = false;
            executedCommands = i;
            break;
        }
        executedCommands = i + 1;
    }

    // Jeśli którakolwiek komenda nie powiodła się, cofnij wszystkie wykonane w odwrotnej kolejności
    if (!success) {
        for (size_t i = 0; i < executedCommands; ++i) {
            size_t index = executedCommands - i - 1;
            m_commands[index]->undo();
        }
    }

    return success;
}

bool CompoundCommand::undo() {
    bool success = true;

    // Cofnij wszystkie komendy w odwrotnej kolejności
    for (auto it = m_commands.rbegin(); it != m_commands.rend(); ++it) {
        if (!(*it)->undo()) {
            success = false;
            break;
        }
    }

    return success;
}

bool CompoundCommand::redo() {
    bool success = true;

    // Ponownie wykonaj wszystkie komendy po kolei
    for (auto& command : m_commands) {
        if (!command->redo()) {
            success = false;
            break;
        }
    }

    return success;
}

// Implementacja AddControlPointCommand
AddControlPointCommand::AddControlPointCommand(std::shared_ptr<InterpolatedCurve> curve, const Point& point, int index)
    : m_curve(curve), m_point(point), m_index(index), m_addedIndex(-1) {
}

bool AddControlPointCommand::execute() {
    if (!m_curve) {
        return false;
    }

    if (m_index < 0 || m_index > static_cast<int>(m_curve->getPointCount())) {
        // Dodaj na końcu krzywej
        m_curve->addPoint(m_point);
        m_addedIndex = static_cast<int>(m_curve->getPointCount()) - 1;
    } else {
        // Dodaj pod określonym indeksem
        m_curve->insertPoint(m_index, m_point);
        m_addedIndex = m_index;
    }

    return true;
}

bool AddControlPointCommand::undo() {
    if (!m_curve || m_addedIndex < 0 ||
        m_addedIndex >= static_cast<int>(m_curve->getPointCount())) {
        return false;
    }

    m_curve->removePoint(m_addedIndex);
    return true;
}

bool AddControlPointCommand::redo() {
    if (!m_curve) {
        return false;
    }

    if (m_addedIndex < 0) {
        return execute(); // Pierwszy raz wykonujemy execute()
    }

    // Ponowne dodanie punktu na odpowiednim indeksie
    if (m_addedIndex > static_cast<int>(m_curve->getPointCount())) {
        m_curve->addPoint(m_point);
    } else {
        m_curve->insertPoint(m_addedIndex, m_point);
    }

    return true;
}

DeleteControlPointCommand::DeleteControlPointCommand(std::shared_ptr<InterpolatedCurve> curve,

    : m_curve(curve), m_pointIndex(pointIndex) {

    // Zapisz punkt przed usunięciem dla operacji undo
    if (m_curve && m_pointIndex >= 0 && m_pointIndex < static_cast<int>(m_curve->getPointCount())) {
        const auto& points = m_curve->getPoints();

    if (!m_curve || m_pointIndex < 0 ||
        m_pointIndex >= static_cast<int>(m_curve->getPointCount())) {
        return false;
    }

    // Sprawdź, czy krzywa będzie miała wystarczającą liczbę punktów po usunięciu

    return true;

    }

    // Przywróć punkt na odpowiedni indeks
    if (m_pointIndex >= static_cast<int>(m_curve->getPointCount())) {
        m_curve->addPoint(m_deletedPoint);
    } else {
        m_curve->insertPoint(m_pointIndex, m_deletedPoint);
    }

    return true;
}

// Implementacja PIMPL
class CommandManager::CommandManagerImpl {
public:
    CommandManagerImpl() = default;
    ~CommandManagerImpl() = default;

    bool executeCommand(std::unique_ptr<AbstractCommand> command) {
        if (!command) {
            return false;
        }

        // Wykonanie komendy
        bool success = command->execute();
        if (!success) {
            return false;
        }

        // Po wykonaniu nowej komendy, czyścimy stos redo
        if (!m_redoStack.empty()) {
            m_redoStack.clear();
        }

        // Dodanie komendy do stosu undo
        m_undoStack.push_back(std::move(command));
        return true;
    }

    bool canUndo() const {
        return !m_undoStack.empty();
    }

    bool canRedo() const {
        return !m_redoStack.empty();
    }

    bool undo() {
        if (!canUndo()) {
            return false;
        }

        // Pobranie ostatniej komendy ze stosu undo
        auto command = std::move(m_undoStack.back());
        m_undoStack.pop_back();

        // Cofnięcie komendy
        bool success = command->undo();

        // Przeniesienie komendy na stos redo
        m_redoStack.push_back(std::move(command));
        return success;
    }

    bool redo() {
        if (!canRedo()) {
            return false;
        }

        // Pobranie ostatniej komendy ze stosu redo
        auto command = std::move(m_redoStack.back());
        m_redoStack.pop_back();

        // Ponowne wykonanie komendy
        bool success = command->redo();

        // Przeniesienie komendy na stos undo
        m_undoStack.push_back(std::move(command));
        return success;
    }

    void clearHistory() {
        m_undoStack.clear();
        m_redoStack.clear();
    }

    int undoStackSize() const {
        return static_cast<int>(m_undoStack.size());
    }

    int redoStackSize() const {
        return static_cast<int>(m_redoStack.size());
    }

private:
    std::vector<std::unique_ptr<AbstractCommand>> m_undoStack;
    std::vector<std::unique_ptr<AbstractCommand>> m_redoStack;
};

InterpolatedCurve::InterpolatedCurve(std::shared_ptr<IMathProvider> mathProvider,
                                    InterpolationType type)
    : type(type),
      mathProvider(std::move(mathProvider)){}

void InterpolatedCurve::addPoint(const Point& point) {
    points.push_back(point);
    needsRebuild = true;
}

void InterpolatedCurve::insertPoint(size_t index, const Point& point) {
    if (index <= points.size()) {
        points.insert(points.begin() + index, point);
        needsRebuild = true;
    }
}

void InterpolatedCurve::removePoint(size_t index) {
    if (index < points.size()) {
        points.erase(points.begin() + index);
        needsRebuild = true;
    }
}

void InterpolatedCurve::rebuildInterpolation() const {
    if (!needsRebuild || points.size() < 2) {
        needsRebuild = false;
        return;
    }
    coefficients.clear();

    switch (type) {
        case InterpolationType::NIFS3:
            buildNIFS3();
        break;
        case InterpolationType::OIFS3:
            buildOIFS3();
        break;
        case InterpolationType::Lagrange:
            buildLagrange();
        break;
        case InterpolationType::Barycentric:
            buildBarycentric();
        break;
    }

    needsRebuild = false;
}
void InterpolatedCurve::buildNIFS3() const {
    const size_t n = points.size();
    if (n < 2) return;

    Eigen::MatrixXd A = Eigen::MatrixXd::Zero(n, n);
    Eigen::VectorXd bx = Eigen::VectorXd::Zero(n);
    Eigen::VectorXd by = Eigen::VectorXd::Zero(n);

    A(0, 0) = 1.0;
    A(n-1, n-1) = 1.0;

    for (size_t i = 1; i < n - 1; ++i) {
        A(i, i-1) = 1.0;
        A(i, i) = 4.0;
        A(i, i+1) = 1.0;

        bx(i) = 3.0 * (points[i+1].x() - points[i-1].x());
        by(i) = 3.0 * (points[i+1].y() - points[i-1].y());
    }

    Eigen::VectorXd cx = A.colPivHouseholderQr().solve(bx);
    Eigen::VectorXd cy = A.colPivHouseholderQr().solve(by);

    for (size_t i = 0; i < n - 1; ++i) {
        Eigen::Matrix<double, 4, 2> coef;

        coef(0, 0) = points[i].x();
        coef(1, 0) = cx(i);
        coef(2, 0) = 3.0 * (points[i+1].x() - points[i].x()) - 2.0 * cx(i) - cx(i+1);
        coef(3, 0) = 2.0 * (points[i].x() - points[i+1].x()) + cx(i) + cx(i+1);

        coef(0, 1) = points[i].y();
        coef(1, 1) = cy(i);
        coef(2, 1) = 3.0 * (points[i+1].y() - points[i].y()) - 2.0 * cy(i) - cy(i+1);
        coef(3, 1) = 2.0 * (points[i].y() - points[i+1].y()) + cy(i) + cy(i+1);

        coefficients.push_back(coef);
    }
}

void InterpolatedCurve::buildOIFS3() const {
    const size_t n = points.size();
    if (n < 3) return;

    Eigen::MatrixXd A = Eigen::MatrixXd::Zero(n, n);
    Eigen::VectorXd bx = Eigen::VectorXd::Zero(n);
    Eigen::VectorXd by = Eigen::VectorXd::Zero(n);

    for (size_t i = 0; i < n; ++i) {
        size_t prev = (i > 0) ? (i - 1) : (n - 1);
        size_t next = (i < n - 1) ? (i + 1) : 0;

        A(i, prev) = 1.0;
        A(i, i) = 4.0;
        A(i, next) = 1.0;

        bx(i) = 3.0 * (points[next].x() - points[prev].x());
        by(i) = 3.0 * (points[next].y() - points[prev].y());
    }

    Eigen::VectorXd cx = A.colPivHouseholderQr().solve(bx);
    Eigen::VectorXd cy = A.colPivHouseholderQr().solve(by);

    for (size_t i = 0; i < n; ++i) {
        size_t next = (i < n - 1) ? (i + 1) : 0;

        Eigen::Matrix<double, 4, 2> coef;

        coef(0, 0) = points[i].x();
        coef(1, 0) = cx(i);
        coef(2, 0) = 3.0 * (points[next].x() - points[i].x()) - 2.0 * cx(i) - cx(next);
        coef(3, 0) = 2.0 * (points[i].x() - points[next].x()) + cx(i) + cx(next);

        coef(0, 1) = points[i].y();
        coef(1, 1) = cy(i);
        coef(2, 1) = 3.0 * (points[next].y() - points[i].y()) - 2.0 * cy(i) - cy(next);
        coef(3, 1) = 2.0 * (points[i].y() - points[next].y()) + cy(i) + cy(next);

        coefficients.push_back(coef);
    }
}

void InterpolatedCurve::buildLagrange() const {
    const size_t n = points.size();
    if (n < 2) return;
}

void InterpolatedCurve::buildBarycentric() const {
    const size_t n = points.size();
    if (n < 2) return;
}

Vector InterpolatedCurve::getTangentAt(double t) const {
    const double epsilon = 1e-6;
    Point p1 = getPointAt(t);
    Point p2 = getPointAt(t + epsilon);

    return Vector(
        (p2.x() - p1.x()) / epsilon,
        (p2.y() - p1.y()) / epsilon,
        (p2.z() - p1.z()) / epsilon
    );
}

Point InterpolatedCurve::getPointAt(double t) const {
    if (points.empty()) return Point(0.0, 0.0, 0.0);
    if (points.size() == 1) return points[0];

    if (needsRebuild) {
        rebuildInterpolation();
    }

    t = std::max(0.0, std::min(1.0, t));

    size_t numSegments = (type == InterpolationType::OIFS3) ? points.size() : (points.size() - 1);
    double segmentLength = 1.0 / numSegments;
    size_t segmentIndex = std::min(static_cast<size_t>(t / segmentLength), numSegments - 1);
    double localT = (t - segmentIndex * segmentLength) / segmentLength;

    if (type == InterpolationType::NIFS3 || type == InterpolationType::OIFS3) {
        if (segmentIndex >= coefficients.size()) {
            return points[points.size() - 1];
        }

        const auto& coef = coefficients[segmentIndex];
        Eigen::Vector4d tVec;
        tVec << 1.0, localT, localT*localT, localT*localT*localT;

        double x = tVec.dot(coef.col(0));
        double y = tVec.dot(coef.col(1));

        return Point(x, y, 0.0);
    }
    else if (type == InterpolationType::Lagrange) {
        double x = 0.0;
        double y = 0.0;

        for (size_t i = 0; i < points.size(); ++i) {
            double L = 1.0;
            for (size_t j = 0; j < points.size(); ++j) {
                if (i != j) {
                    double tj = static_cast<double>(j) / (points.size() - 1);
                    double ti = static_cast<double>(i) / (points.size() - 1);
                    L *= (t - tj) / (ti - tj);
                }
            }
            x += points[i].x() * L;
            y += points[i].y() * L;
        }

        return Point(x, y, 0.0);
    }
    else if (type == InterpolationType::Barycentric) {
        double x = 0.0;
        double y = 0.0;
        double sumW = 0.0;

        std::vector<double> weights(points.size());
        for (size_t i = 0; i < points.size(); ++i) {
            double ti = static_cast<double>(i) / (points.size() - 1);
            weights[i] = 1.0;
            for (size_t j = 0; j < points.size(); ++j) {
                if (i != j) {
                    double tj = static_cast<double>(j) / (points.size() - 1);
                    weights[i] *= (ti - tj);
                }
            }
            weights[i] = 1.0 / weights[i];
        }

        for (size_t i = 0; i < points.size(); ++i) {
            double ti = static_cast<double>(i) / (points.size() - 1);

            if (std::abs(t - ti) < 1e-10) {
                return points[i];
            }

            double w = weights[i] / (t - ti);
            x += points[i].x() * w;
            y += points[i].y() * w;
            sumW += w;
        }
        return Point(x / sumW, y / sumW, 0.0);
    }

    size_t nextIndex = (segmentIndex + 1) % points.size();
    double oneMinusT = 1.0 - localT;
    double x = oneMinusT * points[segmentIndex].x() + localT * points[nextIndex].x();
    double y = oneMinusT * points[segmentIndex].y() + localT * points[nextIndex].y();
    double z = oneMinusT * points[segmentIndex].z() + localT * points[nextIndex].z();

    return Point(x, y, z);
}



double InterpolatedCurve::getLength() const {
    const int steps = 20;
    double length = 0.0;
    Point prev = getPointAt(0.0);

    for (int i = 1; i <= steps; ++i) {
        double t = static_cast<double>(i) / steps;
        Point curr = getPointAt(t);
        length += prev.distanceTo(curr);
        prev = curr;
    }

    return length;
}

void InterpolatedCurve::transform(const TransformMatrix& matrix) {
    for (auto& point : points) {
        point.transform(matrix);
    }
    needsRebuild = true;
}

std::unique_ptr<Curve> InterpolatedCurve::clone() const {
    auto copy = std::make_unique<InterpolatedCurve>(mathProvider, type);
    copy->setPoints(points);
    return copy;
}

Vector InterpolatedCurve::getPosition() const {
    if (points.empty()) return Vector::Zero();

    Vector sum = Vector::Zero();
    for (const auto& p : points) {
        sum += Vector(p.x(), p.y(), p.z());
    }
    return sum / static_cast<double>(points.size());
}

void InterpolatedCurve::setPosition(const Eigen::Vector3d& position) {
    if (points.empty()) return;

    Vector delta = position - getPosition();
    for (auto& p : points) {
        p.setX(p.x() + delta.x());
        p.setY(p.y() + delta.y());
        p.setZ(p.z() + delta.z());
    }
    needsRebuild = true;
}

double Point::distanceTo(const Point& other) const {
    const double dx = other.x() - x();
    const double dy = other.y() - y();
    const double dz = other.z() - z();
    return std::sqrt(dx * dx + dy * dy + dz * dz);
}

Point Point::midPoint(const Point& other) const {
    return Point(
        (x() + other.x()) / 2.0,
        (y() + other.y()) / 2.0,
        (z() + other.z()) / 2.0
    );
}

void Point::transform(const TransformMatrix& matrix) {
    // Konwertujemy punkt do formatu Eigen::Vector4d (reprezentacja homogeniczna)
    Eigen::Vector4d homogeneousPoint;
    homogeneousPoint << x(), y(), z(), 1.0;

    // Pobieramy macierz transformacji i aplikujemy transformację
    Eigen::Vector4d transformedPoint = matrix.getMatrix() * homogeneousPoint;

    // Konwertujemy z powrotem do współrzędnych kartezjańskich (z normalizacją)
    if (std::abs(transformedPoint(3)) > 1e-10) {  // Unikamy dzielenia przez zero
        setX(transformedPoint(0) / transformedPoint(3));
        setY(transformedPoint(1) / transformedPoint(3));
        setZ(transformedPoint(2) / transformedPoint(3));
    }
}

    setZ(position.z());

int Canvas::hitTestControlPoint(const QPointF& pos, double tolerance) {
    auto curve = getSelectedCurve();
    if (!curve || !m_dPtr->viewportManager) return -1;

    const auto& points = curve->getPoints();
    for (size_t i = 0; i < points.size(); ++i) {
        QPointF screenPos = m_dPtr->viewportManager->worldToScreen(
            QPointF(points[i].x(), points[i].y()));

        if (QLineF(pos, screenPos).length() <= tolerance) {
            return static_cast<int>(i);
        }
    }

    return -1;
}

bool Canvas::addControlPoint(const QPointF& screenPos) {
    auto curve = getSelectedCurve();
    if (!curve || !m_dPtr->viewportManager) return false;

    // Konwertuj współrzędne ekranu na współrzędne świata
    QPointF worldPos = m_dPtr->viewportManager->screenToWorld(screenPos);

    // Dodaj punkt do krzywej
    curve->addPoint(Point(worldPos.x(), worldPos.y()));

    // Ustal indeks nowego punktu (ostatni punkt)
    int newPointIndex = static_cast<int>(curve->getPointCount()) - 1;

    // Zaznacz nowy punkt
    selectControlPoint(newPointIndex);

    // Emituj sygnał
    emit controlPointAdded(newPointIndex);

    update();
    return true;
}

bool Canvas::moveControlPoint(int pointIndex, const QPointF& screenPos) {
    auto curve = getSelectedCurve();
    if (!curve || !m_dPtr->viewportManager ||
        pointIndex < 0 || pointIndex >= static_cast<int>(curve->getPointCount())) {
        return false;
    }

    // Konwertuj współrzędne ekranu na współrzędne świata
    QPointF worldPos = m_dPtr->viewportManager->screenToWorld(screenPos);

    // Przesuń punkt
    const auto& points = curve->getPoints();
    auto newPoint = Point(worldPos.x(), worldPos.y(), points[pointIndex].z());
    curve->movePoint(pointIndex, newPoint);

    // Emituj sygnał
    emit controlPointMoved(pointIndex);

    update();
    return true;
}

void Canvas::selectControlPoint(int pointIndex) {
    auto curve = getSelectedCurve();
    if (!curve) return;

    if (pointIndex >= 0 && pointIndex < static_cast<int>(curve->getPointCount())) {
        m_dPtr->selectedControlPointIndex = pointIndex;
        emit controlPointSelected(pointIndex);
    } else {
        m_dPtr->selectedControlPointIndex = -1;
    }

    update();
}

void Canvas::startAddingControlPoints() {
    setEditMode(EditMode::AddPoint);
}

void Canvas::startMovingControlPoints() {
    setEditMode(EditMode::MovePoint);
}

void Canvas::paintEvent(QPaintEvent* event)
{
    Q_UNUSED(event);

    if (!isVisible() || width() <= 0 || height() <= 0) return;

    QPainter painter(this);
    if (!painter.isActive()) {
        qWarning() << "Nie można zainicjować QPainter w Canvas::paintEvent";
        return;
    }

    painter.fillRect(rect(), Qt::white);

    if (!m_dPtr->renderer) return;

    m_dPtr->renderer->beginRender(&painter, m_dPtr->viewportManager.get());

    // Rysowanie elementów pomocniczych
    m_dPtr->renderer->renderGrid(1.0);
    m_dPtr->renderer->renderAxes();

    // Rysowanie obiektów geometrycznych
    for (const auto& object : m_dPtr->objects) {
        m_dPtr->renderer->renderObject(*object);
    }

    // Rysowanie markera zaznaczenia dla wybranego obiektu
    if (m_dPtr->selectedObject) {
        const auto* geometricObj = dynamic_cast<const GeometricObject*>(m_dPtr->selectedObject.get());
        if (geometricObj) {
            auto position = geometricObj->getPosition();
            QPointF selectionPos = m_dPtr->viewportManager->worldToScreen(QPointF(position.x(), position.y()));
            m_dPtr->renderer->renderSelectionMarker(selectionPos);
        }
    }

    // Rysowanie punktów kontrolnych jeśli jesteśmy w trybie edycji
    if (m_dPtr->isEditingPoints && m_dPtr->selectedObject) {
        drawControlPoints();
    }

    m_dPtr->renderer->endRender();
}

void Canvas::drawControlPoints() {
    auto curve = getSelectedCurve();
    if (!curve || !m_dPtr->renderer) return;

    const auto& points = curve->getPoints();

    // Najpierw narysuj poligon kontrolny
    m_dPtr->renderer->renderControlPolygon(points);

    // Potem narysuj punkty kontrolne
    m_dPtr->renderer->renderControlPoints(points, m_dPtr->selectedControlPointIndex, m_dPtr->activeControlPointIndex);
}

void Canvas::resizeEvent(QResizeEvent* event)
{
    if (m_dPtr->viewportManager) {
        m_dPtr->viewportManager->setWindowSize(event->size().width(), event->size().height());
        emit viewportChanged();
    }

    QWidget::resizeEvent(event);
}

void Canvas::mousePressEvent(QMouseEvent* event)
{
    m_dPtr->lastMousePos = event->pos();

    if (event->button() == Qt::LeftButton) {
        if (m_dPtr->editMode == EditMode::AddPoint) {
            // Dodaj nowy punkt kontrolny
            addControlPoint(event->pos());
            return;
        }
        else if (m_dPtr->editMode == EditMode::MovePoint) {
            // Sprawdź czy kliknięto na punkt kontrolny
            int pointIndex = hitTestControlPoint(event->pos());
            if (pointIndex >= 0) {
                selectControlPoint(pointIndex);
                m_dPtr->isDraggingPoint = true;
                m_dPtr->activeControlPointIndex = pointIndex;
                setCursor(Qt::ClosedHandCursor);
                return;
            }
        }
        else {
            // Tryb zaznaczania obiektu
            m_dPtr->isSelecting = true;
            m_dPtr->selectionPoint = event->pos();

            // Sprawdź czy kliknięto na obiekt
            // Ta logika mogłaby być rozszerzona w przyszłości
        }
    }
    else if (event->button() == Qt::MiddleButton) {
        m_dPtr->isPanning = true;
        setCursor(Qt::ClosedHandCursor);
    }

    QWidget::mousePressEvent(event);
}

void Canvas::mouseMoveEvent(QMouseEvent* event)
{
    QPointF currentPos = event->pos();
    QPointF delta = currentPos - m_dPtr->lastMousePos;

    if (m_dPtr->isDraggingPoint && m_dPtr->activeControlPointIndex >= 0) {
        // Przesuń punkt kontrolny
        moveControlPoint(m_dPtr->activeControlPointIndex, currentPos);
    }
    else if (m_dPtr->isPanning) {
        if (m_dPtr->viewportManager) {
            m_dPtr->viewportManager->pan(delta);
            emit viewportChanged();
            update();
        }
    }
    else if (m_dPtr->editMode == EditMode::MovePoint) {
        // Sprawdź czy kursor jest nad punktem kontrolnym
        int pointIndex = hitTestControlPoint(currentPos);
        if (pointIndex >= 0) {
            setCursor(Qt::OpenHandCursor);
        } else {
            setCursor(Qt::ArrowCursor);
        }
    }

    m_dPtr->lastMousePos = currentPos;
    QWidget::mouseMoveEvent(event);
}

void Canvas::mouseReleaseEvent(QMouseEvent* event)
{
    if (event->button() == Qt::LeftButton) {
        if (m_dPtr->isDraggingPoint) {
            m_dPtr->isDraggingPoint = false;
            m_dPtr->activeControlPointIndex = -1;

            if (m_dPtr->editMode == EditMode::MovePoint) {
                setCursor(Qt::OpenHandCursor);
            } else {
                setCursor(Qt::ArrowCursor);
            }
        }
        else if (m_dPtr->isSelecting) {
            // Tutaj logika zaznaczania obiektu
            m_dPtr->isSelecting = false;
        }
    }
    else if (event->button() == Qt::MiddleButton) {
        m_dPtr->isPanning = false;
        setCursor(Qt::ArrowCursor);
    }

    QWidget::mouseReleaseEvent(event);
}

void Canvas::wheelEvent(QWheelEvent* event)
{
    if (m_dPtr->viewportManager) {
        QPointF center = event->position();

        double factor = event->angleDelta().y() > 0 ? 1.1 : 0.9;

        m_dPtr->viewportManager->zoom(factor, center);
        emit viewportChanged();
        update();
    }

    event->accept();
}

void Canvas::keyPressEvent(QKeyEvent* event)
{
    if (event->key() == Qt::Key_Escape) {
        // Escape wychodzi z trybu edycji
        finishEditing();
    }
    else if (event->key() == Qt::Key_Delete && m_dPtr->selectedControlPointIndex >= 0) {
        // Usuń wybrany punkt kontrolny
        auto curve = getSelectedCurve();
        if (curve && m_dPtr->selectedControlPointIndex >= 0 &&
            m_dPtr->selectedControlPointIndex < static_cast<int>(curve->getPointCount())) {

            // Upewnij się, że krzywa będzie miała co najmniej 2 punkty
            if (curve->getPointCount() > 2) {
                curve->removePoint(m_dPtr->selectedControlPointIndex);
                m_dPtr->selectedControlPointIndex = -1;
                update();
            } else {
                QMessageBox::warning(this, "Usuwanie punktu",
                    "Krzywa musi mieć co najmniej 2 punkty.");
            }
        }
    }

    QWidget::keyPressEvent(event);
}

void QtRenderer::beginRender(QPainter* painter, ViewportManager* viewportManager)
{
    if (!painter || !viewportManager) {
        qDebug() << "Nieprawidłowy painter lub viewportManager";
        return;
    }

    if (!painter->isActive()) {
        qWarning() << "QPainter nie jest aktywny w beginRender()";
        return;
    }

    m_dPtr->painter = painter;
    m_dPtr->viewportManager = viewportManager;

    m_dPtr->painter->save();
    m_dPtr->painter->setTransform(viewportManager->viewTransform());
    m_dPtr->painter->setRenderHint(QPainter::Antialiasing, true);
}

void QtRenderer::endRender()
{
    if (m_dPtr->painter) {
        m_dPtr->painter->restore();

        m_dPtr->painter = nullptr;
        m_dPtr->viewportManager = nullptr;
    }
}

void QtRenderer::renderPoint(const Point& point)
{
    if (!m_dPtr->painter || !m_dPtr->viewportManager || !m_dPtr->painter->isActive()) {
        return;
    }

    QPen pen(Qt::NoPen);
    QBrush brush(m_dPtr->pointColor);
    m_dPtr->painter->setPen(pen);
    m_dPtr->painter->setBrush(brush);

    const double x = point.x(), y = point.y();

    m_dPtr->painter->drawEllipse(QPointF(x, y), m_dPtr->pointSize, m_dPtr->pointSize);
}

void QtRenderer::renderCurve(const Curve& curve)
{
    if (!m_dPtr->painter || !m_dPtr->viewportManager) {
        return;
    }

    QPen pen(m_dPtr->curveColor, m_dPtr->curveWidth, m_dPtr->curveStyle);
    m_dPtr->painter->setPen(pen);
    m_dPtr->painter->setBrush(Qt::NoBrush);

    constexpr int numSegments = 5000;
    QPolygonF points;

    for (int i = 0; i <= numSegments; ++i) {
        double t = static_cast<double>(i) / numSegments;
        Point p = curve.getPointAt(t);
        points.append(QPointF(p.x(), p.y()));
    }

    m_dPtr->painter->drawPolyline(points);
}

void QtRenderer::renderObject(const GeometricObject& object)
{
    const Point* point = dynamic_cast<const Point*>(&object);
    if (point) {
        renderPoint(*point);
        return;
    }

    const Curve* curve = dynamic_cast<const Curve*>(&object);
    if (curve)
        renderCurve(*curve);
}

void QtRenderer::renderGrid(double gridSize) {
    if (!m_dPtr->drawGrid || !m_dPtr->painter) return;

    QRectF visibleRect = m_dPtr->viewportManager->visibleWorldRect();

    // Linie pomocnicze
    QPen smallPen(QColor(230, 230, 230), 0.5, Qt::DotLine);
    m_dPtr->painter->setPen(smallPen);

    for (double x = std::floor(visibleRect.left()); x <= visibleRect.right(); x += 1.0) {
        m_dPtr->painter->drawLine(QPointF(x, visibleRect.top()), QPointF(x, visibleRect.bottom()));
    }
    for (double y = std::floor(visibleRect.top()); y <= visibleRect.bottom(); y += 1.0) {
        m_dPtr->painter->drawLine(QPointF(visibleRect.left(), y), QPointF(visibleRect.right(), y));
    }

    // Linie główne
    QPen mainPen(QColor(200, 200, 200), 1.0, Qt::SolidLine);
    m_dPtr->painter->setPen(mainPen);

    for (double x = std::floor(visibleRect.left() / 5) * 5; x <= visibleRect.right(); x += 5.0) {
        m_dPtr->painter->drawLine(QPointF(x, visibleRect.top()), QPointF(x, visibleRect.bottom()));
    }
    for (double y = std::floor(visibleRect.top() / 5) * 5; y <= visibleRect.bottom(); y += 5.0) {
        m_dPtr->painter->drawLine(QPointF(visibleRect.left(), y), QPointF(visibleRect.right(), y));
    }
}

void QtRenderer::renderText(const QPointF& position, const QString& text) {
    if (!m_dPtr->painter) return;
    m_dPtr->painter->save();
    QTransform current = m_dPtr->painter->transform();
    m_dPtr->painter->resetTransform();
    QPointF screenPos = current.map(position);
    m_dPtr->painter->drawText(screenPos, text);
    m_dPtr->painter->restore();
}

void QtRenderer::renderAxes()
{
    if (!m_dPtr->drawAxes || !m_dPtr->painter || !m_dPtr->viewportManager) return;

    QPen pen(m_dPtr->axesColor, m_dPtr->axesWidth, Qt::SolidLine);
    m_dPtr->painter->setPen(pen);

    QRectF visibleRect = m_dPtr->viewportManager->visibleWorldRect();

    // Rysowanie głównych osi
    m_dPtr->painter->drawLine(QPointF(visibleRect.left(), 0), QPointF(visibleRect.right(), 0));
    m_dPtr->painter->drawLine(QPointF(0, visibleRect.top()), QPointF(0, visibleRect.bottom()));

    double arrowSize = 7.0 / m_dPtr->viewportManager->zoomFactor();

    QPolygonF arrowX;
    arrowX << QPointF(visibleRect.right(), 0)
          << QPointF(visibleRect.right() - arrowSize, arrowSize / 2)
          << QPointF(visibleRect.right() - arrowSize, -arrowSize / 2);
    m_dPtr->painter->setBrush(m_dPtr->axesColor);
    m_dPtr->painter->drawPolygon(arrowX);

    QPolygonF arrowY;
    arrowY << QPointF(0, visibleRect.bottom())
          << QPointF(arrowSize / 2, visibleRect.bottom() - arrowSize)
          << QPointF(-arrowSize / 2, visibleRect.bottom() - arrowSize);
    m_dPtr->painter->drawPolygon(arrowY);

    renderText(QPointF(visibleRect.right() - 2 * arrowSize, -arrowSize), "x");
    renderText(QPointF(arrowSize, visibleRect.bottom() - 2 * arrowSize), "y");

    renderAxisTicks();
}

void QtRenderer::renderAxisTicks() {
    if (!m_dPtr->painter || !m_dPtr->viewportManager || !m_dPtr->painter->isActive()) return;

    QRectF visibleRect = m_dPtr->viewportManager->visibleWorldRect();
    double zoomFactor = m_dPtr->viewportManager->zoomFactor();

    // Obliczenie optymalnej odległości między przedziałkami w zależności od szerokości widocznego obszaru
    double viewportWidth = visibleRect.width();
    double viewportHeight = visibleRect.height();

    // Chcemy mieć około 5-10 przedziałek na widocznym obszarze
    double desiredTickCount = 16.0;
    double optimalSpacingX = viewportWidth / desiredTickCount;
    double optimalSpacingY = viewportHeight / desiredTickCount;

    // Funkcja pomocnicza do znajdowania "ładnej" wartości odstępu
    auto findNiceSpacing = [](double optimalSpacing) -> double {
        double magnitude = pow(10, floor(log10(optimalSpacing)));
        double normalized = optimalSpacing / magnitude;

        if (normalized < 1.2) return 1.0 * magnitude;
        if (normalized < 2.5) return 2.0 * magnitude;
        if (normalized < 7.0) return 5.0 * magnitude;
        return 10.0 * magnitude;
    };

    double tickSpacingX = findNiceSpacing(optimalSpacingX);
    double tickSpacingY = findNiceSpacing(optimalSpacingY);

    // Minimalna i maksymalna wartość dla przedziałek
    double xMin = floor(visibleRect.left() / tickSpacingX) * tickSpacingX;
    double xMax = ceil(visibleRect.right() / tickSpacingX) * tickSpacingX;
    double yMin = floor(visibleRect.top() / tickSpacingY) * tickSpacingY;
    double yMax = ceil(visibleRect.bottom() / tickSpacingY) * tickSpacingY;

    double tickLength = 5.0 / zoomFactor;

    auto determinePrecision = [](double spacing) -> int {
        if (spacing >= 1.0) return 0;
        return std::min(6, static_cast<int>(std::ceil(std::abs(std::log10(spacing)))));
    };

    int xPrecision = determinePrecision(tickSpacingX);
    int yPrecision = determinePrecision(tickSpacingY);

    for (double x = xMin; x <= xMax; x += tickSpacingX) {
        if (std::abs(x) < 1e-6) continue;

        m_dPtr->painter->drawLine(QPointF(x, -tickLength/2), QPointF(x, tickLength/2));
        QString label = QString::number(x, 'f', xPrecision);
        // Usunięcie zbędnych zer na końcu
        while (label.contains('.') && label.endsWith('0'))
            label.chop(1);

        if (label.endsWith('.'))
            label.chop(1);

        renderText(QPointF(x, -tickLength - 5.0/zoomFactor), label);
    }

    // Rysowanie przedziałek dla osi Y
    for (double y = yMin; y <= yMax; y += tickSpacingY) {
        if (std::abs(y) < 1e-6) continue;

        m_dPtr->painter->drawLine(QPointF(-tickLength/2, y), QPointF(tickLength/2, y));
        QString label = QString::number(y, 'f', yPrecision);
        // Usunięcie zbędnych zer na końcu
        while (label.contains('.') && label.endsWith('0'))
            label.chop(1);

        if (label.endsWith('.'))
            label.chop(1);

        renderText(QPointF(-tickLength - 15.0/zoomFactor, y), label);
    }
}

void QtRenderer::renderSelectionMarker(const QPointF& position)
{
    if (!m_dPtr->painter || !m_dPtr->painter->isActive())
        return;

    m_dPtr->painter->save();
    m_dPtr->painter->resetTransform();

    QPen pen(m_dPtr->selectionColor, 1.0, Qt::SolidLine);
    m_dPtr->painter->setPen(pen);
    m_dPtr->painter->setBrush(Qt::NoBrush);

    double size = m_dPtr->selectionSize;
    m_dPtr->painter->drawEllipse(position, size, size);

    // Narysuj krzyżyk w środku
    m_dPtr->painter->drawLine(position - QPointF(size / 2, 0),
                             position + QPointF(size / 2, 0));
    m_dPtr->painter->drawLine(position - QPointF(0, size / 2),
                             position + QPointF(0, size / 2));

    m_dPtr->painter->restore();
}

void QtRenderer::renderControlPoint(const QPointF& position, bool isSelected, bool isActive)
{
    if (!m_dPtr->painter || !m_dPtr->painter->isActive()) {
        return;
    }

    m_dPtr->painter->save();
    m_dPtr->painter->resetTransform();

    // Wybór koloru w zależności od stanu punktu
    QColor fillColor;
    if (isActive) {
        fillColor = m_dPtr->controlPointActiveColor;
    } else if (isSelected) {
        fillColor = m_dPtr->controlPointSelectedColor;
    } else {
        fillColor = m_dPtr->controlPointNormalColor;
    }

    // Warstwa tła (cień)
    QPen shadowPen(Qt::black, 1.0);
    QBrush shadowBrush(QColor(0, 0, 0, 50));
    m_dPtr->painter->setPen(shadowPen);
    m_dPtr->painter->setBrush(shadowBrush);
    m_dPtr->painter->drawEllipse(position + QPointF(1, 1), m_dPtr->controlPointSize + 1, m_dPtr->controlPointSize + 1);

    // Warstwa główna (punkt kontrolny)
    QPen pen(Qt::black, 1.0);
    QBrush brush(fillColor);
    m_dPtr->painter->setPen(pen);
    m_dPtr->painter->setBrush(brush);
    m_dPtr->painter->drawEllipse(position, m_dPtr->controlPointSize, m_dPtr->controlPointSize);

    // Dodaj numer punktu kontrolnego jeśli jest wybrany
    if (isSelected) {
        m_dPtr->painter->setPen(Qt::white);
        QFont font = m_dPtr->painter->font();
        font.setPointSize(9);
        m_dPtr->painter->setFont(font);

        // Obliczanie offsetu tekstu dla wyśrodkowania
        QFontMetrics fm(font);
        QString text = QString::number(isActive ? 1 : 0); // Przykładowy tekst
        QRect textRect = fm.boundingRect(text);
        QPointF textPos = position - QPointF(textRect.width() / 2.0, -textRect.height() / 4.0);

        m_dPtr->painter->drawText(textPos, text);
    }

    m_dPtr->painter->restore();
}

void QtRenderer::renderControlPoints(const std::vector<Point>& points, int selectedIndex, int activeIndex)
{
    if (!m_dPtr->painter || !m_dPtr->viewportManager || points.empty()) {
        return;
    }

    // Rysowanie poligonu kontrolnego (linie łączące punkty)
    if (m_dPtr->drawControlPolygon && points.size() >= 2) {
        renderControlPolygon(points);
    }

    // Buforowanie pozycji punktów kontrolnych na ekranie (dla wydajności)
    std::vector<QPointF> screenPositions;
    screenPositions.reserve(points.size());

    for (const auto& point : points) {
        screenPositions.push_back(
            m_dPtr->viewportManager->worldToScreen(QPointF(point.x(), point.y()))
        );
    }

    // Najpierw rysujemy zwykłe punkty kontrolne
    for (size_t i = 0; i < points.size(); ++i) {
        if (static_cast<int>(i) != selectedIndex && static_cast<int>(i) != activeIndex) {
            renderControlPoint(screenPositions[i], false, false);
        }
    }

    // Następnie rysujemy zaznaczony punkt (jeśli istnieje)
    if (selectedIndex >= 0 && selectedIndex < static_cast<int>(points.size())) {
        renderControlPoint(screenPositions[selectedIndex], true, false);
    }

    // Na końcu rysujemy aktywny punkt (jeśli istnieje i jest inny niż zaznaczony)
    if (activeIndex >= 0 && activeIndex < static_cast<int>(points.size()) && activeIndex != selectedIndex) {
        renderControlPoint(screenPositions[activeIndex], false, true);
    }

    // Rysujemy etykiety z numerami punktów
    m_dPtr->painter->save();
    m_dPtr->painter->resetTransform();
    QPen textPen(Qt::black);
    m_dPtr->painter->setPen(textPen);

    QFont font = m_dPtr->painter->font();
    font.setPointSize(8);
    m_dPtr->painter->setFont(font);

    for (size_t i = 0; i < points.size(); ++i) {
        QString label = QString::number(i);
        QFontMetrics fm(font);
        QRect textRect = fm.boundingRect(label);

        QPointF textPos = screenPositions[i] + QPointF(m_dPtr->controlPointSize + 2,
                                                      m_dPtr->controlPointSize / 2.0);

        // Tło dla tekstu dla lepszej czytelności
        QRectF bgRect = QRectF(textPos.x() - 1, textPos.y() - textRect.height(),
                              textRect.width() + 2, textRect.height() + 2);
        m_dPtr->painter->fillRect(bgRect, QColor(255, 255, 255, 180));

        m_dPtr->painter->drawText(textPos, label);
    }

    m_dPtr->painter->restore();
}

void QtRenderer::renderControlPolygon(const std::vector<Point>& points)
{
    if (!m_dPtr->painter || !m_dPtr->viewportManager || points.size() < 2) {
        return;
    }

    m_dPtr->painter->save();
    m_dPtr->painter->resetTransform();

    QPen pen(m_dPtr->controlPolygonColor, m_dPtr->controlPolygonWidth, m_dPtr->controlPolygonStyle);
    m_dPtr->painter->setPen(pen);
    m_dPtr->painter->setBrush(Qt::NoBrush);

    QPolygonF polygon;
    for (const auto& point : points) {
        QPointF screenPos = m_dPtr->viewportManager->worldToScreen(QPointF(point.x(), point.y()));
        polygon << screenPos;
    }

    m_dPtr->painter->drawPolyline(polygon);

    m_dPtr->painter->restore();
}

// Stałe domyślne
constexpr double DEFAULT_MIN_ZOOM = 0.1;
constexpr double DEFAULT_MAX_ZOOM = 100.0;
constexpr double DEFAULT_ZOOM_FACTOR = 1.0;
constexpr QPointF DEFAULT_PAN_OFFSET(0.0, 0.0);
constexpr QRectF DEFAULT_VIEWPORT(-10.0, -10.0, 20.0, 20.0);

ViewportManager::ViewportManager()
    : m_viewport(DEFAULT_VIEWPORT)
    , m_windowWidth(800)
    , m_windowHeight(600)
    , m_zoomFactor(DEFAULT_ZOOM_FACTOR)
    , m_panOffset(DEFAULT_PAN_OFFSET)
    , m_minZoom(DEFAULT_MIN_ZOOM)
    , m_maxZoom(DEFAULT_MAX_ZOOM)
{
    updateTransformations();
}

ViewportManager::~ViewportManager() = default;

void ViewportManager::pan(const QPointF& delta)
{
    // Obliczenie przesunięcia w koordynatach świata
    QPointF worldDelta = screenToWorld(delta) - screenToWorld(QPointF(0, 0));

    // // Ponieważ oś Y nie jest odwrócona, odwracamy składową Y
    // worldDelta.setY(-worldDelta.y());

    // Aktualizacja przesunięcia
    m_panOffset -= worldDelta;

    updateTransformations();
}

void ViewportManager::zoom(double factor, const QPointF& center)
{
    // Sprawdzenie ograniczeń zoomowania
    double newZoom = m_zoomFactor * factor;
    if (newZoom < m_minZoom) {
        factor = m_minZoom / m_zoomFactor;
        newZoom = m_minZoom;
    }
    else if (newZoom > m_maxZoom) {
        factor = m_maxZoom / m_zoomFactor;
        newZoom = m_maxZoom;
    }

    // Jeśli nie ma zmiany, zakończ
    if (std::abs(factor - 1.0) < 1e-6) {
        return;
    }

    // Pobierz punkt środkowy w koordynatach świata
    QPointF worldCenter = screenToWorld(center);

    // Aktualizacja współczynnika zoomowania
    m_zoomFactor = newZoom;

    // Korekta przesunięcia, aby punkt centralny pozostał w miejscu
    QPointF newWorldCenter = screenToWorld(center);
    m_panOffset += worldCenter - newWorldCenter;

    updateTransformations();
}

void ViewportManager::resetView()
{
    // Przywrócenie domyślnych wartości
    m_zoomFactor = DEFAULT_ZOOM_FACTOR;
    m_panOffset = DEFAULT_PAN_OFFSET;
    m_viewport = DEFAULT_VIEWPORT;

    updateTransformations();
}

void ViewportManager::fitView(const QRectF& rect)
{
    if (rect.isEmpty()) {
        return;
    }

    // Obliczenie współczynnika skalowania
    double scaleX = m_windowWidth / rect.width();
    double scaleY = m_windowHeight / rect.height();

    // Użycie mniejszego współczynnika z marginesem 15%
    double newZoom = std::min(scaleX, scaleY) * 0.85;

    // Ograniczenie zoomu do dozwolonych wartości
    newZoom = std::max(m_minZoom, std::min(m_maxZoom, newZoom));

    m_zoomFactor = newZoom;
    m_panOffset = QPointF(rect.center().x(), rect.center().y());

    updateTransformations();
}

void ViewportManager::zoomIn(const QPointF& center)
{
    zoom(1.2, center); // Zbliżenie o 20%
}

void ViewportManager::zoomOut(const QPointF& center)
{
    zoom(0.8, center); // Oddalenie o 20%
}

void ViewportManager::panBy(double dx, double dy)

    pan(QPointF(dx, dy));
}

void ViewportManager::setMinZoom(double minZoom)
{
    m_minZoom = minZoom > 0 ? minZoom : DEFAULT_MIN_ZOOM;
    if (m_zoomFactor < m_minZoom) {
        m_zoomFactor = m_minZoom;
        updateTransformations();
    }
}

void ViewportManager::setMaxZoom(double maxZoom)
{
    m_maxZoom = maxZoom > m_minZoom ? maxZoom : DEFAULT_MAX_ZOOM;
    if (m_zoomFactor > m_maxZoom) {
        m_zoomFactor = m_maxZoom;
        updateTransformations();
    }
}

void ViewportManager::setZoomConstraints(double minZoom, double maxZoom)
{
    m_minZoom = minZoom > 0 ? minZoom : DEFAULT_MIN_ZOOM;
    m_maxZoom = maxZoom > m_minZoom ? maxZoom : DEFAULT_MAX_ZOOM;

    // Dostosowanie aktualnego zoomu do nowych ograniczeń
    if (m_zoomFactor < m_minZoom) m_zoomFactor = m_minZoom;
    else if (m_zoomFactor > m_maxZoom) m_zoomFactor = m_maxZoom;
    updateTransformations();
}

void ViewportManager::updateTransformations()
{
    m_viewTransform.reset();
    // Przesunięcie początku układu współrzędnych do środka okna
    m_viewTransform.translate(m_windowWidth / 2.0, m_windowHeight / 2.0);

    m_viewTransform.scale(m_zoomFactor, -m_zoomFactor);

    // 3. Zastosowanie przesunięcia widoku
    m_viewTransform.translate(-m_panOffset.x(), -m_panOffset.y());

    // Obliczenie transformacji odwrotnej
    m_inverseViewTransform = m_viewTransform.inverted();
}

// Implementacja metod operacji na macierzach
std::shared_ptr<TransformMatrix> EigenMathAdapter::createRotationMatrix(double angle, const std::array<double, 3>& axis) {
    Eigen::Vector3d eigenAxis(axis[0], axis[1], axis[2]);
    Eigen::AngleAxisd rotation(angle, eigenAxis.normalized());
    Eigen::Matrix4d matrix = Eigen::Matrix4d::Identity();
    matrix.block<3, 3>(0, 0) = rotation.toRotationMatrix();
    return std::make_shared<StandardTransformMatrix>(matrix);
}

std::shared_ptr<TransformMatrix> EigenMathAdapter::createTranslationMatrix(const std::array<double, 3>& translation) {
    Eigen::Matrix4d matrix = Eigen::Matrix4d::Identity();
    matrix(0, 3) = translation[0];
    matrix(1, 3) = translation[1];
    matrix(2, 3) = translation[2];
    return std::make_shared<StandardTransformMatrix>(matrix);
}

std::shared_ptr<TransformMatrix> EigenMathAdapter::createScalingMatrix(const std::array<double, 3>& factors) {
    Eigen::Matrix4d matrix = Eigen::Matrix4d::Identity();
    matrix(0, 0) = factors[0];
    matrix(1, 1) = factors[1];
    matrix(2, 2) = factors[2];
    return std::make_shared<StandardTransformMatrix>(matrix);
}

std::vector<double> EigenMathAdapter::solveLinearSystem(
    const std::vector<std::vector<double>>& coefficients,
    const std::vector<double>& constants) {

    // Konwersja danych wejściowych
    Eigen::MatrixXd A = toEigenMatrix(coefficients);
    Eigen::VectorXd b = Eigen::Map<const Eigen::VectorXd>(constants.data(), constants.size());

    // Rozwiązanie układu równań (Ax = b)
    Eigen::VectorXd x = A.colPivHouseholderQr().solve(b);

    // Konwersja wyniku
    return std::vector<double>(x.data(), x.data() + x.size());
}

Eigen::MatrixXd EigenMathAdapter::toEigenMatrix(const std::vector<std::vector<double>>& matrix) const {
    if (matrix.empty()) return Eigen::MatrixXd(0, 0);

    size_t rows = matrix.size();
    size_t cols = matrix[0].size();

    Eigen::MatrixXd result(rows, cols);
    for (size_t i = 0; i < rows; ++i) {
        for (size_t j = 0; j < cols && j < matrix[i].size(); ++j) {
            result(i, j) = matrix[i][j];
        }
    }
    return result;
}

#pragma once

#include <memory>
#include <vector>
#include <array>

class TransformMatrix;

class IMathProvider {
public:
    virtual ~IMathProvider() = default;

    // Podstawowe operacje arytmetyczne (inline)
    virtual double add(double a, double b) const { return a + b; }
    virtual double subtract(double a, double b) const { return a - b; }
    virtual double multiply(double a, double b) const { return a * b; }
    virtual double divide(double a, double b) const {
        if (b == 0) throw std::invalid_argument("Division by zero");
        return a / b;
    }

    // Operacje macierzowe
    virtual std::shared_ptr<TransformMatrix> createRotationMatrix(double angle, const std::array<double, 3>& axis) = 0;
    virtual std::shared_ptr<TransformMatrix> createTranslationMatrix(const std::array<double, 3>& translation) = 0;
    virtual std::shared_ptr<TransformMatrix> createScalingMatrix(const std::array<double, 3>& factors) = 0;

    // Operacje algebry liniowej
    virtual std::vector<double> solveLinearSystem(
        const std::vector<std::vector<double>>& coefficients,
        const std::vector<double>& constants) = 0;
};

    QVBoxLayout *propertiesLayout = new QVBoxLayout(propertiesWidget);
    propertiesLayout->addWidget(new QLabel("Wybierz obiekt, aby edytować jego właściwości"));

    CommandManager commandManager;

    auto updateObjectsTree = [objectsTree, canvas]() {
        objectsTree->clear();

        int index = 0;
        for (const auto& obj : canvas->objects()) {
            QTreeWidgetItem* item = new QTreeWidgetItem(objectsTree);
            item->setText(0, QString("Obiekt %1").arg(std::to_string(index + 1)));

            auto curve = std::dynamic_pointer_cast<InterpolatedCurve>(obj);
            if (curve) {
                item->setText(1, "Krzywa");

                const auto& points = curve->getPoints();
                for (size_t i = 0; i < points.size(); ++i) {
                    QTreeWidgetItem* pointItem = new QTreeWidgetItem(item);
                    const Point& p = points[i];
                    pointItem->setText(0, QString("Punkt %1").arg(std::to_string(i)));
                    pointItem->setText(1, QString("(%1, %2)")
                        .arg(QString::number(p.x(), 'f', 2), QString::number(p.y(), 'f', 2)));

                    pointItem->setData(0, Qt::UserRole, QVariant::fromValue(static_cast<int>(i)));
                    pointItem->setData(0, Qt::UserRole + 1, QVariant::fromValue(index));
                }
            } else {
                item->setText(1, "Inny");
            }

            item->setData(0, Qt::UserRole, QVariant::fromValue(index));
            index++;
        }

        objectsTree->expandAll();
    };

    auto updatePropertiesPanel = [&]() {
        QLayoutItem* child;
        while ((child = propertiesLayout->takeAt(0)) != nullptr) {
            delete child->widget();
            delete child;
        }

        auto selectedObject = canvas->selectedObject();
        if (!selectedObject) {
            propertiesLayout->addWidget(new QLabel("Wybierz obiekt, aby edytować jego właściwości"));
            return;
        }

        auto curve = std::dynamic_pointer_cast<InterpolatedCurve>(selectedObject);
        if (curve) {
            propertiesLayout->addWidget(new QLabel("<b>Właściwości krzywej interpolowanej</b>"));

            // propertiesLayout->addWidget(new QLabel(QString("Liczba punktów kontrolnych: %1")
            //     .arg(curve->getPointCount())));

                    auto type = static_cast<InterpolatedCurve::InterpolationType>(index);
                    curve->setInterpolationType(type);
                    canvas->refresh();

                    QString typeName;
                    switch (type) {
                        case InterpolatedCurve::InterpolationType::NIFS3: typeName = "NIFS3"; break;
                        case InterpolatedCurve::InterpolationType::OIFS3: typeName = "OIFS3"; break;
                        case InterpolatedCurve::InterpolationType::Lagrange: typeName = "Lagrange"; break;
                        case InterpolatedCurve::InterpolationType::Barycentric: typeName = "Barycentric"; break;
                    }

                    statusBar->showMessage(QString("Zmieniono typ interpolacji na %1").arg(typeName));
                });

            propertiesLayout->addWidget(new QLabel(QString("Długość krzywej: -"))); //%1")
                //.arg(QString::number(curve->getLength(), 'f', 2))));

            propertiesLayout->addWidget(new QLabel("Wybierz krzywą interpolowaną, aby edytować jej właściwości"));

        propertiesLayout->addStretch();

        switch (canvas->getEditMode()) {

            case Canvas::EditMode::SelectObject:
                modeLabel->setText("Tryb: Zaznaczanie obiektu");
                break;
            case Canvas::EditMode::AddPoint:
                modeLabel->setText("Tryb: Dodawanie punktu");
                break;
            case Canvas::EditMode::MovePoint:
                modeLabel->setText("Tryb: Przesuwanie punktu");
                break;
        }
    };

    QObject::connect(exitAction, &QAction::triggered, &app, &QApplication::quit);

    QObject::connect(newAction, &QAction::triggered, [&]() {
        canvas->clearObjects();

        canvas->resetView();
        updateObjectsTree();

    QObject::connect(openAction, &QAction::triggered, [&]() {
        QString filePath = QFileDialog::getOpenFileName(&mainWindow, "Otwórz projekt",

        if (!filePath.isEmpty()) {
            statusBar->showMessage("Otwieranie projektu nie jest jeszcze zaimplementowane");
        }
    });

    QObject::connect(saveAction, &QAction::triggered, [&]() {
        statusBar->showMessage("Zapisywanie projektu nie jest jeszcze zaimplementowane");

    QObject::connect(saveAsAction, &QAction::triggered, [&]() {
        QString filePath = QFileDialog::getSaveFileName(&mainWindow, "Zapisz projekt jako",
            QString(), "Pliki projektu (*.crv);;Wszystkie pliki (*.*)");
        if (!filePath.isEmpty()) {

        canvas->zoomOut();
        statusBar->showMessage("Oddalono widok");

        updateObjectsTree();
        statusBar->showMessage("Usunięto obiekt");
    });

    QObject::connect(canvas, &Canvas::objectSelected, [&](std::shared_ptr<GeometricObject> obj) {
        updateObjectsTree();
        updatePropertiesPanel();
        auto objects = canvas->objects();
        auto it = std::find(objects.begin(), objects.end(), obj);
        if (it != objects.end()) {
            int index = static_cast<int>(std::distance(objects.begin(), it));

            for (int i = 0; i < objectsTree->topLevelItemCount(); ++i) {
                QTreeWidgetItem* item = objectsTree->topLevelItem(i);
                if (item->data(0, Qt::UserRole).toInt() == index) {
                    objectsTree->setCurrentItem(item);
                    break;
                }
            }
        }

        statusBar->showMessage("Wybrano obiekt");
    });

        updateObjectsTree();
        statusBar->showMessage(QString("Przesunięto punkt kontrolny %1").arg(std::to_string(pointIndex)));

        updateObjectsTree();

        auto selectedObj = canvas->selectedObject();
        auto objects = canvas->objects();
        auto it = std::find(objects.begin(), objects.end(), selectedObj);

        if (it != objects.end()) {
            int objectIndex = static_cast<int>(std::distance(objects.begin(), it));

            for (int i = 0; i < objectsTree->topLevelItemCount(); ++i) {
                QTreeWidgetItem* objectItem = objectsTree->topLevelItem(i);
                if (objectItem->data(0, Qt::UserRole).toInt() == objectIndex) {
                    for (int j = 0; j < objectItem->childCount(); ++j) {
                        QTreeWidgetItem* pointItem = objectItem->child(j);
                        if (pointItem->data(0, Qt::UserRole).toInt() == pointIndex) {
                            objectsTree->setCurrentItem(pointItem);
                            break;
                        }
                    }
                    break;
                }
            }
        }

        statusBar->showMessage(QString("Wybrano punkt kontrolny %1").arg(std::to_string(pointIndex)));
    });

        updateModeLabel();
        statusBar->showMessage("Rozpoczęto edycję punktów kontrolnych");
    });

        selectModeAction->setChecked(true);
        updateModeLabel();
        statusBar->showMessage("Zakończono edycję punktów kontrolnych");
    });

        if (item->parent() == nullptr) {
            int index = item->data(0, Qt::UserRole).toInt();
            if (index >= 0 && index < static_cast<int>(canvas->objects().size())) {
                canvas->selectObject(canvas->objects()[index]);
                canvas->finishEditing();
                selectModeAction->setChecked(true);
                updateModeLabel();
            }
        }
        else {
            int pointIndex = item->data(0, Qt::UserRole).toInt();
            int objectIndex = item->data(0, Qt::UserRole + 1).toInt();

            if (objectIndex >= 0 && objectIndex < static_cast<int>(canvas->objects().size())) {
                canvas->selectObject(canvas->objects()[objectIndex]);

                auto curve = canvas->getSelectedCurve();
                if (curve && pointIndex >= 0 && pointIndex < static_cast<int>(curve->getPointCount())) {
                    canvas->startMovingControlPoints();
                    movePointModeAction->setChecked(true);
                    canvas->selectControlPoint(pointIndex);
                    updateModeLabel();
                }
            }
        }
    });

    QObject::connect(objectsTree, &QTreeWidget::itemDoubleClicked, [&](QTreeWidgetItem* item, int column) {
        if (item->parent() != nullptr) {
            int pointIndex = item->data(0, Qt::UserRole).toInt();
            int objectIndex = item->data(0, Qt::UserRole + 1).toInt();

            if (objectIndex >= 0 && objectIndex < static_cast<int>(canvas->objects().size())) {
                canvas->selectObject(canvas->objects()[objectIndex]);

                auto curve = canvas->getSelectedCurve();
                if (curve && pointIndex >= 0 && pointIndex < static_cast<int>(curve->getPointCount())) {
                    canvas->startMovingControlPoints();
                    movePointModeAction->setChecked(true);
                    canvas->selectControlPoint(pointIndex);
                    updateModeLabel();
                    statusBar->showMessage(QString("Edytowanie punktu kontrolnego %1. Możesz go przesunąć na canvasie.").arg(std::to_string(pointIndex)));
                }
            }
        }
    });

    updateObjectsTree();

    mainWindow.show();
    statusBar->showMessage("Aplikacja gotowa do użycia");

    return app.exec();
}
